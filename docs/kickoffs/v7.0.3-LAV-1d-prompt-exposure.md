## NODE.JS TEST

Full code, save as `test-lav-1d.mjs`, run with `node test-lav-1d.mjs`, delete before committing.

```js
// test-lav-1d.mjs — pure Node, no app imports.
import { readFileSync } from "node:fs";
let pass = 0, fail = 0;
const assert = (c, l) => { if (c) { pass++; console.log(`PASS: ${l}`); } else { fail++; console.log(`FAIL: ${l}`); } };

const exec = readFileSync("api/capabilities/execute.js", "utf8");
const hook = readFileSync("src/hooks/useHarnessStream.js", "utf8");
const screen = readFileSync("src/screens/LiveAgentViewScreen.jsx", "utf8");
const stripped = s => s.replace(/\/\/[^\n]*/g, "");

// --- emit exists, exactly once, with §19p identity and the full prompt ---
const emits = exec.match(/type: 'prompt_assembled'/g) || [];
assert(emits.length === 1, `exactly one prompt_assembled emit site (got ${emits.length})`);
const emitBlock = exec.slice(exec.indexOf("prompt_assembled") - 400, exec.indexOf("prompt_assembled") + 400);
for (const f of ["trace_id", "span_id", "system_prompt", "agentId"]) {
  assert(emitBlock.includes(f), `emit carries ${f}`);
}
assert(!/system_prompt\.slice|system_prompt\.substring|truncat/i.test(emitBlock), "prompt is not truncated at the emit");

// --- multi-trace collection mirror ---
function collectTraces(evts) {
  const seen = [];
  for (const e of evts) { const t = e.trace_id ?? e.data?.trace_id; if (t && !seen.includes(t)) seen.push(t); }
  return seen;
}
assert(JSON.stringify(collectTraces([
  { trace_id: "t1" }, { data: { trace_id: "t1" } }, { data: { trace_id: "t2" } }, { trace_id: "t3" },
])) === JSON.stringify(["t1", "t2", "t3"]), "distinct trace ids collected in arrival order");

// --- prompt state: latest wins, cleared on reset ---
function promptReducer(state, action) {
  if (action.type === "clear") return null;
  if (action.type === "prompt_assembled") return action.payload;
  return state;
}
let p = promptReducer(null, { type: "prompt_assembled", payload: { agentId: "marcus", prompt_chars: 100 } });
p = promptReducer(p, { type: "prompt_assembled", payload: { agentId: "michelle", prompt_chars: 200 } });
assert(p.agentId === "michelle", "latest prompt replaces prior");
assert(promptReducer(p, { type: "clear" }) === null, "clear resets the prompt box");

// --- fs guards ---
assert(/prompt_assembled/.test(hook), "hook handles prompt_assembled");
assert(/\.in\(/.test(screen) || /\.in\(/.test(hook), "poller uses a multi-trace IN query");
assert(!/#[0-9a-fA-F]{3,8}\b/.test(stripped(screen)), "zero literal hex in screen");
assert(!/Math\.random/.test(hook + screen), "no Math.random");
assert(!/Waiting for a request…[\s\S]{0,80}system:/.test(screen), "no fabricated placeholder prompt copy");

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS PASS: NO — fix before committing"); process.exit(1); }
console.log("ALL TESTS PASS");
```
