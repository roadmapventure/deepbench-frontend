## 1. CONTEXT

`S-APPLE-04b` (`AA-103`) built the generic mechanism that lets "accept" hand a confirmed decision to a second, execute-only intent instead of just filing it away — and used it to complete Nadia's new Data Integrity Patch feature end to end. That same session found and fixed a live safety gap on Nadia's *other*, already-shipped capability — Escalate (`data-escalate-intent`) — but deliberately left Escalate's own round trip incomplete (removed the unsafe `can_request_help`, did not add the replacement mechanism), scoping that completion to its own session per John's direction 2026-07-03 ("fix the safety gap now, complete Escalate in a follow-up session").

This session is that follow-up. It is pure retrofit — zero new harness code, reusing `AA-103` exactly as built. Escalate's decide-step (`data-escalate-intent`) already produces a correctly-shaped proposal (`new_chunks`/`summary`/`citations`); once a human accepts, this session's new execute-intent asks for help to get each chunk in `new_chunks` actually inserted into the Data Room via Eleanor — the same "decide, then delegate the execution" split `S-APPLE-04b` established for Patch, applied to Escalate's different (array-of-chunks, not single-chunk-correction) shape.

**Why this couldn't just be folded into `S-APPLE-04b`:** that session was already at 3 tasks / 2 files touching two different capabilities' safety posture; adding a second full live round-trip test (multi-chunk insert, a structurally different case from Patch's single insert+update_status) would have pushed it past the one-feature-per-session discipline. Splitting keeps each session's own Node test focused on the specific new behavior it introduces.

```js
// test-s-arch-hitl-resume-02-api.mjs — Category F, L, M tests for S-ARCH-HITL-RESUME-02
// Run: node --env-file=.env.local test-s-arch-hitl-resume-02-api.mjs

const SUPABASE_URL = process.env.SUPABASE_URL;
const SUPABASE_KEY = process.env.SUPABASE_SERVICE_KEY;
const HEADERS = { 'Content-Type': 'application/json', apikey: SUPABASE_KEY, Authorization: `Bearer ${SUPABASE_KEY}` };
const DATA_ROOM_TAG = 'apple-cso-data-room';

let pass = 0, fail = 0;
function check(label, cond) {
  if (cond) { console.log(`PASS: ${label}`); pass++; }
  else { console.log(`FAIL: ${label}`); fail++; }
}

async function fetchRow(table, filter) {
  const res = await fetch(`${SUPABASE_URL}/rest/v1/${table}?${filter}`, { headers: HEADERS });
  if (!res.ok) throw new Error(`Fetch failed for ${table}?${filter}: ${res.status}`);
  return res.json();
}

function testCategoryF() {
  console.log('\n--- Category F: environment ---');
  check('SUPABASE_URL set', !!SUPABASE_URL);
  check('SUPABASE_SERVICE_KEY set', !!SUPABASE_KEY);
  check('ANTHROPIC_API_KEY set', !!process.env.ANTHROPIC_API_KEY);
  check('OPENAI_API_KEY set', !!process.env.OPENAI_API_KEY);
}

async function testCategoryFSchema() {
  console.log('\n--- Category F: Supabase column/data alignment ---');

  const exec = await fetchRow('skill_profiles', `slug=eq.data-escalate-execute-intent&select=traits`);
  check('data-escalate-execute-intent exists', exec.length === 1);
  check('data-escalate-execute-intent can_request_help true', exec[0]?.traits?.can_request_help === true);
  check('data-escalate-execute-intent has no requires_human_confirmation', exec[0]?.traits?.requires_human_confirmation !== true);
  check('data-escalate-execute-intent schema requires chunks_written', exec[0]?.traits?.schema?.required?.includes('chunks_written'));

  const decide = await fetchRow('skill_profiles', `slug=eq.data-escalate-intent&select=traits`);
  check('data-escalate-intent now has on_accept_intent_slug', decide[0]?.traits?.on_accept_intent_slug === 'data-escalate-execute-intent');
  check('data-escalate-intent still has no can_request_help (S-APPLE-04b fix holds)', decide[0]?.traits?.can_request_help !== true);

  const links = await fetchRow('capability_skill_profiles', `capability_slug=eq.data-analysis&select=skill_profile_slug`);
  check('exactly 6 skill profiles linked to data-analysis', links.length === 6);
}

async function testCategoryL() {
  console.log('\n--- Category L: live round trip -- Escalate propose, accept, real Eleanor insert (novel path: first live multi-chunk delegation, first time Escalate ever completes a real write) ---');
  const { runCapability, resolveAccept } = await import('./api/capabilities/execute.js');
  const insertedIds = [];

  try {
    // L1 -- propose a single-chunk Escalate research finding (kept to 1 chunk deliberately --
    // see Task 1 notes on the 2-hops-per-chunk depth ceiling).
    const l1 = await runCapability({
      capability_slug: 'data-analysis',
      intent_slug: 'data-escalate-intent',
      agent_id: 'nadia',
      task_context: {
        hypothesis: 'S-ARCH-HITL-RESUME-02 live test -- confirm Escalate accept now produces a real the_library write.',
        flagged_question: 'n/a',
        research_request: 'Draft one new, clearly-marked test finding for the apple-cso-data-room confirming this mechanism works end to end. Mark it synthesized and non-citeable -- this is test content.',
      },
      tenant_id: 'global',
    });
    check('L1: call completed without error', !l1.error);
    check('L1: status is pending_confirmation', l1.status === 'pending_confirmation');
    check('L1: proposed_action has new_chunks array', Array.isArray(l1.proposed_action?.new_chunks));

    // L2 -- accept it.
    const l2 = await resolveAccept({ confirmation_id: l1.confirmation_id });
    check('L2: call completed without error', !l2.error);
    const chunksWritten = l2.content?.chunks_written ?? l2.chunks_written;
    check('L2: chunks_written reported', typeof chunksWritten === 'number' && chunksWritten >= 1);

    // Verify at least one real row landed, tagged as this test's content.
    const newRows = await fetchRow('the_library', `data_room_tag=eq.${DATA_ROOM_TAG}&source=eq.agent&data_type=eq.synthesized&citeable=eq.false&order=created_at.desc&limit=5&select=id,title,created_at`);
    check('L2: at least one new the_library row exists to verify against', newRows.length > 0);
  } finally {
    for (const id of insertedIds) {
      await fetch(`${SUPABASE_URL}/rest/v1/the_library?id=eq.${id}`, { method: 'DELETE', headers: HEADERS });
    }
  }
}

async function testCategoryM() {
  console.log('\n--- Category M: regression -- S-APPLE-04b Patch flow unaffected ---');
  const { runCapability, resolveAccept } = await import('./api/capabilities/execute.js');

  const seedRes = await fetch(`${SUPABASE_URL}/rest/v1/the_library`, {
    method: 'POST', headers: { ...HEADERS, Prefer: 'return=representation' },
    body: JSON.stringify({
      title: 'TEST S-ARCH-HITL-RESUME-02 regression seed', content: 'Placeholder content for regression check.',
      data_room_tag: DATA_ROOM_TAG, embedding: Array(1536).fill(0), status: 'active',
      citeable: false, data_type: 'synthesized', source: 'test-fixture', is_baseline: false,
    }),
  });
  const [seedRow] = await seedRes.json();
  let newRowId = null;

  try {
    const m1 = await runCapability({
      capability_slug: 'data-analysis',
      intent_slug: 'data-patch-intent',
      agent_id: 'nadia',
      task_context: {
        disputed_chunk_id: seedRow.id,
        correction: 'Regression check only -- confirm S-APPLE-04b Patch flow still completes after S-ARCH-HITL-RESUME-02.',
        director_reasoning: 'n/a',
      },
      tenant_id: 'global',
    });
    check('M1: Patch decide-step unaffected, still reaches pending_confirmation', m1.status === 'pending_confirmation');

    if (m1.status === 'pending_confirmation' && m1.proposed_action?.action !== 'no_action') {
      const m1b = await resolveAccept({ confirmation_id: m1.confirmation_id });
      check('M1b: Patch accept still delegates to data-patch-execute-intent (unaffected by this session)', !m1b.error);
      const afterRow = await fetchRow('the_library', `id=eq.${seedRow.id}&select=status`);
      check('M1b: original chunk correctly superseded (Patch flow byte-identical to S-APPLE-04b)', afterRow[0]?.status === 'superseded');
      const newRows = await fetchRow('the_library', `supersedes_id=eq.${seedRow.id}&select=id`);
      newRowId = newRows[0]?.id || null;
    }
  } finally {
    if (newRowId) await fetch(`${SUPABASE_URL}/rest/v1/the_library?id=eq.${newRowId}`, { method: 'DELETE', headers: HEADERS });
    await fetch(`${SUPABASE_URL}/rest/v1/the_library?id=eq.${seedRow.id}`, { method: 'DELETE', headers: HEADERS });
  }
}

(async () => {
  try {
    testCategoryF();
    await testCategoryFSchema();
    await testCategoryL();

  console.log(fail === 0 ? 'ALL TESTS PASS' : 'TESTS FAILED — do not commit');
  process.exit(fail === 0 ? 0 : 1);
})();
```
