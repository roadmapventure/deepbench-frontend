// test-lav-1f.mjs — pure Node, no app imports.
import { readFileSync } from "node:fs";
let pass = 0, fail = 0;
const assert = (c, l) => { if (c) { pass++; console.log(`PASS: ${l}`); } else { fail++; console.log(`FAIL: ${l}`); } };

const hook = readFileSync("src/hooks/useHarnessStream.js", "utf8");
const screen = readFileSync("src/screens/LiveAgentViewScreen.jsx", "utf8");
const net = readFileSync("src/components/AgentNetwork.jsx", "utf8");
const stripped = s => s.replace(/\/\/[^\n]*/g, "");

// --- pending-state reducer mirror ---
function pendingReducer(state, frame) {
  if (frame.type === "result" && frame.result?.status === "pending_confirmation")
    return { confirmation_id: frame.result.confirmation_id, agentId: frame.result.agent_id ?? null };
  if (frame.type === "resolve_dispatched" || frame.type === "clear") return null;
  return state;
}
let st = pendingReducer(null, { type: "result", result: { status: "pending_confirmation", confirmation_id: "c1", agent_id: "nadia" } });
assert(st?.confirmation_id === "c1", "pending_confirmation frame sets pending state with the real confirmation_id");
assert(pendingReducer(st, { type: "resolve_dispatched" }) === null, "dispatching the resolve clears pending");
assert(pendingReducer(null, { type: "result", result: { status: "final_delegation" } }) === null, "ordinary terminal never creates a pending state");

// --- You node existence is purely pending-driven ---
function youNodeVisible(pending) { return !!pending; }
assert(youNodeVisible({ confirmation_id: "c1" }) === true, "You node exists while pending");
assert(youNodeVisible(null) === false, "You node never renders without a real gate");

// --- continuation joins the same run ---
function mergeContinuation(traceIds, contEvents) {
  const out = [...traceIds];
  for (const e of contEvents) { const t = e.trace_id; if (t && !out.includes(t)) out.push(t); }
  return out;
}
assert(JSON.stringify(mergeContinuation(["t1"], [{ trace_id: "t1" }, { trace_id: "t2" }])) === JSON.stringify(["t1", "t2"]),
  "resolve continuation's traces join the run's trace set");

// --- fs guards ---
assert(/pending_confirmation/.test(hook), "hook handles pending_confirmation");
assert(/UserAvatar/.test(net), "You node uses UserAvatar");
assert(!/AgentAvatar\s*who=["']you/i.test(net), "You is never an AgentAvatar");
assert(/FOCUS_AREA_STATUS\.liveAgentView/.test(screen), "title status reads the token key");
assert(!/setTimeout\([^)]*resolve/i.test(stripped(hook + net + screen)), "no auto-resolve timer");
assert(!/#[0-9a-fA-F]{3,8}\b/.test(stripped(screen + net)), "zero literal hex");
assert(!/LAV-1d will extend/.test(hook), "stale placeholder comment updated");

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS PASS: NO — fix before committing"); process.exit(1); }
console.log("ALL TESTS PASS");
