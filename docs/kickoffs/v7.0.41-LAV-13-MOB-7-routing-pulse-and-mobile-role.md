### Task 2 — Mobile node: widen to 84px, add a one-line role
In `AgentNetwork.jsx`.

**CSS** (`NET_CSS`, the mobile block):
- `.lav-mnode` — `width:56px` → **`width:84px`**. Everything else on that rule unchanged.
- Add, immediately after `.lav-mname`:
```css
.lav-mrole{font-family:${mono};font-size:7px;color:${T.muted};line-height:1.15;margin-top:1px;
  white-space:nowrap;overflow:hidden;text-overflow:ellipsis}
```

**JSX** (the mobile node, after the `.lav-mname` div):
```jsx
<div className="lav-mrole" title={a.role}>{a.role}</div>
```

**Order is name-then-role, matching desktop's own card after `LAV-12`** — not role-above-name where the code used to sit. Same vertical order on both breakpoints.

**One line, with an ellipsis, is the specified behaviour — not a compromise to design around.** 84px at 7px mono is ~18 characters, so 4 of the 11 `mi` roles truncate ("Screen Controls Editor", "The Intake Assistant", "AI Prompt Strategist", and "Forecast/Theory/Performance Expert" at 34 chars). The full string is always one long-press away in `title`, the same affordance `.lav-pill` and `.lav-model` already use. **Do not** add a second line, shrink the font to force a fit, wrap, or abbreviate any role string — `agents.js` is canonical agent data and is out of scope.

**Why 84 and not more.** Derived against the real measured stage (402×461 at iPhone 16 Pro), not guessed. With a role line the node is ~62px tall. Ring slots 1↔2, 5↔6, 6↔7 and 10↔1 are only ~31px apart vertically — less than the node height — so they overlap on that axis and are kept apart purely by their ~90px horizontal separation. 84px leaves a 6px gutter; anything wider collides. The other adjacent pairs are ~81–100px apart vertically and cannot overlap at any width. Edge clearance also holds: leftmost slot centre is 56px in (left edge at 14px), rightmost is 346px (right edge at 388px of 402). **`mobileSlot()` itself is not touched** — this is a width change only.

```js
// S-LAV-13/MOB-7 inline test, executed against the REAL sources.
import { readFileSync } from "node:fs";
const SCREEN = "src/screens/LiveAgentViewScreen.jsx";
const NET    = "src/components/AgentNetwork.jsx";
const screen = readFileSync(SCREEN, "utf8");
const net    = readFileSync(NET, "utf8");
let pass = true;
const check = (name, cond, extra = "") => {
  if (!cond) { pass = false; console.error(`FAIL ${name} ${extra}`); } else console.log(`ok ${name}`);
};

// ---- Task 1: the pulsing-mode set, lifted and executed from source -------------------------
const m = screen.match(/const PULSING_MODES\s*=\s*new Set\(\[[^\]]*\]\)/);
check("PULSING_MODES exists in source", !!m);
if (m) {
  const PULSING_MODES = new Function(`${m[0]}; return PULSING_MODES;`)();
  check("route pulses (the fix)",  PULSING_MODES.has("route"));
  check("orch still pulses",       PULSING_MODES.has("orch"));
  check("complete still pulses",   PULSING_MODES.has("complete"));
  check("awaiting stays STEADY",  !PULSING_MODES.has("awaiting"), "LAV-1f: harness is stopped");
  check("idle stays steady",      !PULSING_MODES.has("idle"));
  check("error stays steady",     !PULSING_MODES.has("error"));
}
check("badge reads the set, not an inline mode test",
  /animation:\s*PULSING_MODES\.has\(mode\)/.test(screen));

// ---- Task 2: mobile node width + role line, read from source -------------------------------
const w = net.match(/\.lav-mnode\{[^}]*width:(\d+)px/);
check("mobile node width found", !!w);
const WIDTH = w ? Number(w[1]) : 0;
check("mobile node widened to 84px", WIDTH === 84, `got ${WIDTH}`);
check(".lav-mrole rule exists", /\.lav-mrole\{/.test(net));
check("role is one line (nowrap + ellipsis)",
  /\.lav-mrole\{[^}]*white-space:nowrap[^}]*\}/.test(net) &&
  /\.lav-mrole\{[^}]*text-overflow:ellipsis[^}]*\}/.test(net));
check("role renders a.role with a title attribute",
  /className="lav-mrole"\s+title=\{a\.role\}>\{a\.role\}/.test(net));
check("name still renders before role",
  net.indexOf('className="lav-mname"') < net.indexOf('className="lav-mrole"'));
check("agent code was not re-added", !/lav-mcode/.test(net));
check("no literal hex in the new rule", !/\.lav-mrole\{[^}]*#[0-9a-fA-F]{3,8}/.test(net));

// ---- Task 2: the widened box must still not overlap at any ring slot -----------------------
// mobileSlot lifted from source -- if the geometry changed, this fails loudly.
const ms = net.match(/export function mobileSlot\s*\([\s\S]*?\n\}/);
check("mobileSlot found", !!ms);
if (ms) {
  const mobileSlot = new Function(`${ms[0].replace(/^export\s+/, "")}; return mobileSlot;`)();
  const N = 11, STAGE_W = 402, STAGE_H = 461, NODE_H = 62;   // measured live at iPhone 16 Pro
  const slots = Array.from({ length: N }, (_, i) => mobileSlot(i, N));
  for (let i = 0; i < N; i++) for (let j = i + 1; j < N; j++) {
    const dx = Math.abs(slots[i].x - slots[j].x) / 100 * STAGE_W;
    const dy = Math.abs(slots[i].y - slots[j].y) / 100 * STAGE_H;
    check(`slots ${i}/${j} clear at ${WIDTH}px`, dx >= WIDTH || dy >= NODE_H,
      `dx=${dx.toFixed(0)} dy=${dy.toFixed(0)}`);
  }
  for (let i = 0; i < N; i++) {
    const cx = slots[i].x / 100 * STAGE_W;
    check(`slot ${i} not clipped horizontally`,
      cx - WIDTH / 2 >= 0 && cx + WIDTH / 2 <= STAGE_W, `cx=${cx.toFixed(0)}`);
  }
}
console.log(pass ? "ALL TESTS PASS" : "TESTS FAILED");
if (!pass) process.exit(1);
```
