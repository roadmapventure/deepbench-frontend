## 2. CONTEXT

Three separate John findings, all about how agent cards behave in the queue/bench and on the canvas — distinct from `LAV-9a`'s edge/line work, which this session does not touch.

1. **Delegation targets don't "come forward."** A node only leaves the bench (queue) once it has produced its own credited hop (`deriveNetwork`'s `engaged` array only grows on `h.agentId`). But the line to it is drawn the moment it's merely *named* as a delegation target (`sessionEdges`, unconditional on `secondaryAgentId`). So a targeted agent can sit shrunk in the queue with a live line pointing at it — during the run it reads as "solid line to the queue, agent not displaying"; after the run it settles to "dashed line to the queue" (same root cause, since the node never left). John confirmed the fix — treat "named as a target" the same as "spoke" for the purpose of joining the canvas — and confirmed it's fine for the separate `AGENTS ENGAGED` counter (`LiveAgentViewScreen.jsx`, its own independent count) to keep meaning "actually produced a credited event," so the two numbers are allowed to disagree.
2. **The black activity bubble sticks after an agent's own turn ends.** Its visibility is gated on the whole run being live (`bubble = running ? net.bubbles[a.id] : null`), not on whether *that agent* is currently pulsing — so it keeps showing a finished agent's last line for the rest of the run, even after someone else is now active.
3. **Card spacing is tight enough that the communication lines are hard to trace**, per John's direct ask: retune the layout constants for more breathing room, verified live — not a new interactive control (confirmed with John: a tuning pass, not a UI feature).

```js
// test-lav-9b.mjs -- LAV-9b Category B logic tests (pure, no app imports)
let pass = 0, fail = 0;
function assertEq(actual, expected, msg) {
  const ok = JSON.stringify(actual) === JSON.stringify(expected);
  if (ok) { pass++; } else { fail++; console.error(`FAIL: ${msg} -- got ${JSON.stringify(actual)}, expected ${JSON.stringify(expected)}`); }
}

// Mirrors deriveNetwork()'s engaged-tracking loop, restricted to the one branch this session changes.
function deriveEngaged(runHops) {
  const engaged = [];
  for (const h of runHops) {
    const id = h.agentId;
    if (id && !engaged.includes(id)) engaged.push(id);
    if (h.type === "delegation" && id && h.secondaryAgentId) {
      if (!engaged.includes(h.secondaryAgentId)) engaged.push(h.secondaryAgentId);
    }
  }
  return engaged;
}

// A: a delegation target that never produces its own hop is still engaged (the fix)
const runA = [
  { agentId: "michelle", type: "delegation", secondaryAgentId: "marcus" },
];
assertEq(deriveEngaged(runA), ["michelle", "marcus"], "A: delegation target promoted without its own hop");

// B: an agent that DOES speak still only appears once (no duplicate on later delegation to it)
const runB = [
  { agentId: "michelle", type: "delegation", secondaryAgentId: "marcus" },
  { agentId: "marcus", type: "delegation_complete" },
  { agentId: "michelle", type: "delegation", secondaryAgentId: "marcus" }, // re-dispatch to the same target
];
assertEq(deriveEngaged(runB), ["michelle", "marcus"], "B: no duplicate entry on repeated targeting");

// C: bubble-pulsing gate -- only orch/assembly/active count, is-done and is-benched do not
function isPulsing({ isOrch, isAssembly, isActive, running }) {
  return isOrch || (running && (isAssembly || isActive));
}
assertEq(isPulsing({ isOrch: false, isAssembly: false, isActive: true, running: true }), true, "C: active while running pulses");
assertEq(isPulsing({ isOrch: false, isAssembly: false, isActive: true, running: false }), false, "C: run over -- no longer pulsing even if 'active' flag stale");
assertEq(isPulsing({ isOrch: true, isAssembly: false, isActive: false, running: false }), true, "C: orchestrator pulses regardless of the running flag (matches existing is-orch class logic)");
assertEq(isPulsing({ isOrch: false, isAssembly: false, isActive: false, running: true }), false, "C: is-done agent does not pulse");

// D: bench-stack step formula -- capped at 58, never exceeds available span
function benchStep(m) { return m > 1 ? Math.min(58, 520 / (m - 1)) : 0; }
assertEq(benchStep(1), 0, "D: single benched agent has no step");
assertEq(benchStep(5), 58, "D: few agents -- hits the 58px cap");
assertEq(benchStep(20), 520 / 19, "D: many agents -- falls back to the span-derived step, under the cap");

console.log(fail === 0 ? "ALL TESTS PASS" : `${fail} FAILED, ${pass} PASSED`);
if (fail > 0) process.exit(1);
```
