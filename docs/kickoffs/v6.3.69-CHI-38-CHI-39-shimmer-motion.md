## CONTEXT

Two related UI findings from the 2026-07-19 screenshot walkthrough (`design-chi-ux-0719`), both about the Channel Intelligence screen (`MarketIntelligenceScreen.jsx`) feeling static/lifeless at moments where something is either loading or sitting idle:

- **`CHI-38`:** On initial page load, Column 2's 3 news cards pop in with no loading affordance — `EvidenceColumn`'s true-loading branch (`newsCards === null`) currently shows only a text line (`NewsCardsLoadingLine`, a live elapsed-time caption), no visual shape.
- **`CHI-39`:** The 3 example-question buttons (`CHI-30`/`CHI-32`'s rotation mechanism) only re-roll on Clear or a fresh load — a page sitting idle on the welcome state looks static, doesn't read as "alive."

Bundled into one kickoff (John's explicit call, same pattern as `S-CHI-34-design`'s 5-item bundle) because both are the same underlying visual primitive: a shimmer sweep. DeepBench already has this exact primitive — `tokens.js`'s `shimmer` keyframe, already used once by `TaskInstructionsScreen.jsx`'s step-loading progress bar (`background: linear-gradient(90deg,transparent 0%,${T.brass} 50%,transparent 100%), backgroundSize:"200% 100%", animation:"shimmer 1.4s linear infinite"`). No new keyframe needed — both tasks below reuse it via one new shared local component.

**CHI-39 scope, confirmed live with John (this session):** all 4 visible blocks in the idle question area shimmer together every 10 seconds — the 2 rotating question buttons (slots 1 and 3), the static library-catalog question (slot 2), and the "Browse 20 more example questions" drawer toggle (slot 4) — even though only slots 1 and 3 actually change their text. Shimmering all 4 in sync is deliberate: it reads as one consistent "alive" unit instead of visually singling out the two that change.

**Architect Review finding, also confirmed with John:** the drawer toggle (slot 4) renders via `<Drawer>`, a component from `SharedUI.jsx` reused 5 other times in this same file (Agents, Data Sources, Analysis, Agent Reasoning, Agent Routing drawers). The shimmer must NOT be built into `Drawer` itself — that would make all 6 drawers on this screen shimmer every 10 seconds. Task 3 below wraps only this one `<Drawer>` call site in a local overlay instead of touching `Drawer.jsx`.

### Task 2 — Idle auto-rotation timer (CHI-39)

In `InteractColumn` (after the existing re-roll `useEffect`, L2326–2334), add:

```jsx
// FEATURE: CHI-39 — idle auto-rotation: every 10s while the welcome/idle state is showing
// (messages.length === 0), re-roll rotation.picks via the exact same splitRotation() call Clear
// already uses (L2331). STATIC_QUESTION (slot 2) is never part of picks/leftover — unchanged,
// never rotates. Interval is armed only while idle and torn down the instant that stops being
// true (messages.length leaves 0) or the component unmounts — no orphaned timers.
const [isShimmering, setIsShimmering] = useState(false);
useEffect(() => {
  if (messages.length !== 0) return;
  const shimmerTimeouts = [];
  const interval = setInterval(() => {
    setRotation(splitRotation(ROTATING_POOL, shuffleArray));
    setIsShimmering(true);
    shimmerTimeouts.push(setTimeout(() => setIsShimmering(false), 700));
  }, 10000);
  return () => {
    clearInterval(interval);
    shimmerTimeouts.forEach(clearTimeout);
  };
}, [messages.length]);
```

`isShimmering` is the single shared trigger Task 3 reads to animate all 4 blocks in sync — it says nothing about *which* blocks changed text, only that a rotation tick just happened.

```js
// test-chi-38-39.mjs — pure Node, no app imports. Reimplements the exact pure logic this session
// ships so it can be verified in isolation (React hook behavior itself isn't unit-testable outside
// a DOM, so the effect's *decision logic* is extracted into plain functions here, matching the
// shapes Task 2/3 actually implement).

let pass = 0, fail = 0;
function assert(cond, label) {
  if (cond) { pass++; console.log(`PASS: ${label}`); }
  else { fail++; console.log(`FAIL: ${label}`); }
}

// --- Mirrors of the real pure functions (STATIC_QUESTION/ROTATING_POOL/splitRotation unchanged
// by this session — re-declared here only so the test file has zero app imports) ---
const STATIC_QUESTION = { id: "library-catalog", label: "What data is in the library and how can i use it?" };
const ROTATING_POOL = Array.from({ length: 10 }, (_, i) => ({ id: `q${i}`, label: `question ${i}` }));
const FIXED_DRAWER_TAIL = Array.from({ length: 12 }, (_, i) => ({ id: `tail${i}`, label: `tail ${i}` }));
function splitRotation(pool, shuffleFn) {
  const shuffled = shuffleFn(pool);
  return { picks: shuffled.slice(0, 2), leftover: shuffled.slice(2) };
}
function shuffleArray(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

// --- Category A/B: data shape + logic, this session's new call frequency ---
// Slot 2 (STATIC_QUESTION) must never come from picks/leftover, across many re-rolls (Task 2 calls
// splitRotation() far more often now — every 10s while idle — so this guards against a regression
// where STATIC_QUESTION accidentally leaks into the rotating pool under repeated re-rolls).
{
  let leaked = false;
  for (let i = 0; i < 500; i++) {
    const { picks, leftover } = splitRotation(ROTATING_POOL, shuffleArray);
    const all = [...picks, ...leftover];
    if (all.some(q => q.id === STATIC_QUESTION.id)) leaked = true;
    const visibleQuestions = [picks[0], STATIC_QUESTION, picks[1]];
    if (visibleQuestions[1].id !== STATIC_QUESTION.id) leaked = true;
  }
  assert(!leaked, "STATIC_QUESTION (slot 2) never enters picks/leftover across 500 re-rolls, always visibleQuestions[1]");
}

// Drawer count invariant — John's own reference point ("the 20"): must stay exactly 20 no matter
// how many times Task 2's 10s timer re-rolls picks/leftover.
{
  let allTwenty = true;
  for (let i = 0; i < 500; i++) {
    const { leftover } = splitRotation(ROTATING_POOL, shuffleArray);
    const drawerQuestions = [...leftover, ...FIXED_DRAWER_TAIL];
    if (drawerQuestions.length !== 20) allTwenty = false;
  }
  assert(allTwenty, "drawerQuestions.length stays exactly 20 across 500 re-rolls (8 leftover + 12 fixed tail)");
}

// --- Category D: component lifecycle (useEffect arm/disarm logic) ---
// Mirrors Task 2's effect: interval should be armed only while messages.length === 0, and torn
// down (not accumulated) on every dependency change, including repeated 0->1->0 transitions.
function simulateIdleRotationLifecycle(messageLengthSequence) {
  let activeIntervals = 0;
  let armedCount = 0;
  let prevArmed = false;
  for (const len of messageLengthSequence) {
    const shouldBeArmed = len === 0;
    if (prevArmed) { activeIntervals--; } // cleanup from previous effect run, mirrors the real `return () => clearInterval(interval)`
    if (shouldBeArmed) { activeIntervals++; armedCount++; }
    prevArmed = shouldBeArmed;
  }
  return { activeIntervals, armedCount };
}

{
  const { activeIntervals } = simulateIdleRotationLifecycle([0]);
  assert(activeIntervals === 1, "interval arms when idle (messages.length === 0)");
}
{
  const { activeIntervals } = simulateIdleRotationLifecycle([0, 1]);
  assert(activeIntervals === 0, "interval disarms the instant messages.length leaves 0");
}
{
  // 0 -> 1 -> 0 -> 1 -> 0: at every point exactly 0 or 1 interval is live, never 2+ (no leak)
  const seq = [0, 1, 0, 1, 0];
  let maxObserved = 0, activeIntervals = 0, prevArmed = false;
  for (const len of seq) {
    const shouldBeArmed = len === 0;
    if (prevArmed) activeIntervals--;
    if (shouldBeArmed) activeIntervals++;
    prevArmed = shouldBeArmed;
    maxObserved = Math.max(maxObserved, activeIntervals);
  }
  assert(maxObserved <= 1, "repeated idle<->active transitions never accumulate more than 1 live interval");
  assert(activeIntervals === 1, "final state (idle) leaves exactly 1 armed interval, matching messages.length===0");
}

// --- Category B: shimmer trigger timing (Task 2's setTimeout(700) pairing with setInterval(10000)) ---
{
  // 700ms shimmer window must be well inside the 10000ms tick interval — no overlap risk between
  // consecutive ticks (a design invariant Task 2/3 depend on: isShimmering is always false again
  // long before the next tick fires).
  const SHIMMER_MS = 700, TICK_MS = 10000;
  assert(SHIMMER_MS < TICK_MS, "shimmer duration (700ms) is shorter than the rotation interval (10000ms) — no overlap between ticks");
}

// --- Category C: string safety — STATIC_QUESTION.id used as a stable React key regardless of pick content ---
{
  const { picks } = splitRotation(ROTATING_POOL, shuffleArray);
  const visibleQuestions = [picks[0], STATIC_QUESTION, picks[1]];
  const idsSafe = visibleQuestions.every(q => typeof q?.id === "string" && q.id.length > 0);
  assert(idsSafe, "every visibleQuestions[i].id is a non-empty string (safe React key), including after a re-roll");
}

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS PASS: NO — fix before committing"); process.exit(1); }
console.log("ALL TESTS PASS");
```
