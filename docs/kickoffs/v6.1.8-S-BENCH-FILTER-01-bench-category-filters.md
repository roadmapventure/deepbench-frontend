## 2. CONTEXT

The Bench screen (`/bench`, `RosterScreen.jsx`) currently shows all 21 agents in one undifferentiated 3-column grid, sorted by usage count (`RO-09`). As the roster has grown past the original 7-agent NIGP team to include Market Intelligence loop agents, platform-wide broker/utility agents, and a couple of not-yet-wired personas, there's no way to see just one slice of the bench.

This session adds a left-nav category filter — **All / Market Intel / Platform Wide / NIGP Experts / Special Interests** — reusing the exact Left Sidebar Nav pattern already locked in `STYLE-GUIDE.md` §10 and implemented in `PersonnelScreen.jsx`. Agents can belong to more than one group (e.g. Michelle Manning is both Market Intel and Platform Wide) and show up in every group they belong to. "All" is the default view and is not a stored tag — it is simply the unfiltered roster.

Mock reviewed and approved by John (interactive HTML mock, filter click-through + sort behavior demoed) before this kickoff doc was written.

```js
  { id: "marcus",   name: "Marcus Webb",      benchGroups: ["mi"] },
  { id: "priya",    name: "Priya Nair",       benchGroups: ["mi"] },
  { id: "nadia",    name: "Nadia Farouk",     benchGroups: ["mi"] },
  { id: "owen",     name: "Owen Marsh",       benchGroups: ["mi"] },
  { id: "sam",      name: "Sam Reyes",        benchGroups: ["mi"] },
  { id: "elena",    name: "Elena Cho",        benchGroups: ["mi"] },
];

const BENCH_FILTERS = [
  { id: "mi",       label: "Market Intel"      },
  { id: "platform", label: "Platform Wide"     },
  { id: "nigp",     label: "NIGP Experts"      },
  { id: "special",  label: "Special Interests" },
];

function filterAgents(agents, activeFilter) {
  if (activeFilter === "all") return agents.slice();
  return agents.filter(a => a.benchGroups.includes(activeFilter));
}

function countFor(agents, filterId) {
  if (filterId === "all") return agents.length;
  return agents.filter(a => a.benchGroups.includes(filterId)).length;
}

let pass = 0, fail = 0;
function check(label, cond) {
  if (cond) { pass++; }
  else { fail++; console.error("FAIL:", label); }
}

// Category A — Data Shape: every agent has a non-empty benchGroups array of known ids
const knownIds = new Set(BENCH_FILTERS.map(f => f.id));
for (const a of AGENTS) {
  check(`${a.id} has non-empty benchGroups`, Array.isArray(a.benchGroups) && a.benchGroups.length > 0);
  for (const g of a.benchGroups) {
    check(`${a.id}'s group "${g}" is a known BENCH_FILTERS id`, knownIds.has(g));
  }
}

// Category M — Cross-Reference Consistency: every BENCH_FILTERS id has >=1 agent
for (const f of BENCH_FILTERS) {
  const n = AGENTS.filter(a => a.benchGroups.includes(f.id)).length;
  check(`BENCH_FILTERS id "${f.id}" has at least one agent`, n > 0);
}
// No stray group id used anywhere that isn't in BENCH_FILTERS (redundant with above per-agent check, kept as an explicit cross-check)
const usedIds = new Set(AGENTS.flatMap(a => a.benchGroups));
for (const id of usedIds) {
  check(`used group id "${id}" exists in BENCH_FILTERS`, knownIds.has(id));
}

// Category B — Logic: filter counts match expected group membership
check("all filter returns 21 agents", filterAgents(AGENTS, "all").length === 21);
check("mi filter returns 10 agents", filterAgents(AGENTS, "mi").length === 10);
check("platform filter returns 8 agents", filterAgents(AGENTS, "platform").length === 8);
check("nigp filter returns 5 agents", filterAgents(AGENTS, "nigp").length === 5);
check("special filter returns 2 agents", filterAgents(AGENTS, "special").length === 2);

// Multi-group agents appear in every group they belong to
check("michelle appears in mi filter", filterAgents(AGENTS, "mi").some(a => a.id === "michelle"));
check("michelle appears in platform filter", filterAgents(AGENTS, "platform").some(a => a.id === "michelle"));
check("michelle does NOT appear in nigp filter", !filterAgents(AGENTS, "nigp").some(a => a.id === "michelle"));

// Count badges match filtered lengths exactly (badge must reflect group membership, not current view)
for (const f of BENCH_FILTERS) {
  check(`countFor("${f.id}") matches filterAgents length`, countFor(AGENTS, f.id) === filterAgents(AGENTS, f.id).length);
}
check(`countFor("all") matches full roster`, countFor(AGENTS, "all") === 21);

// Empty-group defensive path (no real group is empty today, but the empty-state branch must be reachable/correct)
check("filtering an unknown id returns empty array (defensive)", filterAgents(AGENTS, "nonexistent-group").length === 0);

console.log(`\n${pass} passed, ${fail} failed`);
if (fail > 0) process.exit(1);
console.log("ALL TESTS PASS");
```
