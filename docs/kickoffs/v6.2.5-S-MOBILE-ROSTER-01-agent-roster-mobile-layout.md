# 2. CONTEXT

`RosterScreen.jsx` ("Your agent roster.", route `/bench`) has zero mobile treatment today — at a ~390px viewport it renders its desktop layout unchanged: a 180px fixed left sidebar (filter nav) + a 3-column card grid + a 5-stat horizontal strip, all of which overflow/break below `MOBILE_BREAKPOINT` (768px).

This is one of the 6 destinations in the mobile hamburger drawer shipped by `S-MOBILE-NAV-01` (`AppShell.jsx`) — the nav itself already works on mobile and routes here correctly; this session is only about what renders once the user arrives.

Design session mock (phone-frame HTML mockup) reviewed and approved live by John, no changes requested. Confirmed explicitly in the same conversation: this session does **not** touch `AppShell.jsx` (nav is already done) and does **not** touch `PersonnelScreen.jsx` (the "View Profile" destination) — that screen has its own, larger mobile-design follow-on session, tracked separately (see Backlog Capture below).

Three compositional changes, one file, same underlying data/logic throughout — only the JSX branches on `useIsMobile()`, nothing about `activeFilter`/`sortedAgents`/`filteredAgents`/`stats` computation changes:
1. Left sidebar filter nav (180px, 5 items) → horizontal scrollable chip row.
2. 5-stat single-row strip (+ button + workspace tag) → 3-column stat grid + full-width button, workspace tag dropped on mobile.
3. 3-column agent card grid → single column, full width. Same `AgentCard` component, unchanged internals (avatar, skill bar, situational awareness, Show/Hide Details, action row).

Scroll behavior: unchanged from desktop's existing model. `AppShell.jsx` is a fixed `height:100vh` shell; the content pane already scrolls internally (`overflowY:"auto"`). Mobile keeps exactly that — one continuously-scrolling column containing the chip row, stat grid, and **every** filtered agent's card, no pagination, no truncation.

# 8. NODE.JS TEST

Save as `test-S-MOBILE-ROSTER-01.mjs`, run with `node test-S-MOBILE-ROSTER-01.mjs`, delete before committing.

```js
// test-S-MOBILE-ROSTER-01.mjs — pure Node, no app imports
let pass = 0, fail = 0;
function check(name, cond) {
  if (cond) { pass++; console.log(`PASS: ${name}`); }
  else { fail++; console.log(`FAIL: ${name}`); }
}

// Category B — Logic: grid column selection by viewport
function gridColumns(isMobile) {
  return isMobile ? "1fr" : "repeat(3,1fr)";
}
check("desktop grid is 3-column", gridColumns(false) === "repeat(3,1fr)");
check("mobile grid is single-column", gridColumns(true) === "1fr");

// Category A — Data shape: stats object always has exactly these 5 keys,
// regardless of which layout (row vs grid) renders them
const STAT_KEYS = ["size", "salary", "value", "reportsPerMonth", "trainable"];
function statsShape(stats) {
  const keys = Object.keys(stats);
  return keys.length === STAT_KEYS.length && STAT_KEYS.every(k => keys.includes(k));
}
const sampleStats = { size: 8, salary: 412000, value: 1100000, reportsPerMonth: 36, trainable: 5 };
check("stats object has all 5 required keys, mobile or desktop", statsShape(sampleStats));

// Category B — filter nav item count matches BENCH_FILTERS + "All", same on both layouts
function navItemCount(benchFiltersLength) {
  return benchFiltersLength + 1; // "All" plus each BENCH_FILTERS group
}
check("nav/chip item count = BENCH_FILTERS.length + 1", navItemCount(4) === 5);

// Category B — workspace tag suppressed on mobile only, explicit pinned decision
function shouldShowWorkspaceTag(isMobile) {
  return !isMobile;
}
check("workspace tag shown on desktop", shouldShowWorkspaceTag(false) === true);
check("workspace tag hidden on mobile", shouldShowWorkspaceTag(true) === false);

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail === 0) console.log("ALL TESTS PASS");
process.exit(fail === 0 ? 0 : 1);
```
