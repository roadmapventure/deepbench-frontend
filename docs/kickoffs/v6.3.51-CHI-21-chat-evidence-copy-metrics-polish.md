### Task 1 — Extract `formatHopSummary()` + `HopSummaryLine`, replace both stacked render sites

Add near `HopBadge`'s own definition (~line 896), after `hopBadgeText()`:

```js
// FEATURE: CHI-21 — single source for the combined hop+elapsed wrap-up line, replacing the
// previously-stacked HopBadge (own line) + elapsed-time caption (own line) at both chat-bubble
// render sites. hopEnd is the cumulative SESSION hop total at the moment this message finished
// (hop numbering never resets within a session — CHI-03c), not this answer's own
// hopStart..hopEnd span — confirmed with John this session: "the answer may have taken 4 hops
// to create, but the full agent routing took 9 hops," so the total stated here must match
// Column 3's running count, not a per-answer span. A later message's own hopEnd will be higher;
// this line is a fixed snapshot of the total as of when THIS message finished, not a live value.
function formatHopSummary(hopEnd, totalElapsedMs) {
  if (hopEnd == null || totalElapsedMs == null) return null;
  const hopWord = hopEnd === 1 ? "hop" : "hops";
  return `Full Agent Routing & Answer in ${hopEnd} ${hopWord} total, ${formatElapsed(totalElapsedMs)}`;
}

// FEATURE: CHI-21 — replaces the two previously-stacked <div>s (HopBadge alone, elapsed-time
// caption alone) with one row: badge on the left, enriched summary sentence to its right. Used
// at both MessageBubble render sites (qa branch, default branch) — was duplicated JSX before
// this extraction (Architect Review: duplicate-functionality check).
function HopSummaryLine({ hopStart, hopEnd, totalElapsedMs, accent }) {
  if (hopStart == null || hopEnd == null) return null;
  const summary = formatHopSummary(hopEnd, totalElapsedMs);
  return (
    <div style={{display:"flex",alignItems:"center",gap:6,marginTop:4,flexWrap:"wrap"}}>
      <HopBadge hopStart={hopStart} hopEnd={hopEnd} accent={accent}/>
      {summary && (
        <span style={{fontFamily:mono,fontSize:9.5,color:T.muted}}>{summary}</span>
      )}
    </div>
  );
}
```

// test-CHI-21.mjs — pure Node.js, no app imports
let pass = 0, fail = 0;
function assert(desc, cond) {
  if (cond) { console.log(`  ✓ ${desc}`); pass++; }
  else       { console.error(`  ✗ ${desc}`); fail++; }
}

// --- inline copies of the pure functions under test (same convention as CHI-10's test) ---
function formatElapsed(ms) {
  // mirrors MarketIntelligenceScreen.jsx's real formatElapsed() contract: ms -> "Xm Ys" / "Xs"
  const totalSec = Math.round(ms / 1000);
  const m = Math.floor(totalSec / 60), s = totalSec % 60;
  return m > 0 ? `${m}m ${s}s` : `${s}s`;
}
function formatHopSummary(hopEnd, totalElapsedMs) {
  if (hopEnd == null || totalElapsedMs == null) return null;
  const hopWord = hopEnd === 1 ? "hop" : "hops";
  return `Full Agent Routing & Answer in ${hopEnd} ${hopWord} total, ${formatElapsed(totalElapsedMs)}`;
}

// T1: uses the cumulative total (hopEnd), not a span -- the exact bug John caught live this
// session (a 4-hop-span answer inside a 9-hop-total session must read "9 hops total", not "4").
assert("T1: cumulative hopEnd (9) used, not the answer's own span (hopStart=6,hopEnd=9 -> span 4)",
  formatHopSummary(9, 83000) === "Full Agent Routing & Answer in 9 hops total, 1m 23s");

// T2: pluralization
assert("T2: hopEnd=1 -> singular 'hop'", formatHopSummary(1, 5000) === "Full Agent Routing & Answer in 1 hop total, 5s");
assert("T2b: hopEnd=2 -> plural 'hops'", formatHopSummary(2, 5000) === "Full Agent Routing & Answer in 2 hops total, 5s");

// T3: null guards (Category C -- String Safety on potentially-undefined message fields)
assert("T3: hopEnd null -> null (no summary line renders)", formatHopSummary(null, 5000) === null);
assert("T3b: totalElapsedMs null -> null", formatHopSummary(5, null) === null);
assert("T3c: both undefined -> null", formatHopSummary(undefined, undefined) === null);

// T4: exact approved copy strings (Task 2 / Task 3), guards against a future drive-by edit
// silently drifting from what John confirmed live.
const POINTER_SENTENCE = "I've pulled together an answer — review the full breakdown in the Evidence column to the right. Let me know if you have questions.";

assert("T4: pointer sentence matches approved copy exactly", POINTER_SENTENCE.includes("Evidence column to the right") && POINTER_SENTENCE.startsWith("I've pulled together an answer"));
assert("T4b: evidence header matches approved copy exactly", EVIDENCE_HEADER === "Analysis & Narrative — based on your question...");

console.log(`\n${pass + fail} tests — ${pass} passed, ${fail} failed`);
if (fail > 0) process.exit(1);
