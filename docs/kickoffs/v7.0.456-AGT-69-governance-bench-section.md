```sql
create view public.governance_agent_activity_7d as
select a.id as agent_id, a.code, a.name, a.role, a.specialty,
       count(l.id)::integer                         as calls,
       coalesce(sum(l.input_tokens), 0)::bigint      as input_tokens,
       coalesce(sum(l.output_tokens), 0)::bigint     as output_tokens,
       count(distinct c.id)::integer                 as cycles
from public.agents a
left join public.ai_activity_log l
       on l.agent_id = a.id
      and l.created_at >= now() - interval '7 days'
left join public.runner_cycles c
       on l.id is not null
      and c.started_at >= now() - interval '8 days'
      and c.started_at <= l.created_at
      and coalesce(c.ended_at, now()) >= l.created_at
where a.lane = 'governance' and a.is_active = true
group by a.id, a.code, a.name, a.role, a.specialty;

comment on view public.governance_agent_activity_7d is
  'AGT-69: the Bench Governance section. The is_active lane=governance rows with 7-day calls/tokens from ai_activity_log and the distinct runner_cycles whose window contains a call. Definer-style like runner_public_cycles on purpose: anon holds no grant on runner_cycles and must not; this view exposes aggregates only (no caller_ip, no cost).';

grant select on public.governance_agent_activity_7d to anon, authenticated;
```
