```sql
-- SES-360 (v7.0.454): governance-agent usage and the four-row ship census are views the standing
-- brief renders. Rationale and measurements: docs/kickoffs/v7.0.454-SES-360-governance-agents-brief-block.md

-- (1) one row per governance agent x call_source over a ROLLING 7 days. The roster is joined by
--     LANE, never by id (Rule #1). ai_activity_log exceeds a REST page (1,311 governance rows in
--     7 days against PostgREST's 1,000-row cap, measured 2026-09-11), so the aggregation lives here.
--     A NULL call_source stays NULL: the renderer prints it as *unlabelled* (LOG-128).
create or replace view public.governance_agent_usage
with (security_invoker = true) as
select a.code,
       a.id                                             as agent_id,
       a.role,
       l.call_source,
       count(*)::int                                    as calls,
       coalesce(sum(l.input_tokens), 0)::bigint         as input_tokens,
       coalesce(sum(l.output_tokens), 0)::bigint        as output_tokens,
       count(*) filter (where l.input_tokens is null and l.output_tokens is null)::int as untokened
  from public.agents a
  join public.ai_activity_log l on l.agent_id = a.id
 where a.lane = 'governance'
   and l.created_at >= now() - interval '7 days'
 group by a.code, a.id, a.role, l.call_source;

comment on view public.governance_agent_usage is
  'SES-360: calls and token sums per governance-lane agent x call_source, rolling 7 days. The standing brief renders it; nothing re-derives it.';

-- (2) the four-row ship census. A ship is a runner_items ship card (unattended) or a
--     runner_decisions ship decision (record_ship_decision, attended and unattended) inside the
--     window. SES-345's four legs as measurable today: automation_rank, kickoff_link, a
--     runner_cycles row with push_sha that NAMES the ticket (item_id is text and mixed -- bare id
--     or uuid -- and backlog_item_id is the uuid where set), and a runner_verdicts row. When
--     SES-345 lands its verdict-side sha, change the sha leg HERE, nowhere else.
create or replace view public.ship_handoff_census
with (security_invoker = true) as
with ships as (
  select i.backlog_id
    from public.runner_items i
   where i.kind = 'ship' and i.backlog_id is not null
     and i.created_at >= now() - interval '7 days'
  union
  select d.backlog_id
    from public.runner_decisions d
   where d.kind = 'ship' and d.backlog_id is not null
     and d.decided_at >= now() - interval '7 days'
), per_row as (
  select s.backlog_id,
         (b.automation_rank is not null) as has_rank,
         (b.kickoff_link   is not null) as has_kickoff,
         exists (select 1 from public.runner_cycles c
                  where c.push_sha is not null
                    and (c.item_id = s.backlog_id
                         or c.item_id = b.id::text
                         or c.backlog_item_id::text = b.id::text)) as has_sha,
         exists (select 1 from public.runner_verdicts v
                  where v.backlog_id = s.backlog_id)                 as has_verdict
    from ships s
    left join public.backlog_items b on b.backlog_id = s.backlog_id
), legs as (
  -- backlog_id carries no unique constraint (CHI-48 has two rows): one row per ship, any row counts.
  select backlog_id,
         bool_or(has_rank)    as has_rank,
         bool_or(has_kickoff) as has_kickoff,
         bool_or(has_sha)     as has_sha,
         bool_or(has_verdict) as has_verdict
    from per_row
   group by backlog_id
)
select count(*)::int                                                                        as ships,
       count(*) filter (where has_rank and has_kickoff and has_sha and has_verdict)::int   as ships_all_four,
       count(*) filter (where not has_rank)::int                                            as missing_rank,
       count(*) filter (where not has_kickoff)::int                                         as missing_kickoff,
       count(*) filter (where not has_sha)::int                                             as missing_sha,
       count(*) filter (where not has_verdict)::int                                         as missing_verdict
  from legs;

comment on view public.ship_handoff_census is
  'SES-360: ships in a rolling 7 days and how many carried all four SES-345 handoff rows, with the missing count per leg. One row, always.';

-- (3) grants: service_role only, the SES-78a rule every brief view follows. Default privileges are
--     closed (DAT-18), so this is belt and braces rather than a fix -- assert both ways in QA.
revoke all on public.governance_agent_usage from public, anon, authenticated;
revoke all on public.ship_handoff_census   from public, anon, authenticated;
grant select on public.governance_agent_usage to service_role;
grant select on public.ship_handoff_census   to service_role;
```

```js
async function restGet(url, key, q) {
  const res = await fetch(`${url.replace(/\/+$/, "")}/rest/v1/${q}`, { headers: { apikey: key, Authorization: `Bearer ${key}` } });
  if (!res.ok) throw new Error(`REST ${q} -> HTTP ${res.status} ${res.statusText}: ${await res.text().catch(() => "")}`);
  return res.json();
}

export async function theViewsAnswerWithAConsistentShape(url, key) {

  const usage = await restGet(url, key, `${USAGE_VIEW}?select=code,agent_id,role,call_source,calls,input_tokens,output_tokens,untokened&order=code,call_source.nullslast`);
  assert.ok(Array.isArray(usage), `${USAGE_VIEW} must answer with an array`);
  for (const r of usage) {
    assert.ok(ids.has(r.agent_id), `${USAGE_VIEW} row names '${r.agent_id}', which is not a governance-lane agent -- the view's lane filter has drifted`);
    assert.ok(Number(r.calls) >= 1, `${USAGE_VIEW} must never return a zero-call row (${r.agent_id}/${r.call_source}) -- a GROUP BY row exists only because rows exist`);
    assert.ok(Number(r.input_tokens) >= 0 && Number(r.output_tokens) >= 0 && Number(r.untokened) >= 0 && Number(r.untokened) <= Number(r.calls),
      `${USAGE_VIEW} sums must be non-negative and untokened <= calls (${r.agent_id}/${r.call_source})`);
  }
  const keys = usage.map(r => `${r.agent_id}|${r.call_source}`);
  assert.strictEqual(new Set(keys).size, keys.length, `${USAGE_VIEW} must return one row per (agent, call_source)`);

  const census = await restGet(url, key, `${CENSUS_VIEW}?select=ships,ships_all_four,missing_rank,missing_kickoff,missing_sha,missing_verdict`);
  assert.strictEqual(census.length, 1, `${CENSUS_VIEW} must always return exactly one row (an aggregate with no GROUP BY), got ${census.length}`);

}
```
