Add and export these two helpers; mobile renders `` `${formatElapsedShort(clock - base)} · ${formatExpectationShort(liveStatus?.expectationMs ?? 120000)}` `` in mono 10 `T.brassDeep`, guarded by the same `base &&` condition desktop uses. **Desktop's `elapsedText` is unchanged.**
```js
// MOB-4a -- mobile-width elapsed/expectation. m:ss instead of "Xm Ys", and the expectation
// abbreviated to its minute figure. ARCHITECTURE.md §19o requires the expectation to remain
// visible on mobile, so it is shortened, never dropped. Mobile always derives from the numeric
// expectationMs (falling back to desktop's own 120000 default) rather than liveStatus.expectation's
// pre-formatted desktop string, which is too long for this row by construction.
export function formatElapsedShort(ms) {
  const s = Math.max(0, Math.floor(ms / 1000));
  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
}
export function formatExpectationShort(ms) {
  const sec = Math.ceil(ms / 1000 / 5) * 5;
  const m = Math.floor(sec / 60);
  return m > 0 ? `>${m}m` : `>${sec}s`;
}
```

// S-MOB-4a inline test: the two mobile time helpers, and the select-background rule that
// determines whether tokens.js's global chevron survives. Pure Node -- the .jsx module can't be
// imported here, so the helpers are mirrored from the kickoff spec; the committed implementation
// must match verbatim.
function formatElapsedShort(ms) {
  const s = Math.max(0, Math.floor(ms / 1000));
  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
}
function formatExpectationShort(ms) {
  const sec = Math.ceil(ms / 1000 / 5) * 5;
  const m = Math.floor(sec / 60);
  return m > 0 ? `>${m}m` : `>${sec}s`;
}
let pass = true;
const check = (name, got, want) => {
  const ok = got === want;
  if (!ok) { pass = false; console.error(`FAIL ${name}\n  got:  ${JSON.stringify(got)}\n  want: ${JSON.stringify(want)}`); }
  else console.log(`ok ${name}`);
};
check("elapsed zero",        formatElapsedShort(0),      "0:00");
check("elapsed pads seconds",formatElapsedShort(34000),  "0:34");
check("elapsed past a minute",formatElapsedShort(112000),"1:52");
check("elapsed exact minute",formatElapsedShort(60000),  "1:00");
check("elapsed negative clamps", formatElapsedShort(-5000), "0:00");
check("expect 2m default",   formatExpectationShort(120000), ">2m");
check("expect rounds up to 5s", formatExpectationShort(31000), ">35s");
check("expect sub-minute",   formatExpectationShort(20000),  ">20s");
check("expect 3m",           formatExpectationShort(170000), ">2m");

// The chevron rule. React's inline `background` is the SHORTHAND and resets background-image to
// none, wiping tokens.js L78's chevron; `backgroundColor` does not. This asserts the property NAME
// the implementation must use -- a test that passed either way would prove nothing.
const styleUsed = { backgroundColor: "#ffffff" };   // mirror of the committed LAV <select> style
check("select sets backgroundColor, not background",
  Object.prototype.hasOwnProperty.call(styleUsed, "background") ? "background" : "backgroundColor",
  "backgroundColor");
console.log(pass ? "ALL TESTS PASS" : "TESTS FAILED");
if (!pass) process.exit(1);
