## 2. CONTEXT

A brokered dispatch's `intent_slug` currently rides the model's *echo* of the candidate list:
`matchedCandidate.intent_slug || null` reads Michelle Manning — Project Manager's
`agent-selection-intent` output, and when her turn omits the optional per-candidate field
(~1 in 5 brokered picks, measured 2026-08-07: 17 labeled vs 4 blank identical
`data-room-custody` evidence dispatches), the call logs `feature = <capability>:none:depthN`,
the Assembly drawer files real work as an unlabeled "In progress" ghost, and hop-budget
stats pool into a junk `:none` key. The direct `delegate_to_agent` path reads the same class
of model-authored `tool_input.intent_slug` untrusted-as-envelope.

**The fix (John-approved 2026-08-07/08):** the model keeps the pick; the platform carries the
envelope. The harness resolves the Intent from its own roster data — validated echo, then
single-Intent stamp, then the capability's own declared `default_intent_slug` (data, already
migrated live: `capabilities.default_intent_slug`, set to `library-evidence-intent` for
`data-room-custody` only), then honest blank. Zero model calls, zero per-capability code
(§19b/§19d — the rule stays generic; the default is a data declaration, same authorship class
as `capabilities.display_phrase`). The §19s amendment documenting this rule is already
committed by the design session — no doc task for you.

The Supabase migration is **already applied** (column exists, one row set). Do not write or

### T1 — `resolveDispatchIntent()` helper (new export in `api/capabilities/execute.js`)

Place it directly below `resolveCapabilityHolder()` (~L500) and mirror its fetch idiom
(`getSupabaseHeaders`, `encodeURIComponent`, plain REST):

```js
// FEATURE: LOO-33 -- the platform carries the envelope: a dispatch's intent_slug is resolved
// from the harness's own roster data, never trusted from a model's echo of the candidate
// list (the transcription drops the optional field ~1 in 5 brokered picks). Ladder: a valid
// echoed intent wins (the picker's reasoned choice, validated by membership); a single-intent
// capability stamps deterministically; a multi-intent capability falls to its own declared
// default_intent_slug (data, §19b -- same authorship class as display_phrase); else null,
// honestly blank, exactly the pre-LOO-33 behavior. On any fetch failure this degrades to the
// raw echo -- resolution must never make a dispatch worse than it was before this feature.
export async function resolveDispatchIntent(capability_slug, echoedIntentSlug) {
  const supabaseUrl = process.env.SUPABASE_URL;
  const supabaseKey = process.env.SUPABASE_SERVICE_KEY;
  if (!supabaseUrl || !supabaseKey) return { intent_slug: echoedIntentSlug || null, resolution: 'unverified' };
  const headers = getSupabaseHeaders(supabaseKey);
  try {
    const linkRes = await fetch(
      `${supabaseUrl}/rest/v1/capability_skill_profiles?capability_slug=eq.${encodeURIComponent(capability_slug)}&select=skill_profile_slug`,
      { headers }
    );
    if (!linkRes.ok) throw new Error('link fetch failed');
    const linkRows = await linkRes.json();
    let intents = [];
    if (linkRows.length) {
      const intentRes = await fetch(
        `${supabaseUrl}/rest/v1/skill_profiles?slug=in.(${encodeURIComponent(linkRows.map(r => r.skill_profile_slug).join(','))})&skill_type_slug=eq.intent&select=slug`,
        { headers }
      );
      if (!intentRes.ok) throw new Error('intent fetch failed');
      intents = (await intentRes.json()).map(r => r.slug);
    }
    if (echoedIntentSlug && intents.includes(echoedIntentSlug)) {
      return { intent_slug: echoedIntentSlug, resolution: 'echo' };
    }
    if (intents.length === 1) {
      return { intent_slug: intents[0], resolution: 'single' };
    }
    if (intents.length > 1) {
      const capRes = await fetch(
        `${supabaseUrl}/rest/v1/capabilities?slug=eq.${encodeURIComponent(capability_slug)}&select=default_intent_slug&limit=1`,
        { headers }
      );
      if (!capRes.ok) throw new Error('capability fetch failed');
      const capRows = await capRes.json();
      const def = capRows?.[0]?.default_intent_slug || null;
      if (def && intents.includes(def)) {
        return { intent_slug: def, resolution: 'default' };
      }
    }
    return { intent_slug: null, resolution: 'none' };
  } catch (e) {
    return { intent_slug: echoedIntentSlug || null, resolution: 'unverified' };
  }
}
```

The `resolution` tag exists for tests and future logging; nothing else consumes it this
session.

```js
// test-loo-33.mjs -- S-LOO-33: resolveDispatchIntent ladder against LIVE Supabase data,
// plus structural asserts that both seams actually consume it. Run: node test-loo-33.mjs
import { readFileSync } from 'node:fs';

// Load .env.local (gitignored, already copied into this worktree) into process.env.
for (const line of readFileSync('.env.local', 'utf8').split(/\r?\n/)) {
  const m = line.match(/^([A-Z0-9_]+)=(.*)$/);
  if (m && !process.env[m[1]]) process.env[m[1]] = m[2];
}
// The api route reads SUPABASE_URL/SUPABASE_SERVICE_KEY; .env.local may use VITE_-prefixed
// names for the URL -- map if needed.
if (!process.env.SUPABASE_URL && process.env.VITE_SUPABASE_URL) process.env.SUPABASE_URL = process.env.VITE_SUPABASE_URL;

const { resolveDispatchIntent } = await import('./api/capabilities/execute.js');

let failures = 0;
const assertEq = (label, got, want) => {
  const ok = JSON.stringify(got) === JSON.stringify(want);
  console.log(`${ok ? 'PASS' : 'FAIL'}: ${label} -> ${JSON.stringify(got)}`);
  if (!ok) failures++;
};

// 1. Blank echo on a multi-intent capability WITH a declared default -> the default.
assertEq('blank echo, data-room-custody -> declared default',
  await resolveDispatchIntent('data-room-custody', null),
  { intent_slug: 'library-evidence-intent', resolution: 'default' });

// 2. Valid echo beats the default -- her reasoned choice wins.
assertEq('valid echo wins over default',
  await resolveDispatchIntent('data-room-custody', 'library-record-lookup-intent'),
  { intent_slug: 'library-record-lookup-intent', resolution: 'echo' });

// 3. Invented echo is rejected by membership -> falls to the default.
assertEq('invented echo -> default',
  await resolveDispatchIntent('data-room-custody', 'totally-invented-intent'),
  { intent_slug: 'library-evidence-intent', resolution: 'default' });

// 4. Single-intent capability stamps deterministically, echo irrelevant.
assertEq('single-intent capability -> stamped',
  await resolveDispatchIntent('memory-consolidation', null),
  { intent_slug: 'reasoner-intent', resolution: 'single' });

// 5. Multi-intent capability with NO declared default -> honest blank (default is opt-in).
assertEq('multi-intent, no default -> honest blank',
  await resolveDispatchIntent('data-analysis', null),
  { intent_slug: null, resolution: 'none' });

// 6. Unknown capability -> honest blank, never a throw (LOO-34's territory, not ours).
assertEq('unknown capability -> honest blank',
  await resolveDispatchIntent('no-such-capability-xyz', null),
  { intent_slug: null, resolution: 'none' });

// 7. Structural: no echo-read survives, both seams consume the resolver.
const src = readFileSync('api/capabilities/execute.js', 'utf8');
const echoReads = (src.match(/matchedCandidate\.intent_slug \|\|/g) || []).length;
console.log(`${echoReads === 0 ? 'PASS' : 'FAIL'}: zero surviving matchedCandidate.intent_slug || reads (found ${echoReads})`);
if (echoReads !== 0) failures++;
const callSites = (src.match(/await resolveDispatchIntent\(/g) || []).length;
console.log(`${callSites >= 2 ? 'PASS' : 'FAIL'}: resolveDispatchIntent consumed at both seams (found ${callSites} call sites)`);
if (callSites < 2) failures++;

console.log(failures === 0 ? 'ALL PASS' : `${failures} FAILURE(S)`);
process.exit(failures === 0 ? 0 : 1);
```
