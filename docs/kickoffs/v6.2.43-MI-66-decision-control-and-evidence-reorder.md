John's call: the warning belongs to the decision moment, not the entire flow.

```js
// test-MI-66.mjs — pure Node.js, no app imports
let pass = 0, fail = 0;
function assert(desc, cond) {
  if (cond) { console.log(`  ✓ ${desc}`); pass++; }
  else       { console.error(`  ✗ ${desc}`); fail++; }
}

// Mirrors EvidenceColumn's `st && hypFlow.stage === "result"` block's render order after MI-66
// (MarketIntelligenceScreen.jsx, EvidenceColumn). Returns an ordered array of section keys that
// would actually render for a given state -- pure logic, no JSX/React involved.
function resultSectionOrder(st, hypFlow, hasActualRows, hasTheorizedRows) {
  const sections = [];
  const showDecisionBlock = !hypFlow.confirmation;
  if (showDecisionBlock) {
    if (st.override_warning) sections.push("override_warning");
    sections.push("decision_copy");
    sections.push("decision_buttons");
  }
  if (hasActualRows) sections.push("actual_table");
  if (hasTheorizedRows) sections.push("theorized_table");
  if (st.visualization) sections.push("chart");
  sections.push("thesis"); // container always present; individual rows conditional inside, unchanged by this session
  return sections;
}

// T1: full result, no confirmation pending, override present -- decision block (with warning) leads
const t1 = resultSectionOrder({ override_warning: true, visualization: {} }, { confirmation: null }, true, true);
assert("T1: override_warning is first", t1[0] === "override_warning");
assert("T1: decision copy/buttons directly follow the warning", t1[1] === "decision_copy" && t1[2] === "decision_buttons");
assert("T1: data tables come after the decision block", t1.indexOf("actual_table") > t1.indexOf("decision_buttons"));
assert("T1: chart comes after data tables", t1.indexOf("chart") > t1.indexOf("theorized_table"));
assert("T1: thesis is last", t1[t1.length - 1] === "thesis");

// T2: no override flagged -- decision block still leads, just without the warning line
const t2 = resultSectionOrder({ override_warning: false }, { confirmation: null }, false, false);
assert("T2: no override_warning present when st.override_warning is false", !t2.includes("override_warning"));
assert("T2: decision block still leads", t2[0] === "decision_copy" && t2[1] === "decision_buttons");

// T3: confirmation pending -- decision block (and override_warning) hides entirely, per John's
// explicit 'hide' call; evidence (tables/chart/thesis) still renders (ConfirmationCard is a sibling)
const t3 = resultSectionOrder({ override_warning: true, visualization: {} }, { confirmation: { confirmation_id: "x" } }, true, false);
assert("T3: override_warning hidden once confirmation is pending", !t3.includes("override_warning"));
assert("T3: decision buttons hidden once confirmation is pending", !t3.includes("decision_buttons"));
assert("T3: evidence (actual table, chart, thesis) still present under confirmation", t3.includes("actual_table") && t3.includes("chart") && t3.includes("thesis"));

// T4: no data-point rows and no chart -- decision block still leads, thesis container always present
const t4 = resultSectionOrder({}, { confirmation: null }, false, false);
assert("T4: decision block leads even with zero evidence rows", t4[0] === "decision_copy");
assert("T4: thesis container always present", t4.includes("thesis"));

console.log(`\n${pass + fail} tests — ${pass} passed, ${fail} failed`);
if (fail > 0) process.exit(1);
else console.log("ALL TESTS PASS");
```
