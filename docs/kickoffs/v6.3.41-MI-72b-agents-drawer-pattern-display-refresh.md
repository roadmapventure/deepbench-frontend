## NODE.JS TEST

Run with `node test-mi-72b.mjs`, delete before committing. Category D (component lifecycle / useEffect+useRef) applies to Task 2's dependency-array change — reimplemented as a plain-function simulation since this is pure Node (no React).

```js
// test-mi-72b.mjs
// Category B (logic) + D (lifecycle semantics of the refreshKey-driven effect, simulated).

let pass = 0, fail = 0;
function check(name, cond) {
  if (cond) { pass++; } else { fail++; console.error(`FAIL: ${name}`); }
}

// Simulates useAgentActivitySummary's effect re-firing exactly when refreshKey changes,
// matching React's dependency-array semantics (Object.is comparison per key), not on every render.
function simulateEffectRuns(refreshKeySequence) {
  let lastKey;
  let runs = 0;
  for (const key of refreshKeySequence) {
    if (lastKey === undefined || !Object.is(lastKey, key)) {
      runs++;
      lastKey = key;
    }
  }
  return runs;
}

// B1: refreshKey never changing (existing call sites, e.g. baselineActivity) -> exactly 1 run
check('B1: stable refreshKey (0,0,0,0) fetches exactly once', simulateEffectRuns([0, 0, 0, 0]) === 1);

// B2: refreshKey bumped 3 times (3 drawer opens) -> 4 total runs (1 initial + 3 refreshes)
check('B2: refreshKey (0,1,2,3) fetches 4 times', simulateEffectRuns([0, 1, 2, 3]) === 4);

// B3: Drawer's onOpen fires only on closed->open, never on open->closed
function toggle(openBefore, onOpenCallCount) {
  let calls = onOpenCallCount;
  const willOpen = !openBefore;
  if (willOpen) calls++;
  return { nowOpen: willOpen, calls };
}
{
  let state = { open: false, calls: 0 };
  state = { open: toggle(state.open, state.calls).nowOpen, calls: toggle(state.open, state.calls).calls }; // open (closed->open): should fire
  check('B3a: closed->open fires onOpen once', state.calls === 1);
  const before = state.calls;
  state = { open: toggle(state.open, state.calls).nowOpen, calls: toggle(state.open, state.calls).calls }; // close (open->closed): should NOT fire
  check('B3b: open->closed does not fire onOpen', state.calls === before);
}

// Data shape (mirrors MI-72a's PATTERN_NAME fallback): unrecognized pattern slug still renders,
// falls back to the raw slug rather than throwing/blank.
{
  const PATTERN_NAME = { rag: 'RAG', 'tool-use': 'Tool Use' };
  const label = (slug) => PATTERN_NAME[slug] || slug;
  check('C1: known slug maps to human name', label('rag') === 'RAG');
  check('C2: unknown slug falls back to raw slug, not undefined/blank', label('case-based-reasoning') === 'case-based-reasoning');
}

if (fail === 0) {
  console.log(`ALL TESTS PASS (${pass}/${pass})`);
} else {
  console.error(`${fail} FAILED, ${pass} passed`);
  process.exit(1);
}
```
