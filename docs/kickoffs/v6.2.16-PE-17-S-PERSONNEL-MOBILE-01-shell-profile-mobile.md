# 2. CONTEXT

`PersonnelScreen.jsx` (route `/bench/:agentId`, "the personnel file of {agent}") has zero mobile treatment today — it's the "View Profile" destination from every `RosterScreen.jsx` agent card, so since `RO-13` shipped, tapping any agent on a phone lands on this screen's unmodified desktop layout: a 180px left sidebar (agent identity strip + Profile/Resume/Training/Playbook tab nav) and a plain-text page header, both of which overflow/break below `MOBILE_BREAKPOINT` (768px).

This session covers the **shared shell** (sidebar → mobile nav, page header) **and the Profile tab only** — Resume/Training/Playbook tab *bodies* stay desktop-styled content for now, a pre-existing gap this session doesn't worsen (they already have zero mobile treatment; this session doesn't touch `ResumeTab.jsx`, or the `TrainingTab`/`PlaybookTab` functions). Those three are separate, larger follow-on sessions — logged as `PE-18` (see Backlog Capture below). `PlaybookTab` in particular needs no layout work when its turn comes (already single-column, full-width cards).

Three rounds of live mock iteration with John landed on a design that's a further consolidation past `RO-13`'s pattern, not just a reapplication of it — worth stating precisely since it's the one genuinely new idea this session introduces:

- **Round 1** mock (sidebar → chip row, plain header unchanged) was rejected for redundancy — the agent's name/role/code appeared three times (page header title, a compact identity strip, and `ProfileTab`'s own ID Badge card).
- **Round 2** dropped the middle identity strip — down to two occurrences (header title, ID Badge card).
- **Round 3 (final, approved):** the page header's plain title and `ProfileTab`'s ID Badge card **merge into one persistent persona block** — avatar, name, role, tenure, status chips, quote, and the three stat badges (Situational/Readiness/Skill) — that renders once, above the tab bar, on **every** tab (not just Profile). `ProfileTab` itself skips rendering its own ID Badge card on mobile since it's now fully redundant with this block. Name/role/code/status now appear exactly once anywhere on the mobile screen.
- **Back navigation**, raised as an open question and resolved: the mobile hamburger's "Agent Roster" item already provides a path back (`SH-19`, no new code), but John asked for a direct one-tap link too — a small "← Agent Roster" link added to the top of the persona block, same "screen grows its own in-body control" pattern `AZ-19` used to close `SH-20` for Analyzer's mapping page (`rightContent`/`onBack` aren't rendered by `AppShell.jsx` on mobile at all, by design — this is not a workaround for that, it's the same intended pattern).

// test-PE-17.mjs — pure Node, no app imports
let pass = 0, fail = 0;

  if (cond) { pass++; console.log(`PASS: ${name}`); }
  else { fail++; console.log(`FAIL: ${name}`); }

  return isMobile ? "1fr" : "1fr 1fr";

check("mobile Profile grid is single-column", profileGridColumns(true) === "1fr");

  return isMobile ? "grid-3col" : "flex-row";
}
check("desktop Intelligence Config stays a flex row", configStripMode(false) === "flex-row");
check("mobile Intelligence Config becomes a 3-col grid", configStripMode(true) === "grid-3col");

// Category B — ID Badge card suppressed on mobile only (redundant with the new persona block)
function shouldShowIdBadgeCard(isMobile) {
  return !isMobile;
}
check("ID Badge card shown on desktop", shouldShowIdBadgeCard(false) === true);
check("ID Badge card hidden on mobile", shouldShowIdBadgeCard(true) === false);

// Category A — tab bar always has exactly the same 4 tabs, same order, mobile or desktop
const NAV_GROUPS_SAMPLE = [
  { id: "overview", tabs: [{ id: "profile", label: "Profile" }] },
  { id: "configure", tabs: [
    { id: "resume", label: "Resume" },
    { id: "training", label: "Training" },
    { id: "playbook", label: "Playbook" },

check("flattened tab list has exactly 4 tabs", flatTabs.length === 4);
check("flattened tab order is Profile, Resume, Training, Playbook",
  flatTabs.map(t => t.id).join(",") === "profile,resume,training,playbook");
