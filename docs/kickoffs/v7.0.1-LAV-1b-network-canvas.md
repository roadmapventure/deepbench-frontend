// --- Mirror: mode derivation over an event sequence (Task 3) ---
function deriveMode(evts, { running, terminal }) {
  if (!running && !terminal) return "idle";
  if (terminal) return terminal === "error" ? "error" : "complete";
  return evts.some(e => e.type === "delegation_return") ? "orch" : "route";
}
const seq = [
  { type: "intent_routing" }, { type: "delegation" }, { type: "delegation_complete" },
];
assert(deriveMode([], { running: false, terminal: null }) === "idle", "no run: idle");
assert(deriveMode(seq, { running: true, terminal: null }) === "route", "delegations without any return: Routing");
assert(deriveMode([...seq, { type: "delegation_return" }], { running: true, terminal: null }) === "orch", "first delegation_return flips to Orchestrating");
assert(deriveMode(seq, { running: false, terminal: "result" }) === "complete", "terminal result: Complete");
assert(deriveMode(seq, { running: false, terminal: "error" }) === "error", "terminal error: error state");

// --- Mirror: open-delegation (orchestrator) tracking (Task 2 node states) ---
function openDispatchers(evts) {
  const open = new Map();
  for (const e of evts) {
    if (e.type === "delegation") open.set(e.agentId, (open.get(e.agentId) || 0) + 1);
    if (e.type === "delegation_return" || e.type === "delegation_complete") {
      const credited = e.type === "delegation_return" ? e.data?.returnTo : e.data?.dispatcher;
      if (credited && open.get(credited)) open.set(credited, open.get(credited) - 1);
    }
  }
  return [...open.entries()].filter(([, n]) => n > 0).map(([id]) => id);
}
assert(openDispatchers([{ type: "delegation", agentId: "marcus" }]).includes("marcus"),
  "a dispatched delegation with no completion leaves the dispatcher open (orchestrating ring)");
assert(!openDispatchers([
  { type: "delegation", agentId: "marcus" },
  { type: "delegation_return", data: { returnTo: "marcus" } },
]).includes("marcus"), "a return crediting the dispatcher closes it");

// --- Mirror: edge materialization + re-dispatch color (Task 2) ---
function edgeColorKey(seen, from, to) {
  const k = `${from}>${to}`;
  const isRedispatch = seen.has(k);
  seen.add(k);
  return isRedispatch ? "flag" : "dispatch";
}
{

  assert(edgeColorKey(seen, "marcus", "nadia") === "flag", "repeat crossing same run: re-dispatch color");
  assert(edgeColorKey(seen, "marcus", "priya") === "dispatch", "different pair unaffected");
}

// --- Mirror: engagement order is arrival order, one at a time (sequential truth) ---
function engagementOrder(evts) {
  const order = [];
  for (const e of evts) { const id = e.agentId; if (id && !order.includes(id)) order.push(id); }
  return order;
}
assert(JSON.stringify(engagementOrder([
  { agentId: "sam" }, { agentId: "michelle" }, { agentId: "sam" }, { agentId: "marcus" },
])) === JSON.stringify(["sam", "michelle", "marcus"]), "arc joins in first-engagement order, no duplicates");
