import { readFileSync } from 'node:fs';
const html = readFileSync('docs/runbooks/briefing-template.html', 'utf8');
const m = html.match(/\.idchip\{([^}]*)\}/);
if (!m) { console.error('FAIL: .idchip rule not found'); process.exit(1); }
const rule = m[1].replace(/\s+/g, '');
const checks = [
  ['max-width:35%',            rule.includes('max-width:35%')],
  ['45% gone from rule',      !rule.includes('45%')],
  ['still shrinkable',         rule.includes('flex:01auto') && rule.includes('min-width:0')],
  ['still wraps',              rule.includes('overflow-wrap:anywhere')],
  ['exactly one .idchip rule', (html.match(/\.idchip\{/g) || []).length === 1],
  ['<title> intact in first 8KB',
    html.slice(0, 8192).includes('<title>DeepBench Morning Briefing</title>')],
];
let fail = 0;
for (const [name, ok] of checks) {
  console.log((ok ? 'PASS' : 'FAIL') + ' — ' + name);
  if (!ok) fail = 1;
}
process.exit(fail);
