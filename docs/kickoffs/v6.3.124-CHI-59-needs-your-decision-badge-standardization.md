```js
// test-chi-59.mjs — pure Node, no app imports. Reimplements each drawer's "is a decision still
// pending" predicate to verify badge visibility across every relevant state shape.

let pass = 0, fail = 0;
function assert(cond, label) {
  if (cond) { pass++; console.log(`PASS: ${label}`); }
  else { fail++; console.log(`FAIL: ${label}`); }
}

// --- Mirrors of Task 3's inline JSX conditions ---
function isQaPending(qaEvidence) { return !!qaEvidence && !qaEvidence.reviewChoice; }
function isCandidatesPending(hypFlow) { return !!hypFlow && hypFlow.stage === "choosing"; }
function isResultPending(hypFlow) { return !!hypFlow && hypFlow.stage === "result" && !hypFlow.resolution; }
function isDraftPending(hypFlow) { return !!(hypFlow && hypFlow.confirmation); } // unchanged from existing isHypAwaitingConfirmation

// --- Analysis & Narrative (qa) ---
{
  assert(isQaPending({ text: "an answer" }) === true, "qa: fresh answer, no reviewChoice yet — badge shows");
  assert(isQaPending({ text: "an answer", reviewChoice: "good" }) === false, "qa: reviewChoice='good' set — badge gone");
  assert(isQaPending({ text: "an answer", reviewChoice: "exploring" }) === false, "qa: reviewChoice='exploring' set — badge gone");
  assert(isQaPending(null) === false, "qa: no qaEvidence at all — badge false (drawer doesn't render anyway)");
}

// --- Candidates (choosing) ---
{
  assert(isCandidatesPending({ stage: "choosing" }) === true, "candidates: stage=choosing — badge shows");
  assert(isCandidatesPending({ stage: "generating" }) === false, "candidates: stage=generating (not chosen yet, not choosable yet) — badge false");
  assert(isCandidatesPending({ stage: "testing" }) === false, "candidates: stage=testing (already chosen, moved on) — badge false");
  assert(isCandidatesPending(null) === false, "candidates: no hypFlow — badge false");
}

// --- Result (hyp-result) ---
{
  assert(isResultPending({ stage: "result" }) === true, "result: stage=result, no resolution yet — badge shows");
  assert(isResultPending({ stage: "result", resolution: "committed" }) === false, "result: resolution set — badge gone");
  assert(isResultPending({ stage: "result", resolution: "discarded" }) === false, "result: discarded also counts as resolved — badge gone");
  assert(isResultPending({ stage: "testing" }) === false, "result: still testing, not at result stage yet — badge false");
  assert(isResultPending({ stage: "committing" }) === false, "result: past result stage, mid-commit — badge false (Draft Forecast/committing has its own gate)");
}

// --- Draft Forecast (hyp-confirmation) — unchanged behavior, confirms refactor didn't alter it ---
{
  assert(isDraftPending({ confirmation: { proposed_action: {} } }) === true, "draft: confirmation landed — badge shows, same as before this session");
  assert(isDraftPending({ confirmation: null }) === false, "draft: no confirmation yet — badge false");
  assert(isDraftPending(null) === false, "draft: no hypFlow — badge false");
}

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS PASS: NO — fix before committing"); process.exit(1); }
console.log("ALL TESTS PASS");
```
