## CONTEXT

Third `LAV-1` sub-session. Adds the dark HARNESS TRACE console (real event lines + Raw JSON toggle + click-an-id record inspector resolving the **actual Supabase row**), the span waterfall, the title-bar meters, and the node card's pattern pill + model tag. Everything here is **row-derived data**: it arrives by reading `ai_activity_log` keyed by the run's `trace_id` — the same read-time posture as the pattern pill (`LOG-79`/`LOG-95b`) — plus the live event ledger 1b already keeps. Meters tick stepwise as rows land; that is the honest cadence, do not smooth or interpolate.

**Vocabulary rule (John/design, hard):** the console speaks the platform's REAL event vocabulary — `delegation`, `delegation_complete`, `delegation_return`, `intent_routing`, `qa_answer`, `agent_selection`, `result`, `error`, plus row-derived `model_call` lines. No invented taxonomy (`agent.invoke`, `llm.usage`, `orchestration.reduce` do not exist and must not appear).

**Concurrency truth (settled):** within one trace, execution is sequential; the only real overlap is **nesting** (a parent span open over its child). The meter is therefore **Active Spans** (currently-open nesting, live) with a peak; there is no sibling-concurrency meter and no k-of-N chip. The waterfall's honest shape is an enclosing parent bar over a staircase of children.

// test-lav-1c.mjs — pure Node, no app imports. Mirrors the derivations this session ships and
// fs-guards the new/changed files.
import { readFileSync } from "node:fs";
let pass = 0, fail = 0;
const assert = (c, l) => { if (c) { pass++; console.log(`PASS: ${l}`); } else { fail++; console.log(`FAIL: ${l}`); } };

// --- loops/retries parsed from real feature encodings (execute.js L308 shape) ---
function deriveLoopsRetries(rows) {
  let maxDepth = 0, retries = 0;
  for (const r of rows) {
    const d = r.feature?.match(/:depth(\d+)/); if (d) maxDepth = Math.max(maxDepth, +d[1]);
    const a = r.feature?.match(/:apiRetry(\d+)/); if (a) retries += 1;
  }
  return { maxDepth, retries };
}
{
  const rows = [
    { feature: "channel-intelligence:ci-answer-intent:depth0" },
    { feature: "project-manager:agent-selection-intent:depth1" },
    { feature: "qa-guard:qg-review-intent:depth2:apiRetry1" },
  ];
  const { maxDepth, retries } = deriveLoopsRetries(rows);
  assert(maxDepth === 2, "loops = max :depthN across rows");
  assert(retries === 1, ":apiRetryN counts as a retry");
  assert(deriveLoopsRetries([]).maxDepth === 0, "no rows: loops 0 (renders as dash)");
}

// --- p50 over latency_ms ---
function p50(rows) { const s = rows.map(r => r.latency_ms).filter(v => v != null).sort((a, b) => a - b); return s.length ? s[Math.floor(s.length / 2)] : 0; }
assert(p50([{ latency_ms: 100 }, { latency_ms: 900 }, { latency_ms: 300 }]) === 300, "p50 is the real median");
assert(p50([]) === 0, "p50 of nothing is 0, never invented");

// --- active spans: open-nesting from the ledger ---
function activeSpans(evts, running) {
  let open = running ? 1 : 0;
  for (const e of evts) {
    if (e.type === "delegation") open += 1;
    if (e.type === "delegation_complete" || e.type === "delegation_return") open = Math.max(running ? 1 : 0, open - 1);
  }
  return open;
}
assert(activeSpans([], true) === 1, "running root counts as one open span");
assert(activeSpans([{ type: "delegation" }], true) === 2, "open delegation nests: 2 active spans");
assert(activeSpans([{ type: "delegation" }, { type: "delegation_return" }], true) === 1, "closed delegation returns to 1");

// --- waterfall tree: nest by parent_span_id, never flatten a child above its parent ---
function treeDepth(rows, id, depth = 0) {
  const row = rows.find(r => r.span_id === id);
  if (!row || !row.parent_span_id) return depth;
  return treeDepth(rows, rows.find(r => r.span_id === row.parent_span_id)?.span_id ?? null, depth + 1);
}
{
  const rows = [
    { span_id: "a", parent_span_id: null },
    { span_id: "b", parent_span_id: "a" },
    { span_id: "c", parent_span_id: "b" },
  ];
  assert(treeDepth(rows, "a") === 0 && treeDepth(rows, "b") === 1 && treeDepth(rows, "c") === 2, "span tree nests by parent_span_id");
}

// --- cost is delegated, never computed locally ---
const screen = readFileSync("src/screens/LiveAgentViewScreen.jsx", "utf8");
const consoleSrc = readFileSync("src/components/HarnessTraceConsole.jsx", "utf8");

assert(!/agent\.invoke|llm\.usage|orchestration\.(reduce|begin|complete)/.test(all), "no invented event vocabulary");
assert(!/#[0-9a-fA-F]{3,8}\b/.test(all.replace(/\/\/[^\n]*/g, "")), "zero literal hex in touched files");
assert(!/Math\.random/.test(all), "no Math.random");
assert(!/patterns_used/.test(all), "pattern pill never reads frozen patterns_used");
assert(/ai_activity_log/.test(screen + consoleSrc), "rows come from ai_activity_log");
assert(!/realtime|\.channel\(/i.test(all), "no Realtime subscription — bounded polling only");

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS PASS: NO — fix before committing"); process.exit(1); }
