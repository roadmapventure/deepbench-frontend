## 8. NODE.JS TEST (full code — save as `test-mob-8.mjs`, delete before commit)

**Bind to the committed bytes** (`SES-69`), and **blank template-literal interpolations before matching any CSS rule body** (`SES-72` — `NET_CSS` is a template literal, so `${mono}` contains a literal `}` that terminates a naive `[^}]*` match and can produce a *vacuous* pass).

```js
// S-MOB-8 inline test: the view rename, read from the real source.
import { readFileSync } from "node:fs";
const NET = "src/components/AgentNetwork.jsx";
const net = readFileSync(NET, "utf8");
// SES-72: neutralise ${...} so rule-body matches can't terminate inside an interpolation.
const flat = net.replace(/\$\{[^}]*\}/g, "___");
let pass = true;
const check = (name, cond, extra = "") => {
  if (!cond) { pass = false; console.error(`FAIL ${name} ${extra}`); } else console.log(`ok ${name}`);
};

// --- the rename itself -----------------------------------------------------------------
check("button label reads Single",           />Single<\/button>/.test(net));
check("old Active label is gone",            !/>Active<\/button>/.test(net));
check("click sets the single view",          /setMobileView\("single"\)/.test(net));
check("no setMobileView\\(\"active\"\\) left", !/setMobileView\("active"\)/.test(net));
check("on-class compares to single",         /mobileView === "single"/.test(net));
check("no mobileView === \"active\" left",   !/mobileView === "active"/.test(net));

// --- what must NOT have changed --------------------------------------------------------
check("Bench label intact",                  />Bench<\/button>/.test(net));
check("bench comparison intact",             /mobileView === "bench"/.test(net));
check("Bench is still the default view",     /useState\(\s*"bench"\s*\)/.test(net));
check("node width still 84px",               /\.lav-mnode\{[^}]*width:84px/.test(flat));
check("mobileSlot untouched (hub 50/52)",    /return \{ x: 50, y: 52 \}/.test(net));
check("role line still one line",
  /\.lav-mrole\{[^}]*white-space:nowrap/.test(flat) && /\.lav-mrole\{[^}]*text-overflow:ellipsis/.test(flat));
check("toggle pill still 19px",              /\.lav-mseg\{[^}]*height:19px/.test(flat));

// --- the working-agent concept must keep its own name ----------------------------------
check("net.activeId preserved",              /activeId/.test(net));
check("is-active class preserved",           /is-active/.test(net));
check("assemblyActive preserved",            /assemblyActive/.test(net));
check("no stray view-word collision",        !/mobileView === "activeId"/.test(net));

// --- SES-72 self-check: prove the interpolation-blanking actually matters ---------------
check("SES-72 guard is load-bearing",
  /\$\{/.test(net) && flat.indexOf("___") !== -1,
  "expected interpolations present in source and blanked in flat");
console.log(pass ? "ALL TESTS PASS" : "TESTS FAILED");
if (!pass) process.exit(1);
```
