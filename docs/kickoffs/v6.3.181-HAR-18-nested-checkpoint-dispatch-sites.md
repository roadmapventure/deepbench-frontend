## 2. CONTEXT

`runCapability()` can legitimately return `{status:'in_progress', job_id}` instead of a result — that is the durable-checkpoint mechanism (budget starvation today; every transient-failure recovery after S-HAR-17b ships). Four places in `execute.js` invoke a nested `runCapability()`. Two of them handle an `in_progress` return correctly by converting it into a `nested_checkpoint` outcome (the `delegationRequired` auto-resolve at L592 and the plain `delegate_to_agent` at L635). The other two do not:

- **Broker site (L547–551):** `request_help`'s dispatch to the project-manager broker. An `in_progress` return flows into the `lastHelpSelection` rebuild (L558, reads `.content` → nulls), then either the L569 throw (`delegationRequired` path — masking a healthy checkpoint as `no valid recommended_agent_id`) or the L650 `tool_result` append (plain path — the model literally receives `{"status":"in_progress"...}` as its help result).
- **Critique site (~L809–823):** the pre-confirmation critique dispatch. An `in_progress` return is stored **verbatim** as the `critique` field of a `pending_confirmations` row — the user gets a confirmation card whose critique is a job stub, and the checkpointed critique job is orphaned.

This is live today (a budget checkpoint inside either chain corrupts), and becomes routine once S-HAR-17b makes every transient failure checkpoint. Fix ordering is deliberate: this ships before the recovery seam.

```js
// test-har-17a-api.mjs — HAR-18 Category L live test (run from the worktree root)
// Proves: (1) a broker-chain checkpoint now yields a resumable in_progress outer job instead of
// the L569 throw / stringified-blob tool_result; (2) resume drives it to a real terminal result;
// (3) a critique-chain checkpoint yields pending_confirmation with critique === null, never a stub.
import { runCapability, resumeCapability, __setTestBudgetMs } from './api/capabilities/execute.js';

const SUPA = process.env.SUPABASE_URL, KEY = process.env.SUPABASE_SERVICE_KEY;
const H = { apikey: KEY, Authorization: `Bearer ${KEY}` };
const sb = async (path) => (await fetch(`${SUPA}/rest/v1/${path}`, { headers: H })).json();

let pass = 0, fail = 0;
const ok = (cond, label) => { if (cond) { pass++; console.log('  PASS', label); } else { fail++; console.error('  FAIL', label); } };

// --- Resolve real config rows dynamically (no hardcoded slugs; schema verified via REST) ---
const delegReq = (await sb(`intent_skill_config?delegation_required=eq.true&select=capability_slug,intent_slug,agent_id&limit=1`))[0]
  ?? (await sb(`skill_profiles?delegation_required=eq.true&select=capability_slug,intent_slug&limit=1`))[0];
// ^ If neither table matches the live schema, STOP and locate the table that carries
//   delegation_required / requires_human_confirmation / critique_capability_slug (read
//   api/_lib/handlers/db-assembly.js for the real source of these traits), then fix the two
//   queries in this test. Do not hardcode a slug from memory.
const critiqued = (await sb(`intent_skill_config?critique_capability_slug=not.is.null&select=capability_slug,intent_slug,agent_id&limit=1`))[0];
if (!delegReq) { console.error('no delegation_required intent found — fix the lookup, see comment'); process.exit(1); }

// --- Test 1: broker checkpoint -> in_progress outer, waiting_on_job_id set, resume completes ---
{
  let flipped = false;
  const onEvent = (evt) => {
    if (!flipped && evt.type === 'delegation' && evt.viaTool === 'request_help') {
      flipped = true; __setTestBudgetMs(0); // nested broker runLoop checkpoints at its first top-of-loop check
    }
  };
  const res = await runCapability({
    capability_slug: delegReq.capability_slug, intent_slug: delegReq.intent_slug,
    agent_id: delegReq.agent_id, task_context: { question: 'HAR-18 L-test: broker checkpoint round trip' },
    tenant_id: 'global', _onEvent: onEvent,
  });
  __setTestBudgetMs(null);
  ok(res?.status === 'in_progress' && !!res.job_id, 'T1a outer returns in_progress (not the L569 throw)');
  const [row] = await sb(`durable_hops?id=eq.${res.job_id}&select=status,pending_delegation`);
  ok(row?.status === 'in_progress' && !!row?.pending_delegation?.waiting_on_job_id, 'T1b pending_delegation.waiting_on_job_id persisted');
  // Resume until terminal (fresh budgets each leg; cap generous over MAX_LOOP_DEPTH)
  let cur = res, hops = 0;
  while (cur?.status === 'in_progress' && hops++ < 12) cur = await resumeCapability({ job_id: cur.job_id ?? res.job_id });
  ok(cur && cur.status !== 'in_progress', 'T1c resume chain reaches a terminal result');
  ok(JSON.stringify(cur).includes('"status":"in_progress"') === false, 'T1d no in_progress blob embedded in terminal result');
}

// --- Test 2: critique checkpoint -> pending_confirmation with critique === null ---
if (critiqued) {
  let flipped = false;
  const onEvent = (evt) => {
    if (!flipped && evt.type === 'delegation' && evt.viaTool === 'critique') { flipped = true; __setTestBudgetMs(0); }
  };
  const res = await runCapability({
    capability_slug: critiqued.capability_slug, intent_slug: critiqued.intent_slug,
    agent_id: critiqued.agent_id, task_context: { question: 'HAR-18 L-test: critique checkpoint degrade' },
    tenant_id: 'global', _onEvent: onEvent,
  });
  __setTestBudgetMs(null);
  ok(res?.status === 'pending_confirmation', 'T2a confirmation gate still fires');
  ok(res?.critique === null, 'T2b critique degraded to null (never a job stub)');
  const [conf] = await sb(`pending_confirmations?id=eq.${res.confirmation_id}&select=critique,status`);
  ok(!conf?.critique || conf.critique === null, 'T2c persisted row carries no in_progress stub');
  // Cleanup: reject the test confirmation so no live pending card lingers
  await fetch(`${SUPA}/rest/v1/pending_confirmations?id=eq.${res.confirmation_id}`, {
    method: 'PATCH', headers: { ...H, 'Content-Type': 'application/json' }, body: JSON.stringify({ status: 'rejected' }),
  });
} else {
  console.log('  SKIP T2 (no critique-configured intent found live) — record this in the completion report');
}

console.log(fail === 0 ? `ALL TESTS PASS (${pass})` : `FAILURES: ${fail}`);
process.exit(fail === 0 ? 0 : 1);
```
