### Task 2 — `useHarnessStream.js`: capture the family into the ledger

- Append via `appendEvent(buildHopEvent(evt.type, evt.agentId ?? null, { …fields }, durationMs, {}))` — **plain append, never `logEvent`**: an assembly frame awaits nothing and replaces nothing, and `logEvent`'s pending-row path would let it claim an open "X is routing to Y" row for the same agent (the LAV-1d lesson, same reasoning verbatim).
- Data fields carried onto the hop event (real values only, omit what the frame lacks): `message` (composed below), `work`, `source`, `matchCount`, `tokens`, `model`, `forAgentId`, `trace_id`, `span_id`.

  - `work: 'reflect' | 'synthesis'` → `` `${evt.agentId ?? "—"} · ${evt.work}${evt.type === 'assembly_work_complete' && evt.tokens != null ? ` · ${evt.tokens} tok` : ''}` ``
- Do **not** call `setPrompt`, `writeStatus`, or touch `pendingDelegationsRef`. The scripted status line stays as-is; the canvas treatment is d's job.

// test-aa-179a.mjs — pure Node, no app imports.
import { readFileSync } from "node:fs";
let pass = 0, fail = 0;
const assert = (c, l) => { if (c) { pass++; console.log(`PASS: ${l}`); } else { fail++; console.log(`FAIL: ${l}`); } };

const hook = readFileSync("src/hooks/useHarnessStream.js", "utf8");
const chi = readFileSync("src/screens/MarketIntelligenceScreen.jsx", "utf8");

// --- CHI guard covers all three non-delegation types ---
const guard = chi.slice(chi.indexOf("onDelegationProgress"), chi.indexOf("onDelegationProgress") + 2600);
for (const t of ["prompt_assembled", "assembly_work'", "assembly_work_complete"]) {
  assert(guard.includes(t), `CHI guard names ${t}`);
}

// --- hook handles the family, via plain append, never logEvent's pending path ---
const aIdx = hook.indexOf("assembly_work");
assert(aIdx > -1, "hook has an assembly branch");
const branch = hook.slice(aIdx - 1200, aIdx + 1600);
assert(branch.includes("appendEvent"), "assembly branch uses appendEvent (plain append)");
assert(!/pendingDelegationsRef/.test(branch), "assembly branch never touches the pending map");
assert(branch.includes("resolveDelegationDuration"), "arrival-delta duration, not resolveEventDuration");
assert(!/setPrompt|writeStatus/.test(branch), "no prompt/status writes from assembly frames");

// --- message composition mirror (real fields only) ---
const msg = (evt) => evt.work === "fetch"
  ? `${evt.agentId ?? "—"} · ${evt.source} · ${evt.matchCount ?? 0} chunks`
  : `${evt.agentId ?? "—"} · ${evt.work}${evt.type === "assembly_work_complete" && evt.tokens != null ? ` · ${evt.tokens} tok` : ""}`;
assert(msg({ work: "fetch", agentId: "eleanor", source: "the_library", matchCount: 3 }) === "eleanor · the_library · 3 chunks", "fetch message");

assert(msg({ work: "fetch", agentId: null, source: "roster", matchCount: 0 }) === "— · roster · 0 chunks", "null-safe");

// --- no third spelling of the family anywhere in either file ---
assert(!/assembly_started|assemblyWork|assembly_complete[^_]/.test(hook + chi), "no variant spellings");

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS PASS: NO — fix before committing"); process.exit(1); }
