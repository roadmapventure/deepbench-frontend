```js
// test-log-93.mjs — S-LOG-93 (v6.3.173) — Active Agents tile
// Categories: A (data shape), M-lite (the tile's source of truth is the real roster file, not a literal)
import { readFileSync } from 'node:fs';

let pass = 0, fail = 0;
const ok = (name, cond, detail = '') => {
  if (cond) { pass++; console.log(`  PASS  ${name}`); }
  else { fail++; console.log(`  FAIL  ${name} ${detail}`); }
};

const agentsSrc = readFileSync('src/data/agents.js', 'utf8');
const panelSrc = readFileSync('src/components/AIActivityPanel.jsx', 'utf8');

// A. Roster shape: count top-level agent entries (id+name pairs on one line, the file's fixed idiom)
const rosterCount = (agentsSrc.match(/^\s*id:\s*["'][a-z]+["'],\s*name:/gm) || []).length;
ok(`roster parse finds a plausible count (${rosterCount} >= 20)`, rosterCount >= 20);

// M. The panel imports AGENTS and uses AGENTS.length — no hardcoded roster number
ok('panel imports AGENTS from ../data/agents.js', /import\s*\{\s*AGENTS\s*\}\s*from\s*["']\.\.\/data\/agents\.js["']/.test(panelSrc));
ok('tile reads AGENTS.length', /\[\s*["']Active Agents["']\s*,\s*AGENTS\.length\s*\]/.test(panelSrc));
ok('no hardcoded 22 in the tile entry', !/\[\s*["']Active Agents["']\s*,\s*22\s*\]/.test(panelSrc));

// A. Tile order: Active Agents sits between Services and Patterns Logged
const strip = panelSrc.slice(panelSrc.indexOf('"Total Calls"'), panelSrc.indexOf('"Models in Use"'));
const order = ['"Total Calls"', '"Total Cost"', '"Services"', '"Active Agents"', '"Patterns Logged"'];
let last = -1, ordered = true;
for (const t of order) { const i = strip.indexOf(t); if (i < last) ordered = false; last = i; }
ok('tile order: Calls, Cost, Services, Active Agents, Patterns Logged, Models in Use', ordered && strip.includes('"Active Agents"'));

console.log(fail === 0 ? `\nALL TESTS PASS (${pass})` : `\n${fail} FAILURES (${pass} passed)`);
if (fail > 0) process.exit(1);
```
