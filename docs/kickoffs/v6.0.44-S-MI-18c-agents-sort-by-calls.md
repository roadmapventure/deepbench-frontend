```js
// test-s-mi-18c.mjs — Category B (sort logic). Pure Node.js, no app imports.

let pass = 0, fail = 0;
function check(name, cond) {
  if (cond) { pass++; console.log(`PASS: ${name}`); }
  else { fail++; console.log(`FAIL: ${name}`); }
}

const PROPOSED_MI_AGENT_IDS = ["marcus", "priya", "nadia", "owen", "sam", "elena", "michelle", "alex", "dan", "eleanor"];

function sortActiveIds(agentActivity) {
  return PROPOSED_MI_AGENT_IDS
    .filter(id => agentActivity[id]?.calls > 0)
    .sort((a, b) => (agentActivity[b]?.calls || 0) - (agentActivity[a]?.calls || 0));
}

// Real-shaped sample matching this session's own live-verified S-MI-18b numbers
const realShape = {
  marcus: { calls: 205 }, priya: { calls: 112 }, nadia: { calls: 21 }, owen: { calls: 66 },
  sam: { calls: 20 }, elena: { calls: 13 }, michelle: { calls: 467 }, alex: { calls: 86 },
  dan: { calls: 32 }, eleanor: { calls: 408 },
};
const sorted = sortActiveIds(realShape);
check("sorted descending by calls", JSON.stringify(sorted) === JSON.stringify(["michelle","eleanor","marcus","priya","alex","owen","dan","nadia","sam","elena"]));
check("all 10 ids present, none dropped", sorted.length === 10);

// Zero-calls agents excluded entirely (Potential bucket), not sorted-to-bottom
const withPotential = { marcus: { calls: 5 }, priya: { calls: 0 }, nadia: undefined };
check("zero/missing-calls agents excluded from activeIds", sortActiveIds(withPotential).length === 1 && sortActiveIds(withPotential)[0] === "marcus");

// Tie handling — stable enough, both present, order between ties not asserted (no spec requirement)
const tied = { marcus: { calls: 10 }, priya: { calls: 10 } };
const tiedResult = sortActiveIds(tied);
check("tied calls: both agents still present", tiedResult.length === 2 && tiedResult.includes("marcus") && tiedResult.includes("priya"));

// Empty activity — no active agents
check("empty activity produces empty activeIds", sortActiveIds({}).length === 0);

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS DID NOT PASS"); process.exit(1); }
console.log("ALL TESTS PASS");
```
