## 2. CONTEXT

Found live during `MI-47`'s own self-QA (this design session, against the real deployed dev URL, asking the real EMEA co-op-budget question) — **not a regression introduced by `MI-47`'s Task 1-4, a pre-existing bug from `S-MI-42` (v6.1.43) that `MI-47`'s own Task 3 (logging every handoff to the permanent Agent Routing drawer) made visible and confusing for the first time.**

Confirmed via the real live SSE event payload (`read_network_requests` on the actual `/api/capabilities/execute` call):
```
data: {"type":"delegation_return","toAgentId":"marcus","toCapabilitySlug":"channel-intelligence","fromAgentId":"michelle","fromCapabilitySlug":"project-manager"}
```
`fromAgentId` is the sub-agent who was helping and is now done (Michelle); `toAgentId` is who control returns to (Marcus, who never left). `describeDelegationEvent()`'s `delegation_return` branch (line 356) reads:
```js
if (evt.type === 'delegation_return') {
  return `${toName} is back — wrapping up…`; // toName = agentById(evt.toAgentId) = "Marcus Webb"
}
```
This produces **"Marcus Webb is back — wrapping up…"** — backwards. Marcus never left; Michelle is the one who was away and is now done. It should read **"Michelle Manning is back — wrapping up…"** (`fromName`, not `toName`).

Live-observed consequence (screenshot-equivalent via `get_page_text` on the real dev URL): `MI-47`'s new drawer row correctly shows Michelle Manning's avatar/name as the row's primary agent (`MI-47`'s `agentId: evt.fromAgentId` is correct), but the row's own summary text underneath says "Marcus Webb is back — wrapping up…" — an avatar/text mismatch sitting permanently in the drawer, where before `MI-47` it only ever flashed briefly in the ephemeral chat line and was easy to miss.

**Why this slipped through `S-MI-42`'s own test:** that session's Node test asserted `describeDelegationEvent({ type:'delegation_return', fromAgentId:'michelle', toAgentId:'marcus' }, AGENTS) === "Marcus Webb is back — wrapping up…"` under the check name *"names who's back, not who left"* — the test's own stated intent contradicts its assertion; it encoded the bug as expected behavior instead of catching it. This session's test corrects the assertion to match the actual intent.

## 5. TASKS (1)

### Task 1 — `MarketIntelligenceScreen.jsx`: fix `describeDelegationEvent()`'s `delegation_return` branch

Line 356, swap `toName` → `fromName`:
```js
function describeDelegationEvent(evt, agents) {
  const agentById = (id) => agents.find(a => a.id === id);
  const fromName = agentById(evt.fromAgentId)?.name || evt.fromAgentId;
  const toName = agentById(evt.toAgentId)?.name || evt.toAgentId;
  if (evt.type === 'delegation_return') {
    return `${fromName} is back — wrapping up…`; // FEATURE: MI-48 -- was toName (named who control
    // returns TO, not who was actually away and is now done) -- confirmed against the real live SSE
    // payload: fromAgentId is the sub-agent who was helping, toAgentId is who receives control back.
  }
  switch (evt.viaTool) {
    case 'request_help':
      return `${fromName} is asking ${toName} who should help…`;
    case 'critique':
      return `${toName} is reviewing ${fromName}'s proposal…`;
    case 'delegate_to_agent':
    default:
      return `${fromName} is routing this to ${toName}…`;
  }
}
```

```js
// test-s-mi-48-delegation-return-naming.mjs — Category B: describeDelegationEvent's
// delegation_return branch names who was actually away and is now done (fromAgentId),
// not who control returns to (toAgentId).

let pass = 0, fail = 0;
function check(name, cond) {
  if (cond) { pass++; console.log(`PASS: ${name}`); }
  else { fail++; console.log(`FAIL: ${name}`); }
}

const AGENTS = [
  { id: 'marcus', name: 'Marcus Webb' }, { id: 'michelle', name: 'Michelle Manning' },
  { id: 'eleanor', name: 'Eleanor Voss' }, { id: 'owen', name: 'Owen Marsh' },
];

function describeDelegationEvent(evt, agents) {
  const agentById = (id) => agents.find(a => a.id === id);
  const fromName = agentById(evt.fromAgentId)?.name || evt.fromAgentId;
  const toName = agentById(evt.toAgentId)?.name || evt.toAgentId;
  if (evt.type === 'delegation_return') {
    return `${fromName} is back — wrapping up…`;
  }
  switch (evt.viaTool) {
    case 'request_help': return `${fromName} is asking ${toName} who should help…`;
    case 'critique': return `${toName} is reviewing ${fromName}'s proposal…`;
    case 'delegate_to_agent': default: return `${fromName} is routing this to ${toName}…`;
  }
}

// Real shape confirmed live: fromAgentId = who was helping (now done), toAgentId = who control returns to.
check("delegation_return names the sub-agent who was away, not the one control returns to", describeDelegationEvent({ type: 'delegation_return', fromAgentId: 'michelle', toAgentId: 'marcus' }, AGENTS) === "Michelle Manning is back — wrapping up…");
check("delegation_return works for any agent pair (Eleanor returning to Marcus, real live-observed case)", describeDelegationEvent({ type: 'delegation_return', fromAgentId: 'eleanor', toAgentId: 'marcus' }, AGENTS) === "Eleanor Voss is back — wrapping up…");
check("unknown fromAgentId falls back to the raw id, never crashes", describeDelegationEvent({ type: 'delegation_return', fromAgentId: 'ghost-agent', toAgentId: 'marcus' }, AGENTS) === "ghost-agent is back — wrapping up…");
check("other delegation branches (request_help/critique/delegate_to_agent) unchanged", (() => {
  const rh = describeDelegationEvent({ type: 'delegation', viaTool: 'request_help', fromAgentId: 'marcus', toAgentId: 'michelle' }, AGENTS);
  const cr = describeDelegationEvent({ type: 'delegation', viaTool: 'critique', fromAgentId: 'marcus', toAgentId: 'owen' }, AGENTS);
  const dg = describeDelegationEvent({ type: 'delegation', viaTool: 'delegate_to_agent', fromAgentId: 'michelle', toAgentId: 'eleanor' }, AGENTS);
  return rh === "Marcus Webb is asking Michelle Manning who should help…"
    && cr === "Owen Marsh is reviewing Marcus Webb's proposal…"
    && dg === "Michelle Manning is routing this to Eleanor Voss…";
})());

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS DID NOT PASS"); process.exit(1); }
console.log("ALL TESTS PASS");
```
