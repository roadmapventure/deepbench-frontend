// test-s-mi-30-agents-drawer.mjs
// Pure Node.js — no app imports. Re-implements the exact logic under test inline.

let pass = 0, fail = 0;
function check(label, cond) {
  if (cond) { pass++; console.log(`PASS: ${label}`); }
  else { fail++; console.log(`FAIL: ${label}`); }
}

// --- rollupBaseline() — copy of the Task 3 implementation ---
function rollupBaseline(stats) {
  if (!stats?.byKind) return null;
  const kinds = Object.values(stats.byKind).filter(k => k.latencyCount > 0);
  if (kinds.length === 0) return null;
  const totalLatency = kinds.reduce((sum, k) => sum + k.totalLatency, 0);
  const latencyCount = kinds.reduce((sum, k) => sum + k.latencyCount, 0);
  const maxLatency = Math.max(...kinds.map(k => k.maxLatency));
  const calls = kinds.reduce((sum, k) => sum + k.calls, 0);
  return { avgLatency: Math.round(totalLatency / latencyCount), maxLatency, calls };
}

// Category B: rollup math across multiple kinds (mirrors Michelle's real baseline shape:
// project-manager wrapper + 2 agent-turn depth buckets)
const michelleStats = {
  byKind: {
    'guardrails-check': { calls: 4, totalLatency: 4728, latencyCount: 4, maxLatency: 2347 },
    'agent-selection-intent': { calls: 4, totalLatency: 27536, latencyCount: 4, maxLatency: 9198 },
  },
};
const r1 = rollupBaseline(michelleStats);
check('rollup: avgLatency is total/count across all kinds', r1.avgLatency === Math.round((4728 + 27536) / 8));
check('rollup: maxLatency is the max across all kinds', r1.maxLatency === 9198);
check('rollup: calls is summed across all kinds', r1.calls === 8);

// Category A: no baseline data at all (Dan Bingham today) -> null, row omitted
check('rollup: undefined stats -> null', rollupBaseline(undefined) === null);
check('rollup: empty byKind -> null', rollupBaseline({ byKind: {} }) === null);

// Category C: string safety — a kind with latencyCount 0 (no latency ever recorded) must not
// be included (would divide by zero / poison the average)
const withZeroLatencyKind = {
  byKind: {
    'x': { calls: 1, totalLatency: 0, latencyCount: 0, maxLatency: null },
    'y': { calls: 2, totalLatency: 1000, latencyCount: 2, maxLatency: 600 },
  },
};
const r2 = rollupBaseline(withZeroLatencyKind);
check('rollup: kinds with latencyCount 0 excluded from average', r2.avgLatency === 500 && r2.calls === 2);

// --- classifyRow() dedup regression — confirms adding 'html-display' to
// CAPABILITY_WRAPPER_TYPES actually suppresses the real double-count found live ---
const CAPABILITY_WRAPPER_TYPES = new Set([
  "channel-intelligence", "hypothesis-evaluation", "quality-gate", "pipeline-triage",
  "memory-consolidation", "data-analysis", "project-manager", "screen-controls",
  "html-display", // <-- this session's fix

const PAIR_WINDOW_MS = 2000;

function classifyRow(row, turnTimestampsByAgent) {
  if (row.ai_type === 'agent-turn') {
    const intentSlug = row.feature ? row.feature.split(':')[1] : null;
    return { kind: intentSlug || row.feature || 'unknown', include: true };
  }
  if (CAPABILITY_WRAPPER_TYPES.has(row.ai_type) && row.feature === 'request-receivable') {
    const rowTime = new Date(row.created_at).getTime();
    const turnTimes = turnTimestampsByAgent.get(row.agent_id) || [];
    const paired = turnTimes.some(t => Math.abs(t - rowTime) < PAIR_WINDOW_MS);
    return { kind: row.ai_type, include: !paired };
  }
  return { kind: row.ai_type, include: true };
}

// Real timestamps from live data (riley, tenant='global', 2026-07-08 15:44:38/41)

function inScope(row, scope) {

  if (scope.aiTypes?.includes(row.ai_type)) return true;
