```js
import { execFileSync } from "node:child_process";
function run(args) {
  const out = execFileSync("node", ["scripts/chi-true-regression.mjs", ...args], { encoding: "utf8", timeout: 900000, cwd: new URL("..", import.meta.url).pathname.replace(/^\/(\w:)/, "$1") });
  const m = out.match(/REPORT_JSON_START\n([\s\S]*?)\nREPORT_JSON_END/);
  if (!m) throw new Error("no report JSON in output:\n" + out.slice(-2000));
  return JSON.parse(m[1]);
}
let passN = 0, failN = 0;
const check = (name, cond) => { cond ? passN++ : failN++; console.log(`${cond ? "PASS" : "FAIL"} ${name}`); };
// T1-T4: one direct case, judged
const r1 = run(["--only", "library-catalog"]);
const c1 = r1.cases[0];
check("T1 one case ran", r1.cases.length === 1 && c1.id === "library-catalog");
check("T2 direct journey", c1.expected_journey === "direct" && ["direct", "direct+review"].includes(c1.actual_journey));
check("T3 judge ran with evidence-bearing verdict", Array.isArray(c1.judge_verdicts) && c1.judge_verdicts.length >= 1 && typeof c1.judge_verdicts[0].pass === "boolean");
check("T4 trace ids captured", c1.trace_ids.length >= 3);
// T5-T6: question extraction integrity (no journey run)
const r2 = run(["--limit", "0"]);
check("T5 baseline holds 24 cases", r2.totals && r2.totals.baseline_cases === 24);
check("T6 extraction counts", r2.totals.extracted === 23);
console.log(`\n${passN} pass / ${failN} fail`);
if (failN) process.exit(1);
```
