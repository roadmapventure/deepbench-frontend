### Task 2 — `src/hooks/useAIActivity.js`: extend `buildActivitySummary()` with a `byPattern` bucket

`buildActivitySummary()` currently builds `d.byKind[kind]` per row (keyed by capability/intent "kind," via `classifyRow()`). Add a parallel `d.byPattern[patternSlug]` bucket, populated in the **same row loop**, gated by the **same `include` flag** `classifyRow()` already returns (so pattern counts respect the same capability-wrapper/agent-turn dedup `byKind` already applies — a row that's excluded from `byKind` because it's a duplicate wrapper row must also be excluded from `byPattern`, or pattern counts would double-count the same real call).

Unlike `byKind` (one row → one kind, mutually exclusive), a single row can carry **multiple** patterns (`row.patterns_used` is an array) — the row contributes to every pattern bucket in that array, not just one.

Exact shape, mirroring `byKind`'s existing fields exactly (`calls`, `totalLatency`, `latencyCount`, `maxLatency`, finalized `avgLatency`):

```js
// inside the same per-row loop, after the existing byKind block, still gated by `if (!include) continue;`
if (Array.isArray(row.patterns_used)) {
  if (!d.byPattern) d.byPattern = {};
  for (const patternSlug of row.patterns_used) {
    if (!d.byPattern[patternSlug]) {
      d.byPattern[patternSlug] = { calls: 0, totalLatency: 0, latencyCount: 0, maxLatency: null };
    }
    const p = d.byPattern[patternSlug];
    p.calls++;
    if (row.latency_ms) {
      p.totalLatency += row.latency_ms;
      p.latencyCount++;
      p.maxLatency = p.maxLatency == null ? row.latency_ms : Math.max(p.maxLatency, row.latency_ms);
    }
  }
}
```

And in the existing finalization pass (where `k.avgLatency` is computed for `byKind`), add the equivalent for `byPattern`:
```js
Object.values(d.byPattern || {}).forEach(p => {
  p.avgLatency = p.latencyCount ? Math.round(p.totalLatency / p.latencyCount) : null;
});
```

`d.byPattern` is `undefined` (not an empty object) for any agent with zero patterned rows — `MI-72b`'s render must handle this, same as it already handles `stats?.byKind` being absent.

## NODE.JS TEST

Run with `node test-mi-72a.mjs`, delete before committing. Pure Node — reimplements the exact logic being tested, no app imports.

```js
// test-mi-72a.mjs
// Category A (data shape), B (logic), C (string safety) — reimplements buildActivitySummary()'s
// new byPattern block exactly as specified in Task 2, tested standalone.

function buildByPattern(rows) {
  const byPattern = {};
  for (const row of rows) {
    if (!Array.isArray(row.patterns_used)) continue;
    for (const patternSlug of row.patterns_used) {
      if (!byPattern[patternSlug]) {
        byPattern[patternSlug] = { calls: 0, totalLatency: 0, latencyCount: 0, maxLatency: null };
      }
      const p = byPattern[patternSlug];
      p.calls++;
      if (row.latency_ms) {
        p.totalLatency += row.latency_ms;
        p.latencyCount++;
        p.maxLatency = p.maxLatency == null ? row.latency_ms : Math.max(p.maxLatency, row.latency_ms);
      }
    }
  }
  Object.values(byPattern).forEach(p => {
    p.avgLatency = p.latencyCount ? Math.round(p.totalLatency / p.latencyCount) : null;
  });
  return byPattern;
}

let pass = 0, fail = 0;
function check(name, cond) {
  if (cond) { pass++; } else { fail++; console.error(`FAIL: ${name}`); }
}

// A. Data shape — a row with 2 patterns contributes to both buckets
{
  const rows = [{ patterns_used: ['rag', 'tool-use'], latency_ms: 1000 }];
  const bp = buildByPattern(rows);
  check('A1: rag bucket exists', bp.rag != null);
  check('A2: tool-use bucket exists', bp['tool-use'] != null);
  check('A3: both buckets got the same call counted', bp.rag.calls === 1 && bp['tool-use'].calls === 1);
}

// B. Logic — avg/max/calls correct across multiple rows for the same pattern
{
  const rows = [
    { patterns_used: ['rag'], latency_ms: 1000 },
    { patterns_used: ['rag'], latency_ms: 3000 },
    { patterns_used: ['rag'], latency_ms: 2000 },
  ];
  const bp = buildByPattern(rows);
  check('B1: calls = 3', bp.rag.calls === 3);
  check('B2: avgLatency = 2000', bp.rag.avgLatency === 2000);
  check('B3: maxLatency = 3000', bp.rag.maxLatency === 3000);
}

// C. String/shape safety — null/undefined/empty patterns_used never throws, contributes nothing
{
  const rows = [
    { patterns_used: null, latency_ms: 500 },
    { patterns_used: undefined, latency_ms: 500 },
    { patterns_used: [], latency_ms: 500 },
    { latency_ms: 500 }, // field absent entirely
  ];
  let threw = false;
  let bp;
  try { bp = buildByPattern(rows); } catch (e) { threw = true; }
  check('C1: does not throw on null/undefined/missing patterns_used', !threw);
  check('C2: produces zero buckets', Object.keys(bp).length === 0);
}

// B. Logic — a row with latency_ms: 0 (falsy) is excluded from latency stats, same as byKind's
// existing `if (row.latency_ms)` guard (not a bug — matches existing byKind semantics exactly).
{
  const rows = [
    { patterns_used: ['guardrails'], latency_ms: 0 },
    { patterns_used: ['guardrails'], latency_ms: 100 },
  ];
  const bp = buildByPattern(rows);
  check('B4: calls counts both rows', bp.guardrails.calls === 2);
  check('B5: latencyCount excludes the zero-latency row', bp.guardrails.latencyCount === 1);
  check('B6: avgLatency reflects only the real-latency row', bp.guardrails.avgLatency === 100);
}

if (fail === 0) {
  console.log(`ALL TESTS PASS (${pass}/${pass})`);
} else {
  console.error(`${fail} FAILED, ${pass} passed`);
  process.exit(1);
}
```
