// FEATURE: CHI-32 patched split (2/8, was CHI-30's 6/4)
function splitRotation(pool, shuffleFn) {
  const shuffled = shuffleFn(pool);
  return { picks: shuffled.slice(0, 2), leftover: shuffled.slice(2) };
}

function shuffleArray(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

  if (cond) { console.log(`PASS: ${name}`); pass++; }
  else { console.log(`FAIL: ${name}`); fail++; }

check("default split: 2 picks, 8 leftover", defaultSplit.picks.length === 2 && defaultSplit.leftover.length === 8);

  "What data is in the library and how can i use it?",
  "What made Crest Wireless's recent upgrade promotion successful, and can we replicate it with other US partners?",
];
check("default visible 3 slots match today's confirmed positions 1-3 exactly", JSON.stringify(visibleQuestions.map(q => q.label)) === JSON.stringify(DEFAULT_VISIBLE_REFERENCE));
const defaultDrawer = [...defaultSplit.leftover, ...FIXED_DRAWER_TAIL];
check("default drawer is always 20 questions", defaultDrawer.length === 20);
check("total visible(3) + drawer(20) = 23", visibleQuestions.length + defaultDrawer.length === 23);

// Category B — Rotated split: correctness + statistical sanity over many trials
let allTrialsValid = true;
let sawAtLeastOneDifferentOrder = false;
const firstTrialPicks = JSON.stringify(splitRotation(ROTATING_POOL, shuffleArray).picks.map(q => q.id));
for (let i = 0; i < 500; i++) {
  const { picks, leftover } = splitRotation(ROTATING_POOL, shuffleArray);
  if (picks.length !== 2 || leftover.length !== 8) allTrialsValid = false;
  const combinedIds = new Set([...picks, ...leftover].map(q => q.id));
  if (combinedIds.size !== 10) allTrialsValid = false; // no lost/duplicated pool item
  if (JSON.stringify(picks.map(q => q.id)) !== firstTrialPicks) sawAtLeastOneDifferentOrder = true;
}
check("500 rotated trials: always 2 picks + 8 leftover, no lost/duplicated pool item", allTrialsValid);
check("500 rotated trials: at least one differs from the first (proves real randomization)", sawAtLeastOneDifferentOrder);

console.log(`\n${pass} passed, ${fail} failed`);
if (fail > 0) { console.log("ALL TESTS FAIL"); process.exit(1); }
console.log("ALL TESTS PASS");
