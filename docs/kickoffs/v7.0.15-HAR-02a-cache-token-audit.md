## 2. CONTEXT
S-HAR-02b/c will enable Anthropic prompt caching platform-wide. Once caching is live, Anthropic's `usage.input_tokens` stops meaning "all input tokens" — it becomes **uncached input only**; the remainder is reported in two fields we currently discard: `usage.cache_creation_input_tokens` (billed at 1.25× the model's input rate) and `usage.cache_read_input_tokens` (billed at ~0.1×). If those fields aren't captured before caching ships, every `ai_activity_log`-derived metric (the AI Audit header, `LOG-81`'s counting, the cost math that justified `HAR-02` itself) silently undercounts. This session is pure audit plumbing: capture the two fields, store them, and price them at read time. **Zero prompt/behavior change** — until b/c ship, both fields simply arrive as 0/undefined and write NULL.

The two columns **already exist** — the design session ran the migration (`har_02a_cache_token_columns`): `ai_activity_log.cache_creation_input_tokens integer NULL`, `ai_activity_log.cache_read_input_tokens integer NULL`. Do not write a migration.

## 8. NODE.JS TEST (full code — save as `test-har-02a.mjs`, delete before committing)
```js
// test-har-02a.mjs — S-HAR-02a (v7.0.15). Categories B, F, G.
import assert from 'node:assert';

// --- B: usage normalization + retry merge (mirror the exact expressions used in request-receivable.js)
function normalizeUsage(llmUsage) {
  return { cache_creation_input_tokens: 0, cache_read_input_tokens: 0, ...(llmUsage || { input_tokens: 0, output_tokens: 0 }) };
}
function mergeUsage(usage, llmUsage) {
  return {
    input_tokens: usage.input_tokens + (llmUsage?.input_tokens || 0),
    output_tokens: usage.output_tokens + (llmUsage?.output_tokens || 0),
    cache_creation_input_tokens: (usage.cache_creation_input_tokens || 0) + (llmUsage?.cache_creation_input_tokens || 0),
    cache_read_input_tokens: (usage.cache_read_input_tokens || 0) + (llmUsage?.cache_read_input_tokens || 0),
  };
}
let u = normalizeUsage({ input_tokens: 100, output_tokens: 50 });
assert.equal(u.cache_creation_input_tokens, 0); assert.equal(u.cache_read_input_tokens, 0);
u = normalizeUsage({ input_tokens: 10, output_tokens: 5, cache_read_input_tokens: 4000 });
assert.equal(u.cache_read_input_tokens, 4000);
u = mergeUsage(u, { input_tokens: 7, output_tokens: 3, cache_creation_input_tokens: 900 });
assert.equal(u.input_tokens, 17); assert.equal(u.cache_creation_input_tokens, 900); assert.equal(u.cache_read_input_tokens, 4000);
u = normalizeUsage(undefined);
assert.equal(u.input_tokens, 0); assert.equal(u.cache_read_input_tokens, 0);

// --- F: logActivity body carries the two columns; omitted params write null
function buildLogBody({ cacheCreationInputTokens = null, cacheReadInputTokens = null } = {}) {
  return { cache_creation_input_tokens: cacheCreationInputTokens, cache_read_input_tokens: cacheReadInputTokens };
}
assert.deepEqual(buildLogBody(), { cache_creation_input_tokens: null, cache_read_input_tokens: null });
assert.deepEqual(buildLogBody({ cacheCreationInputTokens: 12, cacheReadInputTokens: 34 }), { cache_creation_input_tokens: 12, cache_read_input_tokens: 34 });

// --- B: computeCallCost math (rates: illustrative $/Mtok pair; mirror the real table's shape)
function cost(row, inRate, outRate) {
  const cc = row.cache_creation_input_tokens, cr = row.cache_read_input_tokens;
  return (row.input_tokens || 0) / 1e6 * inRate
    + (cc != null ? cc / 1e6 * inRate * 1.25 : 0)
    + (cr != null ? cr / 1e6 * inRate * 0.10 : 0)
    + (row.output_tokens || 0) / 1e6 * outRate;
}
// historical row (no cache fields) — identical to today's math
assert.equal(cost({ input_tokens: 1e6, output_tokens: 0 }, 1, 5), 1);
// cached row: 0.2M uncached + 0.8M read + 0 creation
const c = cost({ input_tokens: 2e5, output_tokens: 0, cache_creation_input_tokens: 0, cache_read_input_tokens: 8e5 }, 1, 5);
assert.ok(Math.abs(c - (0.2 + 0.8 * 0.10)) < 1e-9, `expected 0.28, got ${c}`);

console.log('ALL TESTS PASS');
```
