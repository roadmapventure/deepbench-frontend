### Task 3 — `promptPreviewText`: loop the OUTPUT FORMAT skip
Replace the single-block skip with a loop that strips every consecutive leading OUTPUT FORMAT block (there can legitimately be more than one from a still-misconfigured capability elsewhere; the fix must not assume exactly one or exactly two):
```js
// FEATURE: LAV-7a -- loops instead of stripping one block. Found live 2026-08-01: Alex Reeves's
// screen-controls capability had TWO Format Skill Profiles attached in Supabase (a duplicate-
// attachment data bug, fixed this session -- capability_skill_profiles row d1bff23b-... deleted),
// so a single-strip skip left the second copy showing. The loop is defense in depth: it makes the
// preview correct regardless of how many leading OUTPUT FORMAT blocks the real prompt happens to
// carry, without asserting there will only ever be one or two.
export function promptPreviewText(text) {
  if (typeof text !== "string" || !text) return text;
  const blocks = text.split("\n\n---\n\n");
  let start = 0;
  while (start < blocks.length && /^===\s*OUTPUT FORMAT\s*===/.test(blocks[start])) start += 1;
  return start > 0 ? blocks.slice(start).join("\n\n---\n\n") : text;
}
```
(Signature and call site unchanged — this replaces the function body only.)

// S-LAV-7a inline test: promptPreviewText loop + answerQa/answerText mapping. Pure Node (.jsx not
// importable) -- logic mirrored exactly from the kickoff spec; committed implementation must match.
function promptPreviewText(text) {
  if (typeof text !== "string" || !text) return text;
  const blocks = text.split("\n\n---\n\n");
  let start = 0;
  while (start < blocks.length && /^===\s*OUTPUT FORMAT\s*===/.test(blocks[start])) start += 1;
  return start > 0 ? blocks.slice(start).join("\n\n---\n\n") : text;
}
function mapAnswer(result) {
  const answerQa = result && result.kind === "qa"
    ? { ...result, keyDataPoints: result.key_data_points }
    : null;
  const answerText = result && result.kind !== "qa" ? (result.text || null) : null;
  return { answerQa, answerText };
}
const SEP = "\n\n---\n\n";
const fmt = "=== OUTPUT FORMAT ===\nOutput type: json";
const intent = "=== INTENT ===\nAnswer sourced questions";
let pass = true;
const check = (name, got, want) => {
  const ok = JSON.stringify(got) === JSON.stringify(want);
  if (!ok) { pass = false; console.error(`FAIL ${name}\n  got:  ${JSON.stringify(got)}\n  want: ${JSON.stringify(want)}`); }
  else console.log(`ok ${name}`);
};
check("single leading block still skipped", promptPreviewText(fmt + SEP + intent), intent);
check("TWO leading blocks both skipped (the live bug case)", promptPreviewText(fmt + SEP + fmt + SEP + intent), intent);
check("three leading blocks all skipped", promptPreviewText(fmt + SEP + fmt + SEP + fmt + SEP + intent), intent);
check("no format block -> unchanged", promptPreviewText(intent), intent);
check("format-only prompt -> unchanged (never blank)", promptPreviewText(fmt), fmt);
check("mid-prompt format block NOT skipped", promptPreviewText(intent + SEP + fmt), intent + SEP + fmt);
check("null passthrough", promptPreviewText(null), null);
check("qa result maps keyDataPoints", mapAnswer({ kind: "qa", headline: "H", key_data_points: [1,2] }),
  { answerQa: { kind: "qa", headline: "H", key_data_points: [1,2], keyDataPoints: [1,2] }, answerText: null });
check("qa_failed maps to plain text only", mapAnswer({ kind: "qa_failed", text: "blocked" }),
  { answerQa: null, answerText: "blocked" });
check("null result maps to both null", mapAnswer(null), { answerQa: null, answerText: null });
console.log(pass ? "ALL TESTS PASS" : "TESTS FAILED");
