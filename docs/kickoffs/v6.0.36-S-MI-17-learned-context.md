Same direct-frontend-Supabase-read shape as the existing `useAgentUsageCounts()` in this file (reused precedent — no new pattern). Add below `useAgentUsageCounts()`:

```js
// FEATURE: MI-17 — Learned Context drawer data source. Direct frontend Supabase read,
// same precedent as useAgentUsageCounts() above — no api/ route, RLS is disabled platform-wide
// (AI-42, known accepted gap), this inherits that existing exposure, does not introduce a new one.
// data_room_tag is hardcoded to this demo's single tenant (confirmed live in Supabase: every
// the_library/the_reasoning row and every MI-scoped agent's data_room_access uses this one tag;
// no other data room exists in this build) — same implicit single-tenant scoping the rest of the
// MI screen already assumes without threading a param.
const APPLE_DATA_ROOM_TAG = "apple-cso-data-room";

export function useLearnedContext() {
  const [entries, setEntries] = useState([]);

  useEffect(() => {
    supabase
      .from('the_reasoning')
      .select('id,agent_id,confidence,source_question,content,created_at')
      .eq('data_room_tag', APPLE_DATA_ROOM_TAG)
      .eq('status', 'active')
      .order('created_at', { ascending: false })
      .limit(20)
      .then(({ data, error }) => {
        if (error || !data) return;
        setEntries(data);
      });
  }, []);

  return entries;
}
```

// ---- shapeForLog reuse — same function already shipped in MarketIntelligenceScreen.jsx this
// session's day (S-ARCH-PIPELINE-LOG-01/AA-125); re-implemented here byte-for-byte since this is
// a pure Node test with no app imports, to confirm the exact behavior this session relies on. ----
function shapeForLog(text, maxLen = 140) {
  const s = (text || "").trim();
  if (!s) return "";
  const firstSentence = s.match(/^[^.!?]*[.!?]/);
  if (firstSentence && firstSentence[0].trim().length <= maxLen) {
    return firstSentence[0].trim();
  }
  if (s.length <= maxLen) return s;
  return `${s.slice(0, maxLen).trim()}…`;
}
const longContent = "Japan's +14.57% YoY revenue growth in 2025 (fastest among all Apple GEOs) is attributed to two factors: an objectively short upgrade cycle of 26 months and documented channel-program depth.";
check("shapeForLog(content, 220) returns the first sentence when it fits under 220", shapeForLog(longContent, 220) === "Japan's +14.57% YoY revenue growth in 2025 (fastest among all Apple GEOs) is attributed to two factors: an objectively short upgrade cycle of 26 months and documented channel-program depth.");
check("shapeForLog handles empty/undefined content without throwing", shapeForLog(undefined) === "" && shapeForLog("") === "");
const sourceQ = "Japan is Apple's fastest-growing GEO in 2025 — what is driving that?";
check("shapeForLog(source_question, 100) fits within bound or truncates with ellipsis", shapeForLog(sourceQ, 100).length <= 101);

check("agentById returns undefined (not throw) for an unknown agent_id, falls back to raw id", agentById("unknown-agent") === undefined);
