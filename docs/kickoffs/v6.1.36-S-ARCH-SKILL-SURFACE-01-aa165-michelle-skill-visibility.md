## NODE.JS TEST

Pure Node, talks to real Supabase and calls the real broker function. Save as `test-s-arch-skill-surface-01.mjs`, run with `node --env-file=.env.local test-s-arch-skill-surface-01.mjs`, delete before committing.

```js
// test-s-arch-skill-surface-01.mjs -- AA-165 verification (pure Node)
import { getRosterCandidates } from './lib/project-manager.js';

let failures = 0;
function check(label, cond) {
  if (cond) { console.log(`PASS: ${label}`); }
  else { console.error(`FAIL: ${label}`); failures++; }
}

async function checkSupabaseRow() {
  const SUPABASE_URL = process.env.SUPABASE_URL;
  const SUPABASE_SERVICE_KEY = process.env.SUPABASE_SERVICE_KEY;
  if (!SUPABASE_URL || !SUPABASE_SERVICE_KEY) {
    console.error('FAIL: SUPABASE_URL/SUPABASE_SERVICE_KEY not set -- cannot verify');
    process.exit(1);
  }
  const headers = { apikey: SUPABASE_SERVICE_KEY, Authorization: `Bearer ${SUPABASE_SERVICE_KEY}` };
  const res = await fetch(`${SUPABASE_URL}/rest/v1/skill_profiles?slug=eq.qa-answer-format&select=objective,output_desc,method,traits`, { headers });
  const [row] = await res.json();
  check('qa-answer-format row found', !!row);
  check('objective no longer null', typeof row?.objective === 'string' && row.objective.length > 0);
  check('output_desc no longer null', typeof row?.output_desc === 'string' && row.output_desc.length > 0);
  check('method unchanged (still starts with "Reformat")', typeof row?.method === 'string' && row.method.startsWith('Reformat'));

  const rileyRes = await fetch(`${SUPABASE_URL}/rest/v1/skill_profiles?slug=eq.html-display-format&select=objective,output_desc,method`, { headers });
  const [rileyRow] = await rileyRes.json();
  check('Riley\'s html-display-format completely untouched', rileyRow?.objective === 'Transform content specialist prose and structured data into semantically correct, visually considered web HTML ready for direct rendering.');
}

async function checkRosterContext() {
  const result = await getRosterCandidates({ requestingAgentId: 'michelle' });
  check('roster broker returned a context string', typeof result.context === 'string' && result.context.length > 0);
  check('Alex\'s qa-answer-format objective now appears in Michelle\'s context', result.context.includes('Present a completed, review-passed answer as clean, structured in-app content'));
  check('Riley\'s html-display-format objective also appears (broker fix is symmetric, not Alex-only)', result.context.includes('Transform content specialist prose and structured data into semantically correct'));
  check('existing intent_slugs mechanism still present (AA-108, unaffected)', result.context.includes('intents:'));
  check('existing format_output_types mechanism still present (AA-129, unaffected)', result.context.includes('format_output_types:'));
}

await checkSupabaseRow();
await checkRosterContext();

if (failures > 0) { console.error(`\n${failures} FAILURE(S)`); process.exit(1); }
console.log('\nALL TESTS PASS');
```
