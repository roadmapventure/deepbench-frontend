```js
// test-abt-1a.mjs — S-ABT-1a verification (Categories B, C, F, M-source checks)
import { readFileSync } from "fs";
import assert from "assert";

const src = readFileSync("src/components/AboutPanel.jsx", "utf8");
const tokens = readFileSync("src/tokens.js", "utf8");
const regr = readFileSync("tests/regression/LOG-36-pattern-display-sources.js", "utf8");
let pass = 0;
const ok = (cond, msg) => { assert.ok(cond, msg); console.log("PASS:", msg); pass++; };

// — Source-shape checks —
ok(!/PATTERN_CATALOG|SERVICE_CATALOG/.test(src), "AboutPanel has zero catalog references (LOG-70 consumer retired)");
ok(/useState\(\s*["']architecture["']\s*\)/.test(src), "default tab is architecture");
ok(!/RoadmapTab/.test(src), "RoadmapTab deleted entirely");
for (const t of ["PurposeTab", "QuickStartTab", "RevenueTab", "ShowcaseTab", "JohnTab"])
  ok(new RegExp(`function ${t}\\b`).test(src), `${t} kept in file (hidden, not deleted)`);
const tabsArr = (src.match(/const TABS = \[[\s\S]*?\];/) || [""])[0];
ok(/architecture/.test(tabsArr) && !/(purpose|quickstart|revenue|roadmap|showcase|john)/.test(tabsArr),
  "TABS renders only the architecture tab");
ok(/ai_pattern_classification_rollup/.test(src), "patterns tile reads the Log Displayer rollup");
ok(/platform_schema_stats/.test(src), "schema RPC wired");
ok(/platform_stats/.test(src), "platform_stats row wired");
ok(/count:\s*['"]exact['"]/.test(src), "count queries use head/exact count, not row transfer");
const hexInPanel = (src.match(/#[0-9a-fA-F]{6}\b/g) || []).filter(h => !/rgba/.test(h));
ok(hexInPanel.length === 0, "zero hex literals in AboutPanel (design-tokens rule)");
ok(/navyTextHi/.test(tokens) && /navyTextLo/.test(tokens), "navy text tokens promoted into tokens.js");
ok(!/AboutPanel\.jsx/.test(regr.split("const ALLOWANCES")[1].split("];")[0]), "LOG-36 regression allowance for AboutPanel removed");

// — Live Supabase checks (Category F) — anon key, same client path the browser uses —
const url = process.env.VITE_SUPABASE_URL, key = process.env.VITE_SUPABASE_ANON_KEY;
assert.ok(url && key, "VITE_SUPABASE_URL / VITE_SUPABASE_ANON_KEY must be in .env.local");
const rest = async (path, opts = {}) => {
  const r = await fetch(`${url}/rest/v1/${path}`, { ...opts, headers: { apikey: key, Authorization: `Bearer ${key}`, ...(opts.headers || {}) } });
  return r;
};
const stats = await (await rest("platform_stats?id=eq.1&select=*")).json();
ok(Array.isArray(stats) && stats.length === 1, "platform_stats row id=1 readable via anon");
for (const col of ["lines_of_code", "governance_docs", "commits_dev", "dev_toolchain_services"])
  ok(Number.isInteger(stats[0][col]) && stats[0][col] > 0, `platform_stats.${col} is a positive int (${stats[0][col]})`);
const rpc = await (await rest("rpc/platform_schema_stats", { method: "POST", headers: { "Content-Type": "application/json" }, body: "{}" })).json();
const row = Array.isArray(rpc) ? rpc[0] : rpc;
ok(row.table_count > 0 && row.column_count > 0 && row.rag_table_count > 0, `platform_schema_stats() live: ${row.table_count}/${row.column_count}/${row.rag_table_count}`);
const cnt = await rest("ai_pattern_classification_rollup?select=pattern_slug", { headers: { Prefer: "count=exact", Range: "0-0" } });
ok(cnt.headers.get("content-range") !== null, "rollup count query works via anon");

console.log(`ALL TESTS PASS (${pass})`);
```
