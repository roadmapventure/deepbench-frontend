## TASKS (4)

1. **`src/components/AssemblyView.jsx` (new) — the fold + the drawer.**
   `buildAssemblyStages(events, eventTimes, { runStartedAt, agents, running, terminal })` —
   pure, exported, same input contract as `buildRunTaskEntries`:
   - **Completions open/fill sections.** Stage by declared work: `qa_answer`/
     `hypothesis_generation`/`patch_proposed` → `draft`; `proofreader`/`failure_triage` →
     `verification`; `display_format` → `final`; `assembly_work_complete` `work==='fetch'` →
     `evidence` (but see nesting below); assembly `reflect`/`synthesis` → `draft` sub-entry;
     `error` → `error`. Routing/delegation/`question_boundary`/`transaction_complete` types
     build no section (the canvas tells choreography). A completion fills its section with the
     exported `deriveDid`/`deriveContent` line, the agent (name+role via the roster map, id
     verbatim when unresolved), and the `formatTimeSignature` time signature.
   - **Start signals open ghost sections.** A `prompt_assembled` or `delegation`-start entry
     whose `toIntentSlug` maps via `STAGE_OF_INTENT` (or whose assembly `work` is
     `reflect`/`synthesis` → `draft`) opens a ghost for that stage: worker's agent row +
     shimmer bars + an in-progress line (`deriveAsked` text where derivable, else the stage
     label alone). A start whose slug maps to nothing opens an UNLABELED in-progress ghost
     (mono label `IN PROGRESS`) — never a guessed stage. `agent-selection-intent` and other
     routing work opens nothing. A ghost resolves when a completion for the same stage (or
     same agent, whichever lands) arrives — the completion's own derived stage always wins
     over the ghost's prediction. Ghosts unresolved at terminal are dropped, never kept as
     fake stages.
   - **Fetch nesting (span parentage, §19r made visible).** A fetch completion nests as a
     sub-entry inside an OPEN non-draft section when its `parent_span_id` matches that
     section's span identity, or (fallback when spans don't resolve) its `forAgentId` matches
     that section's agent. Otherwise it lands in the top-level `evidence` section (created on
     first fetch). Sub-entry renders the smaller agent row + `deriveDid` line, per the mock.
   - **Order and scroll.** Sections newest-first by open time (§19n arrival-order posture;
     the mock's Frame 2/3 stacks). On any entries change, pin `scrollTop` to 0 (John's spec:
     the most recent event is always in view). Drawer chrome: title
     `` `${ASSEMBLY_TITLE} · ${plural(n, "stage")}` `` (reuse the feed's `plural`; import or
     re-export from `RunTasks.jsx` — do not redefine), `defaultOpen`, measured body height to
     the container floor + `ScrollFadeHint` — the exact geometry mechanism `RunTasks.jsx`
     already implements (`DRAWER_BODY_PAD`/`MIN_BODY_H`/ResizeObserver): reuse the pattern.
   - **States.** `running && no sections` → `ASSEMBLY_LIVE_TEXT` (italic muted, empty-state
     register). At rest with no sections → `ASSEMBLY_EMPTY_TEXT`. Terminal with sections → the
     dashed-tan cap line `Question answered · build complete` (mono uppercase, `T.line`,
     dashed border — the LAV-20 settle register) at the TOP of the stack.
   - Colors/typography: Treasury tokens per the mock (`T.navy` names, `T.muted` roles/labels,
     moss ✓ on completed stage labels, `T.cardAlt` + dashed `T.line` ghost boxes, flag-red
     only for the `error` stage label).

```js
// Fixture ledger — shapes copied from buildHopEvent output as RunTasks.jsx documents them.
const T0 = 1000;
const roster = [
  { id: "eleanor", name: "Eleanor Voss", role: "The Librarian" },
  { id: "marcus", name: "Marcus Webb", role: "GEO CSO Expert" },
  { id: "owen", name: "Owen Marsh", role: "The Proofreader" },
];
const ev = (type, agentId, data, id) => ({ id, type, agentId, data, durationMs: 100 });
const run = [
  ev("intent_routing", "michelle", { intent: "qa" }, 0),                                   // no section
  ev("assembly_work_complete", "eleanor", { work: "fetch", source: "the_library",
     matchCount: 12, forAgentId: "marcus", span_id: "s-fetch1", parent_span_id: "s-qa" }, 1), // evidence
  ev("prompt_assembled", "marcus", { toIntentSlug: "ci-answer-intent", prompt_chars: 900 }, 2), // draft ghost
  ev("qa_answer", "marcus", { citations: [1,2,3,4,5,6,7], confidence_tier: "high" }, 3),   // fills draft
  ev("prompt_assembled", "owen", { toIntentSlug: "qg-review-intent", prompt_chars: 700 }, 4),  // verification ghost
  ev("assembly_work_complete", "eleanor", { work: "fetch", source: "the_library",
     matchCount: 4, forAgentId: "owen", span_id: "s-fetch2", parent_span_id: "s-gate" }, 5),   // nests in verification
  ev("proofreader", "owen", { eval: { result: "pass", critique: "Cycle figure may be projected." } }, 6),
  ev("display_format", "michelle", { headline: "26-vs-46-month spread", key_data_points: [1,2,3,4] }, 7),
];
const times = run.map((_, i) => T0 + i * 1000);

const { buildAssemblyStages, STAGE_OF_INTENT, ASSEMBLY_LIVE_TEXT, ASSEMBLY_EMPTY_TEXT,
  DELIVERABLE_TITLE, ASSEMBLY_TITLE } = await importAssembly(); // per note above
let fail = 0; const ok = (n, c) => { console.log(`${c ? "PASS" : "FAIL"}: ${n}`); if (!c) fail++; };

// Mid-run at index 4 (owen ghost open, draft filled, evidence filled):
const mid = buildAssemblyStages(run.slice(0, 5), times.slice(0, 5), { runStartedAt: T0, agents: roster, running: true, terminal: false });
ok("3 sections mid-run", mid.sections.length === 3);
ok("newest-first: verification ghost on top", mid.sections[0].stage === "verification" && mid.sections[0].ghost === true);
ok("draft filled below", mid.sections[1].stage === "draft" && !mid.sections[1].ghost);
ok("evidence at bottom", mid.sections[2].stage === "evidence");
ok("no routing section exists", mid.sections.every(s => !["routing", "delegation"].includes(s.stage)));

// Full run, terminal:
const full = buildAssemblyStages(run, times, { runStartedAt: T0, agents: roster, running: false, terminal: true });
ok("4 stages at terminal", full.sections.length === 4);
ok("final form on top", full.sections[0].stage === "final");
ok("verification carries verbatim critique", JSON.stringify(full.sections[1]).includes("Cycle figure may be projected."));
ok("review fetch nested in verification, not evidence", full.sections[1].subEntries?.length === 1
  && JSON.stringify(full.sections[1].subEntries[0]).includes("4 chunk")
  && !JSON.stringify(full.sections.find(s => s.stage === "evidence")).includes("4 chunk"));
ok("no ghost survives terminal", full.sections.every(s => !s.ghost));
ok("terminal flag set for the cap line", full.terminal === true);

// Discriminating negatives:
const noIntent = buildAssemblyStages([ev("prompt_assembled", "x", { prompt_chars: 1 }, 0)], [T0], { runStartedAt: T0, agents: roster, running: true, terminal: false });
ok("unknown/absent slug -> unlabeled ghost, never a guessed stage", noIntent.sections.length === 1 && noIntent.sections[0].stage == null);
const empty = buildAssemblyStages([], [], { runStartedAt: T0, agents: roster, running: true, terminal: false });
ok("live+empty -> live chrome state", empty.sections.length === 0);
const blocked = buildAssemblyStages([ev("proofreader", "owen", { guardrail: { result: "block", reason: "policy" } }, 0)], [T0], { runStartedAt: T0, agents: roster, running: false, terminal: true });
ok("guardrail block builds its own honest verification stage (§19r no-storyboard)", blocked.sections.length === 1 && blocked.sections[0].stage === "verification" && JSON.stringify(blocked.sections[0]).includes("Blocked"));
ok("scores never rendered", !JSON.stringify(full).match(/\d\s*\/\s*5/));
ok("titles exact", ASSEMBLY_TITLE === "Assembly" && DELIVERABLE_TITLE === "Deliverable");
ok("STAGE_OF_INTENT covers exactly the three pipeline slugs", Object.keys(STAGE_OF_INTENT).length === 3);

console.log(fail === 0 ? "ALL TESTS PASS" : `${fail} FAILURES`);
process.exit(fail === 0 ? 0 : 1);
```
