const agents = [
  { id: "marcus", name: "Marcus Webb" },

  { id: "michelle", name: "Michelle Manning" },
];
const first = (id) => (agents.find(a => a.id === id)?.name || id || "").split(" ")[0];
// Task 2's priority order, restated as executable spec:
function expectedLine(evt) {
  const from = first(evt.fromAgentId), to = first(evt.toAgentId);
  const acct = typeof evt.account === "string" && evt.account.trim();
  const task = typeof evt.task === "string" && evt.task.trim();
  if ((evt.type === "delegation_complete" || evt.type === "delegation_return") && acct) {
    const credit = evt.type === "delegation_complete" ? to : from;
    return `${credit} — \u201C${evt.account}\u201D`;
  }
  if (evt.type === "delegation" && task) return `${from} → ${to} — \u201C${evt.task}\u201D`;
  return "TEMPLATE"; // degrade sentinel — must be reachable
}
let pass = 0, fail = 0;
const check = (name, got, want) => { if (got === want) { pass++; } else { fail++; console.error(`FAIL ${name}: got ${JSON.stringify(got)} want ${JSON.stringify(want)}`); } };
// 1. return with account credits the RETURNING agent (from), not the receiver — MI-48's exact bug class

  expectedLine({ type: "delegation_return", fromAgentId: "eleanor", toAgentId: "marcus", account: "Retrieved 14 passages." }),
  "Eleanor — \u201CRetrieved 14 passages.\u201D");

  expectedLine({ type: "delegation_complete", fromAgentId: "marcus", toAgentId: "michelle", account: "Picked the librarian." }),

  expectedLine({ type: "delegation", fromAgentId: "marcus", toAgentId: "eleanor", task: "Pull churn evidence." }),
  "Marcus → Eleanor — \u201CPull churn evidence.\u201D");
// 4-7. degrade: absent/empty/whitespace/null content NEVER renders a quoted line
