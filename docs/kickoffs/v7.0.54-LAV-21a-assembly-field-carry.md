## NODE.JS TEST

`test-lav-21a.mjs` — run from the worktree root with `node test-lav-21a.mjs`. Because the hook
is JSX-adjacent and stateful, the test asserts on the two seams as pure functions: import
`deriveAsked`/`deriveDid`/`deriveContent` from `RunTasks.jsx` is not possible under plain node
(JSX file) — instead assert the exports exist by grepping the source, and assert the field-carry
logic by replaying the exact build expressions:

```js
import { readFileSync } from "node:fs";
const hook = readFileSync("src/hooks/useHarnessStream.js", "utf8");
const feed = readFileSync("src/components/RunTasks.jsx", "utf8");
let fail = 0;
const ok = (name, cond) => { console.log(`${cond ? "PASS" : "FAIL"}: ${name}`); if (!cond) fail++; };

// 1 — delegation START build carries the two declared-work fields (the L220 build site).
const startBuild = hook.split("secondaryAgentId: evt.type === 'delegation'")[0].split("logEvent").pop();
ok("delegation start carries toCapabilitySlug", /toCapabilitySlug:\s*evt\.toCapabilitySlug\s*\?\?\s*null/.test(startBuild));
ok("delegation start carries toIntentSlug", /toIntentSlug:\s*evt\.toIntentSlug\s*\?\?\s*null/.test(startBuild));

// 2 — completion build carries toIntentSlug too (it already had toCapabilitySlug).
const completionBuild = hook.split("delegation_complete' || evt.type === 'delegation_return'")[1] || "";
ok("completion carries toIntentSlug", /toIntentSlug:\s*evt\.toIntentSlug\s*\?\?\s*null/.test(completionBuild.split("return;")[0]));

// 3 — prompt_assembled ledger event carries toIntentSlug (not only the setPrompt state).
const promptBuild = hook.split("buildHopEvent('prompt_assembled'")[1]?.split(")")[0] ?? "";
ok("prompt_assembled ledger carries toIntentSlug", /toIntentSlug/.test(hook.split("buildHopEvent('prompt_assembled'")[1]?.split("promptDurationMs")[0] ?? ""));

// 4 — assembly copy-loop source object includes parent_span_id + toCapabilitySlug.
const assemblySrc = hook.split("const assemblyData = { message: assemblyMessage };")[1]?.split("appendEvent")[0] ?? "";
ok("assembly frames carry parent_span_id", /parent_span_id:\s*evt\.parent_span_id/.test(assemblySrc));
ok("assembly frames carry toCapabilitySlug", /toCapabilitySlug:\s*evt\.toCapabilitySlug/.test(assemblySrc));

// 5 — discriminating negative: the null-guard loop is unchanged (fields absent stay absent).
ok("carried-only-when-present guard intact", /if \(v != null\) assemblyData\[k\] = v;/.test(hook));

// 6 — RunTasks exports the three derivations, and its rendering logic is otherwise untouched.
ok("deriveAsked exported", /export function deriveAsked\(/.test(feed));
ok("deriveDid exported", /export function deriveDid\(/.test(feed));
ok("deriveContent exported", /export function deriveContent\(/.test(feed));
ok("feed derivation bodies unchanged (spot: proofreader pass line)", feed.includes("Scored the draft — verdict ${verdict}: ${critique}") || /Scored the draft/.test(feed));

console.log(fail === 0 ? "ALL TESTS PASS" : `${fail} FAILURES`);
process.exit(fail === 0 ? 0 : 1);
```

Note: assertions 1-4 must FAIL against the pre-change file (run once before editing to confirm
they discriminate), and PASS after. Delete the file before committing.
