// test-s-mi-21.mjs — Category B (order/formatting logic) + source-structure checks.
// Pure Node.js: fs.readFileSync only, no app/React imports.

import { readFileSync } from "fs";

let pass = 0, fail = 0;
function check(name, cond) {
  if (cond) { pass++; console.log(`PASS: ${name}`); }
  else { fail++; console.log(`FAIL: ${name}`); }
}

// --- Task 1: AppShell.jsx nav tab order ---
const appShell = readFileSync("src/AppShell.jsx", "utf8");

const idxMI = appShell.indexOf('label="Market Intelligence"');
const idxWork = appShell.indexOf('label="Work"');
const idxBench = appShell.indexOf('label="Bench"');
check("all three NavTab labels found", idxMI > -1 && idxWork > -1 && idxBench > -1);
check("Market Intelligence renders before Work", idxMI < idxWork);
check("Work renders before Bench", idxWork < idxBench);

// hasBorderLeft: extract each NavTab's own block (from its opening tag to its own closing '/>')
function navTabBlock(source, labelIdx) {
  const blockStart = source.lastIndexOf("<NavTab", labelIdx);
  const blockEnd = source.indexOf("/>", labelIdx);
  return source.slice(blockStart, blockEnd);
}
const miBlock = navTabBlock(appShell, idxMI);
const workBlock = navTabBlock(appShell, idxWork);
const benchBlock = navTabBlock(appShell, idxBench);
check("Market Intelligence tab has hasBorderLeft={true}", /hasBorderLeft=\{true\}/.test(miBlock));
check("Work tab has hasBorderLeft={false}", /hasBorderLeft=\{false\}/.test(workBlock));
check("Bench tab has hasBorderLeft={false}", /hasBorderLeft=\{false\}/.test(benchBlock));

// --- Task 2: MarketIntelligenceScreen.jsx Pipeline Log drawer ---
const miScreen = readFileSync("src/screens/MarketIntelligenceScreen.jsx", "utf8");

check("Pipeline Log now uses Drawer", /<Drawer title="Pipeline Log"/.test(miScreen));
check("no leftover hardcoded Pipeline Log header div", !/>Pipeline Log<\/div>/.test(miScreen));

const pipelineDrawerIdx = miScreen.indexOf('<Drawer title="Pipeline Log"');
const pipelineDrawerTag = miScreen.slice(pipelineDrawerIdx, miScreen.indexOf(">", pipelineDrawerIdx) + 1);
check("Pipeline Log Drawer has defaultOpen={true}", /defaultOpen=\{true\}/.test(pipelineDrawerTag));

const learnedDrawerIdx = miScreen.indexOf('<Drawer title="Learned Context"');
const agentsDrawerIdx = miScreen.indexOf('<Drawer title="Agents"');
const dataSourcesDrawerIdx = miScreen.indexOf('<Drawer title="Data Sources"');
const learnedDrawerTag = miScreen.slice(learnedDrawerIdx, miScreen.indexOf(">", learnedDrawerIdx) + 1);
const agentsDrawerTag = miScreen.slice(agentsDrawerIdx, miScreen.indexOf(">", agentsDrawerIdx) + 1);
const dataSourcesDrawerTag = miScreen.slice(dataSourcesDrawerIdx, miScreen.indexOf(">", dataSourcesDrawerIdx) + 1);
check("Learned Context Drawer untouched (no defaultOpen)", !/defaultOpen/.test(learnedDrawerTag));
check("Agents Drawer untouched (no defaultOpen)", !/defaultOpen/.test(agentsDrawerTag));
check("Data Sources Drawer untouched (no defaultOpen)", !/defaultOpen/.test(dataSourcesDrawerTag));

// Pipeline Log renders before Learned Context (position unchanged, only wrapper changed)
check("Pipeline Log Drawer still renders before Learned Context Drawer", pipelineDrawerIdx > -1 && learnedDrawerIdx > -1 && pipelineDrawerIdx < learnedDrawerIdx);

// --- Category B: event-count pluralization logic (reproduced, matches the count expression above) ---
