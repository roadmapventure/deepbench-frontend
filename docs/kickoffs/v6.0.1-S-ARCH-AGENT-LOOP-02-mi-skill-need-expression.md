1. **Nadia is out of scope.** She has zero `skill_profiles` rows and zero `agent_capability_assignments` in Supabase today (verified live) — `S-APPLE-04a`/`04b` (her Escalate / Data Integrity Patch capabilities) haven't shipped. John rejected this: naming a fixed *capability_slug* for a fixed relationship is Rule #1's violation one level removed from naming a fixed *agent_id* — it still removes Priya's actual judgment about whether/who to ask. Per the §19d sniff test (run explicitly against this relationship, `CLAUDE-DESIGN.md` Architect Review, added this session): Priya gets `can_request_help` on her Stress Test intent instead, same mechanism as Marcus. In practice Michelle will very likely resolve it to Nadia once Nadia's capability exists — but the decision path stays reasoned, not hardcoded.
3. **`agents.rating` is out of scope.** No component in the codebase reads it and no prior session defines what it represents (verified: zero references in `src/`). Populating it here would be fabricating a signal with no defined meaning — logged as a new backlog item (`AA-96`, see close-out) rather than invented.

**What ships this session:**
- Marcus (`channel-intelligence`/`ci-answer-intent`, his front-door Q&A capability) gets `can_request_help: true` — a genuine "I don't know who can help with this" moment.
- Priya (`hypothesis-evaluation`/`hyp-stress-test-intent`) gets `can_request_help: true` — her reasoned Escalate trigger.

```js
// test-s-arch-agent-loop-02-api.mjs — Category F, L, M tests for S-ARCH-AGENT-LOOP-02
// Run: node --env-file=.env.local test-s-arch-agent-loop-02-api.mjs
// Read-only against Tasks 1-3's Supabase changes -- run this AFTER Tasks 1-3 are applied.

const SUPABASE_URL = process.env.SUPABASE_URL;
const SUPABASE_KEY = process.env.SUPABASE_SERVICE_KEY;
const HEADERS = { 'Content-Type': 'application/json', apikey: SUPABASE_KEY, Authorization: `Bearer ${SUPABASE_KEY}` };

let pass = 0, fail = 0;
function check(label, cond) {
  if (cond) { console.log(`PASS: ${label}`); pass++; }
  else { console.log(`FAIL: ${label}`); fail++; }
}

function testCategoryF() {
  console.log('\n--- Category F: environment ---');
  check('SUPABASE_URL set', !!SUPABASE_URL);
  check('SUPABASE_SERVICE_KEY set', !!SUPABASE_KEY);
  check('ANTHROPIC_API_KEY set', !!process.env.ANTHROPIC_API_KEY);
}

async function fetchSkillProfile(slug) {
  const res = await fetch(`${SUPABASE_URL}/rest/v1/skill_profiles?slug=eq.${slug}&select=slug,traits`, { headers: HEADERS });
  if (!res.ok) throw new Error(`Fetch failed for ${slug}: ${res.status}`);
  const rows = await res.json();
  return rows[0] || null;
}

async function fetchAgent(id) {
  const res = await fetch(`${SUPABASE_URL}/rest/v1/agents?id=eq.${id}&select=id,skill_score,situational_awareness,rating`, { headers: HEADERS });
  if (!res.ok) throw new Error(`Fetch failed for ${id}: ${res.status}`);
  const rows = await res.json();
  return rows[0] || null;
}

async function testCategoryFSchema() {
  console.log('\n--- Category F: Supabase column/data alignment ---');

  const ciAnswer = await fetchSkillProfile('ci-answer-intent');
  check('ci-answer-intent: can_request_help true', ciAnswer?.traits?.can_request_help === true);
  check('ci-answer-intent: schema key preserved (not overwritten)', !!ciAnswer?.traits?.schema);

  const hypStress = await fetchSkillProfile('hyp-stress-test-intent');
  check('hyp-stress-test-intent: can_request_help true', hypStress?.traits?.can_request_help === true);

  // Regression controls -- must remain untouched
  const ciRouting = await fetchSkillProfile('ci-routing-intent');
  check('ci-routing-intent: can_request_help NOT set (regression)', ciRouting?.traits?.can_request_help !== true);

  const hypGen = await fetchSkillProfile('hyp-generation-intent');
  check('hyp-generation-intent: can_request_help NOT set (regression)', hypGen?.traits?.can_request_help !== true);

  const marcus = await fetchAgent('marcus');
  check('marcus: skill_score = 82', marcus?.skill_score === 82);
  check('marcus: situational_awareness = 38', marcus?.situational_awareness === 38);
  check('marcus: rating untouched (still 0)', Number(marcus?.rating) === 0);

  const priya = await fetchAgent('priya');
  check('priya: skill_score = 80', priya?.skill_score === 80);
  check('priya: situational_awareness = 42', priya?.situational_awareness === 42);

  const nadia = await fetchAgent('nadia');
  check('nadia: skill_score = 78', nadia?.skill_score === 78);
  check('nadia: situational_awareness = 35', nadia?.situational_awareness === 35);
}
```
