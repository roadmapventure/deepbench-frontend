```jsx
// DeepBench v7.0.0 | useHarnessStream.js | LAV-1a -- run a capability request and consume its
// live event stream outside the CHI screen. Reuses the CHI screen's exported primitives --
// one implementation, two consumers (FEATURE: LAV-1). CHI's own wiring is intentionally not
// refactored onto this hook.
import { useState, useRef, useCallback } from "react";
import { useAgents } from "./useAgents.js";
import {
  runQaWithQualityGate, buildHopEvent, describeDelegationEvent,
  pickCreditedSpan, resolveEventDuration,
} from "../screens/MarketIntelligenceScreen.jsx";

export function useHarnessStream() {
  const agents = useAgents();
  const [events, setEvents] = useState([]);        // same stored-event shape as CHI's pipelineEvents
  const [status, setStatus] = useState(null);      // { message, kind, startedAt, expectationMs } | null
  const [running, setRunning] = useState(false);
  const [result, setResult] = useState(null);      // terminal result frame, verbatim
  const [error, setError] = useState(null);
  const [recovery, setRecovery] = useState(null);  // last HAR-17 recovery notice | null
  const lastEventAtRef = useRef(null);
  const runIdRef = useRef(0);

  const logEvent = (evt) => setEvents(prev => [...prev, evt]);

  // Mirrors MarketIntelligenceScreen.jsx's onDelegationProgress (~L3724-3749) exactly:
  // same attribution branch, same data fields, same span spread, same arrival-delta timing.
  // Re-read that function fresh and mirror the CURRENT branching — do not simplify it.
  const onDelegationProgress = (evt) => {
    const message = describeDelegationEvent(evt, agents);
    const now = Date.now();
    const durationMs = resolveEventDuration(evt.type, { nowMs: now, lastEventAtMs: lastEventAtRef.current });
    lastEventAtRef.current = now;
    if (evt.type === 'delegation_complete' || evt.type === 'delegation_return') {
      const attributedAgentId = evt.type === 'delegation_complete' ? evt.toAgentId : evt.fromAgentId;
      logEvent(buildHopEvent(evt.type, attributedAgentId, { message, viaTool: evt.viaTool || null, reasoning: evt.reasoning ?? null, task: evt.task ?? null, toCapabilitySlug: evt.toCapabilitySlug ?? null, ...pickCreditedSpan(evt) }, durationMs, {}));
      return;
    }
    logEvent(buildHopEvent(evt.type, evt.fromAgentId, { message, viaTool: evt.viaTool || null, ...pickCreditedSpan(evt) }, durationMs, { secondaryAgentId: evt.type === 'delegation' ? evt.toAgentId : null }));
  };

  const runQuestion = useCallback(async (text) => { /* guard re-entry; setRunning(true); clear per-run state;
    call runQaWithQualityGate(text, conversationContext=[], onEvent=logEvent, setStatus-adapter,
    onProgress=onDelegationProgress, isStale=stale-run-id check, backgroundContext=null,
    onRecovery=(r)=>setRecovery(r)); capture result/error; setRunning(false) */ }, [agents]);

  const clear = useCallback(() => { /* reset all state + lastEventAtRef + bump runIdRef */ }, []);

  return { events, status, running, result, error, recovery, runQuestion, clear };
}
```

The sketch's `runQuestion` body is intent, not final code — read `runQaWithQualityGate`'s **current** signature (`(message, conversationContext, onEvent, setStatus, onProgress, isStale, backgroundContext, onRecovery)`, ~L1480) and CHI's own call site fresh, and match how CHI adapts `setStatus` (it passes a `(message, opts)` writer — the hook's adapter stores `{ message, kind: opts?.kind, startedAt }` into hook state; preserve any expectation/`expectationMs` mechanics the current writer carries). One event vocabulary, no renames: whatever `buildHopEvent` stores is what `events` holds — `AuditColumn` must be able to render `events` directly (that is `S-LAV-1b`'s right rail).

**Do not** wire `resolveConfirmation`/HITL into the hook this session — that is `S-LAV-1d`'s scope; leave an explicit `// FEATURE: LAV-1d will extend this hook with the confirmation-resolve path` comment where it will land.

// test-lav-1a.mjs — pure Node, no app imports. Verifies (a) the extracted questions module is
// complete and single-sourced, (b) the hook's event-attribution mirror logic, via fs reads and
// reimplemented pure logic only.
import { readFileSync } from "node:fs";

let pass = 0, fail = 0;
function assert(cond, label) {
  if (cond) { pass++; console.log(`PASS: ${label}`); }
  else { fail++; console.log(`FAIL: ${label}`); }
}

const qSrc = readFileSync("src/data/chiQuestions.js", "utf8");
const miSrc = readFileSync("src/screens/MarketIntelligenceScreen.jsx", "utf8");
const hookSrc = readFileSync("src/hooks/useHarnessStream.js", "utf8");

// --- Category M: single source — the module exports all three, the screen imports and no longer declares them ---
assert(/export const STATIC_QUESTION/.test(qSrc), "chiQuestions.js exports STATIC_QUESTION");
assert(/export const ROTATING_POOL/.test(qSrc), "chiQuestions.js exports ROTATING_POOL");
assert(/export const FIXED_DRAWER_TAIL/.test(qSrc), "chiQuestions.js exports FIXED_DRAWER_TAIL");
assert(/from ["']..\/data\/chiQuestions.js["']/.test(miSrc), "MarketIntelligenceScreen imports from chiQuestions.js");
assert(!/const STATIC_QUESTION\s*=/.test(miSrc), "screen no longer declares STATIC_QUESTION");
assert(!/const ROTATING_POOL\s*=/.test(miSrc), "screen no longer declares ROTATING_POOL");
assert(!/const FIXED_DRAWER_TAIL\s*=/.test(miSrc), "screen no longer declares FIXED_DRAWER_TAIL");

// --- Category B: 23 questions total (1 static + pool + tail), every label non-empty ---
const labels = [...qSrc.matchAll(/label:\s*"((?:[^"\\]|\\.)+)"/g)].map(m => m[1]);
assert(labels.length === 23, `exactly 23 question labels found (got ${labels.length})`);
assert(labels.every(l => l.trim().length > 10), "every question label is a real sentence");

// --- Category B: hook attribution mirror — same branch logic the screen uses ---
function attributedAgent(evt) {
  if (evt.type === 'delegation_complete') return evt.toAgentId;
  if (evt.type === 'delegation_return') return evt.fromAgentId;
  return evt.fromAgentId;
}
assert(attributedAgent({ type: 'delegation_complete', fromAgentId: 'marcus', toAgentId: 'michelle' }) === 'michelle',
  "delegation_complete credits the target (toAgentId)");
assert(attributedAgent({ type: 'delegation_return', fromAgentId: 'michelle', toAgentId: 'marcus' }) === 'michelle',
  "delegation_return credits the returning delegate (fromAgentId)");
assert(attributedAgent({ type: 'delegation', fromAgentId: 'marcus', toAgentId: 'nadia' }) === 'marcus',
  "delegation credits the dispatcher (fromAgentId), target rides secondaryAgentId");

// --- Category M: hook reuses exported primitives, defines none of its own copies ---
assert(/from ["']..\/screens\/MarketIntelligenceScreen.jsx["']/.test(hookSrc), "hook imports primitives from the screen module");
for (const fn of ["runQaWithQualityGate", "buildHopEvent", "describeDelegationEvent", "pickCreditedSpan", "resolveEventDuration"]) {
  assert(new RegExp(`\\b${fn}\\b`).test(hookSrc), `hook references ${fn}`);
  assert(!(new RegExp(`function ${fn}\\b`).test(hookSrc)), `hook does NOT re-implement ${fn}`);
}
// --- Guardrail: no simulator control flow in the new files ---
assert(!/Math\.random|setTimeout/.test(qSrc + hookSrc), "no Math.random/setTimeout in the new modules");

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS PASS: NO — fix before committing"); process.exit(1); }
console.log("ALL TESTS PASS");
