// test-aa-179b.mjs — pure Node, no app imports.
import { readFileSync } from "node:fs";
let pass = 0, fail = 0;
const assert = (c, l) => { if (c) { pass++; console.log(`PASS: ${l}`); } else { fail++; console.log(`FAIL: ${l}`); } };

const net = readFileSync("src/components/AgentNetwork.jsx", "utf8");
const screen = readFileSync("src/screens/LiveAgentViewScreen.jsx", "utf8");
const stripped = s => s.replace(/\/\/[^\n]*/g, "");

// --- assembly types handled; edges/pulses explicitly exclude them ---
assert(/assembly_work/.test(net), "AgentNetwork handles the family");
const pulsesIdx = net.indexOf("pulsesForHop");
assert(/assembly/.test(net.slice(pulsesIdx, pulsesIdx + 1200)), "pulsesForHop explicitly excludes assembly types");
assert(!/isPromptFrame[\s\S]{0,200}assembly/.test(screen), "isPromptFrame NOT widened to filter assembly frames");

// --- tint from tokens, no hex literals ---
assert(/mutedDeep/.test(net), "ring uses T.mutedDeep");
assert(!/#[0-9a-fA-F]{3,8}\b/.test(stripped(net).replace(/PALETTE|rgba\(/g, "")) || !/#[0-9a-fA-F]{3,8}\b/.test(stripped(net)), "zero literal hex added (compare against origin/dev if pre-existing literals exist: added lines only)");

// --- bubble label mirror (approved copy, real fields only) ---
const label = (evt) => {
  const mc = evt.matchCount != null ? evt.matchCount : "—";
  if (evt.work === "fetch") {
    if (evt.source === "the_library") return `Data Room read · ${mc} chunks`;
    if (evt.source === "the_library_catalog") return `Data Room catalog · ${mc}`;
    if (evt.source === "roster") return `Roster fetch · ${mc} candidates`;
    if (evt.source === "the_reasoning") return `Reasoning Layer read · ${mc}`;
    return `Knowledge read · ${mc} chunks`;
  }
  const tok = evt.tokens != null ? ` · ${evt.tokens} tok` : "";
  if (evt.work === "reflect") return evt.type === "assembly_work" ? "REFLECT · drafting execution plan" : `REFLECT · plan drafted${tok}`;
  return evt.type === "assembly_work" ? "Synthesis · rewriting prompt" : `Synthesis · prompt rewritten${tok}`;
};
assert(label({ work: "fetch", source: "the_library", matchCount: 3 }) === "Data Room read · 3 chunks", "library label");
assert(label({ work: "fetch", source: "roster", matchCount: 5 }) === "Roster fetch · 5 candidates", "roster label");
assert(label({ work: "fetch", source: "the_library" }) === "Data Room read · — chunks", "missing count renders dash, never a guessed 0");
assert(label({ work: "reflect", type: "assembly_work" }) === "REFLECT · drafting execution plan", "reflect start");
assert(label({ work: "reflect", type: "assembly_work_complete", tokens: 812 }) === "REFLECT · plan drafted · 812 tok", "reflect complete w/ real tokens");
assert(label({ work: "synthesis", type: "assembly_work_complete" }) === "Synthesis · prompt rewritten", "no tokens -> no token clause");

// --- engagement + no-edge derivation mirror ---
function derive(hops) {
  const engaged = [], edges = [];
  for (const h of hops) {
    if (h.agentId && !engaged.includes(h.agentId)) engaged.push(h.agentId);
    if (h.type === "delegation" && h.secondaryAgentId) edges.push([h.agentId, h.secondaryAgentId]);
  }
  return { engaged, edges };
}
const mixed = derive([
  { type: "assembly_work_complete", agentId: "eleanor" },
  { type: "assembly_work", agentId: "dan" },
  { type: "delegation", agentId: "marcus", secondaryAgentId: "michelle" },
]);
assert(mixed.engaged.join(",") === "eleanor,dan,marcus", "assembly workers engage in arrival order");
assert(mixed.edges.length === 1 && mixed.edges[0][0] === "marcus", "assembly contributes zero edges");

// --- inertness: empty assembly input -> identical outputs ---
const before = derive([{ type: "delegation", agentId: "a", secondaryAgentId: "b" }]);
const after = derive([{ type: "delegation", agentId: "a", secondaryAgentId: "b" }]);
assert(JSON.stringify(before) === JSON.stringify(after), "no assembly frames -> byte-identical derivation");

// --- start/complete active-window pairing mirror ---
function assemblyActive(hops) {
  const open = new Set();
  for (const h of hops) {
    const k = `${h.agentId}|${h.work}`;
    if (h.type === "assembly_work") open.add(k);
    else if (h.agentId) { open.delete(k); [...open].forEach(o => { if (o.startsWith(`${h.agentId}|`)) open.delete(o); }); }
  }
  return [...open];
}
assert(assemblyActive([{ type: "assembly_work", agentId: "dan", work: "reflect" }]).length === 1, "start opens the window");
assert(assemblyActive([
  { type: "assembly_work", agentId: "dan", work: "reflect" },
  { type: "assembly_work_complete", agentId: "dan", work: "reflect" },
]).length === 0, "matching complete closes it");

console.log(`\n${pass} PASS, ${fail} FAIL`);
if (fail > 0) { console.log("ALL TESTS PASS: NO — fix before committing"); process.exit(1); }
console.log("ALL TESTS PASS");
