## 2. CONTEXT

The Supabase backup tool (`C:/Projects/deepbench-backups/dump-supabase.mjs`, machine-local
working copy — **not** inside this git repo) dumps every table over PostgREST and captures
schema DDL from the server-side `public._backup_schema_ddl` view. When that view 404'd from
~2026-08-08 to 2026-08-23, snapshots silently shipped with **no `schema.sql`/`migrations.sql`**:
the script catches the schema error, writes `manifest.schema = { failed: true }`, and still
exits 0 — `totals.failures` counts only table failures (line 296) and the exit-code gate
(line 309) never looks at `manifest.schema`. That 16-day silence is the remaining live defect
of `DAT-21`; the view itself was rebuilt (`SES-81`) and now emits executable grants (`SES-216`),
so once the run fails loudly the ticket's whole surface is covered.

## 8. NODE.JS TEST (full code — save as `C:/Projects/deepbench-backups/tests/DAT-21-schema-failure-fails-run.mjs`)

```js
#!/usr/bin/env node
// DAT-21 regression guard: a schema-capture failure must FAIL the dump run
// (non-zero exit + manifest.totals.schema_failed = 1), while row capture
// still completes. Drives the REAL dump-supabase.mjs against a stub PostgREST
// that reproduces the exact historical failure (PGRST205 on _backup_schema_ddl)
// with healthy tables — so a pass proves the exit code came from the schema
// path, not a table failure.
//
// Run: node C:/Projects/deepbench-backups/tests/DAT-21-schema-failure-fails-run.mjs

import http from 'node:http';
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { spawn } from 'node:child_process';

const DUMP = 'C:/Projects/deepbench-backups/dump-supabase.mjs';

const server = http.createServer((req, res) => {
  const u = new URL(req.url, 'http://x');
  const json = (code, body, extra = {}) => {
    res.writeHead(code, { 'content-type': 'application/json', ...extra });
    res.end(JSON.stringify(body));
  };
  if (u.pathname === '/rest/v1/_backup_schema_ddl') {
    // The exact live failure shape from snapshots 2026-08-08/2026-08-20.
    return json(404, { code: 'PGRST205', details: null, hint: null,
      message: "Could not find the table 'public._backup_schema_ddl' in the schema cache" });
  }
  if (u.pathname === '/rest/v1/_backup_inventory') {
    return json(200, [{ name: 't1', pk: 'id', kind: 'table' }]);
  }
  if (u.pathname === '/rest/v1/t1') {
    // Serves countOf (content-range), the keyset dump page, and verify()'s
    // pk re-read. One row; every page after gt.1 is empty.
    const gt = [...u.searchParams.entries()].some(([k, v]) => k === 'id' && v.startsWith('gt.'));
    return json(200, gt ? [] : [{ id: 1 }], { 'content-range': '0-0/1' });
  }
  return json(404, { message: 'not found: ' + u.pathname });
});

await new Promise((r) => server.listen(0, '127.0.0.1', r));
const port = server.address().port;
const outDir = fs.mkdtempSync(path.join(os.tmpdir(), 'dat21-'));

// Async spawn, NOT spawnSync: spawnSync freezes this process's event loop,
// which silently deadlocks the stub server (it can never answer the child).
const run = await new Promise((resolve) => {
  const child = spawn(process.execPath, [DUMP, outDir], {
    env: { ...process.env, SUPABASE_URL: `http://127.0.0.1:${port}`, SUPABASE_SERVICE_KEY: 'test-key' },
  });
  let stdout = '', stderr = '';
  child.stdout.on('data', (d) => (stdout += d));
  child.stderr.on('data', (d) => (stderr += d));
  const killer = setTimeout(() => child.kill(), 60000);
  child.on('close', (status) => { clearTimeout(killer); resolve({ status, stdout, stderr }); });
});
server.close();

if (process.env.DAT21_DEBUG) {
  console.log('exit:', run.status, '\nSTDOUT:\n', run.stdout, '\nSTDERR:\n', run.stderr);
}
const manifest = JSON.parse(fs.readFileSync(path.join(outDir, 'manifest.json'), 'utf8'));
fs.rmSync(outDir, { recursive: true, force: true });

let failed = 0;
const assert = (n, cond, detail) => {
  console.log(`  ${cond ? 'PASS' : 'FAIL'}  A${n}: ${detail}`);
  if (!cond) failed++;
};

// Preconditions (pin the branch as fired + the discriminator):
assert(1, manifest.schema && manifest.schema.failed === true,
  `schema capture took the failure branch (manifest.schema.failed=${manifest.schema?.failed})`);
assert(2, manifest.tables?.t1 && !manifest.tables.t1.failed && manifest.tables.t1.rows === 1,
  `table capture stayed healthy (t1 rows=${manifest.tables?.t1?.rows}, failed=${manifest.tables?.t1?.failed})`);
// Content assertions (the DAT-21 fix — both must FAIL pre-change):
assert(3, run.status !== 0,
  `dump exited non-zero on schema failure (exit=${run.status})`);
assert(4, manifest.totals?.schema_failed === 1,
  `manifest.totals.schema_failed counts it (schema_failed=${manifest.totals?.schema_failed})`);

console.log(failed === 0 ? '\nALL TESTS PASS' : `\n${failed} ASSERTION(S) FAILED`);
process.exitCode = failed === 0 ? 0 : 1;
```
