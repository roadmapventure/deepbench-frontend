## 2. CONTEXT

`LAV-9b` (v7.0.31, just shipped) widened `engaged` so a delegation target joins the canvas the instant it's named, not only once it produces its own hop — that part is correct and working. But the coding session that built it flagged a real side effect it deliberately did not fix, because fixing it changes visual appearance beyond that kickoff's stated diff: `done` is computed as `[...engaged, ...finished].filter(a => a !== activeId && !orchestrators.has(a))`. A newly-addressed agent is now in `engaged`, is not `activeId`, and is not an orchestrator — so it lands in `done` and renders with the finished-look moss border (`.lav-node.is-done`) while it's still waiting its turn to actually speak. The card does come forward onto the canvas as `LAV-9b` intended; it just wears the wrong skin while it waits.

Fix: `done` should mean "has a hop really credited to its own id, and isn't currently active" — the same test `engaged` used *before* `LAV-9b` widened it — not "is on the canvas at all." Track that narrower condition separately from the now-broader `engaged` (which stays broad, for the arc/bench promotion `LAV-9b` built), so a card only earns the moss "done" border once it has actually spoken.

// test-lav-9b-patch.mjs -- LAV-9b-patch Category B logic test (pure, no app imports)
function deriveDone(runHops) {
  const engaged = [], selfEngaged = new Set(), finished = new Set();
  const openByTarget = new Map();
  let activeId = null;
  for (const h of runHops) {
    const id = h.agentId;
    if (id) { if (!engaged.includes(id)) engaged.push(id); selfEngaged.add(id); activeId = id; }
    if (h.type === "delegation" && id && h.secondaryAgentId) {
      if (!engaged.includes(h.secondaryAgentId)) engaged.push(h.secondaryAgentId);
      openByTarget.set(h.secondaryAgentId, id);
    }
    if ((h.type === "delegation_complete" || h.type === "delegation_return") && id) {
      openByTarget.delete(id); finished.add(id);
    }
  }
  const orchestrators = new Set([...openByTarget.values()]);
  return new Set([...selfEngaged, ...finished].filter(a => a !== activeId && !orchestrators.has(a)));
}

let pass = 0, fail = 0;
function assertEq(actual, expected, msg) {
  const ok = JSON.stringify([...actual].sort()) === JSON.stringify([...expected].sort());
  if (ok) { pass++; } else { fail++; console.error(`FAIL: ${msg} -- got ${JSON.stringify([...actual])}, expected ${JSON.stringify([...expected])}`); }
}

// A: a delegation target that hasn't spoken yet is NOT done (the bug this patch fixes)
const runA = [{ agentId: "michelle", type: "delegation", secondaryAgentId: "marcus" }];
assertEq(deriveDone(runA), new Set(), "A: freshly-addressed, silent agent is not done");

// B: once that target actually reports, it IS done (once no longer active)
const runB = [
  { agentId: "michelle", type: "delegation", secondaryAgentId: "marcus" },
  { agentId: "marcus", type: "delegation_complete" },
  { agentId: "michelle", type: "agent-turn" }, // michelle becomes active again after marcus finishes
];
assertEq(deriveDone(runB), new Set(["marcus"]), "B: agent that actually reported is done once no longer active");

console.log(fail === 0 ? "ALL TESTS PASS" : `${fail} FAILED, ${pass} PASSED`);
if (fail > 0) process.exit(1);
