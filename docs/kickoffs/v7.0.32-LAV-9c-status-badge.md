## 2. CONTEXT

The mode badge already has a real `"complete"` state (`deriveMode()`, green tint, `MODE_COPY.complete`) — but it's only visible for 3 seconds. `terminal` (the state that drives it) is set to `"result"` when a run finishes (line ~408: `useEffect(() => { if (result) setTerminal("result"); }, [result])`), and correctly cleared back to `null` the moment a *new* question starts (line 554, inside the `onRun` handler, before `runQuestion`). The bug is a THIRD effect (currently ~line 410-413) that also clears it on a bare 3-second timer, independent of whether a new run has actually started. Once that timer fires, `deriveMode()` has nothing left to key off but `running: false`, which falls through to `"idle"` (line 142) — so anyone who isn't staring at the exact moment a run finishes only ever sees "Idle," indistinguishable from "nothing has run yet."

John's fix, confirmed in the design session: remove the timer entirely (the existing new-run-start clear at line 554 is the only reset that should exist), rename the copy from "Complete" to **"Question Answered"** (John's own framing — the specific thing that's true is the question, not some larger process), and give it a **pulse**, matching the existing `orch`-state pulse mechanism, so it visibly announces "go look at the metrics around the screen" rather than sitting there passively.

```js
// test-lav-9c.mjs -- LAV-9c Category B logic tests (pure, no app imports)
function deriveMode(runHops, { running, terminal, awaiting = false }) {
  if (awaiting) return "awaiting";
  if (terminal) return terminal === "error" ? "error" : "complete";
  if (!running) return "idle";
  return runHops.some(h => h.type === "delegation_return") ? "orch" : "route";
}
const MODE_COPY = { idle: "Idle", route: "Routing", orch: "Orchestrating", awaiting: "Awaiting confirmation", complete: "Question Answered", error: "Error" };
function pulses(mode) { return mode === "orch" || mode === "complete"; }

let pass = 0, fail = 0;
function assertEq(actual, expected, msg) {
  const ok = JSON.stringify(actual) === JSON.stringify(expected);
  if (ok) { pass++; } else { fail++; console.error(`FAIL: ${msg} -- got ${JSON.stringify(actual)}, expected ${JSON.stringify(expected)}`); }
}

// A: terminal="result" always yields "complete" -- no time dimension, so nothing can revert it
// except a real state change (this IS the fix: the old bug was an extra timer outside this function
// clearing `terminal` back to null after 3000ms; this asserts the function itself has no such decay)
assertEq(deriveMode([], { running: false, terminal: "result" }), "complete", "A: terminal=result -> complete, immediately");
assertEq(deriveMode([], { running: false, terminal: "result" }), "complete", "A2: still complete on a later call with identical args (no hidden timer state)");

// B: only a real state change -- terminal going back to null -- falls through to idle
assertEq(deriveMode([], { running: false, terminal: null }), "idle", "B: terminal cleared (e.g. by onRun's own setTerminal(null)) -> idle");

// C: error still routes to "error", not "complete"
assertEq(deriveMode([], { running: false, terminal: "error" }), "error", "C: terminal=error unaffected by this session's changes");

// D: copy and pulse
assertEq(MODE_COPY.complete, "Question Answered", "D: renamed copy");
assertEq(pulses("complete"), true, "D2: complete pulses");
assertEq(pulses("orch"), true, "D3: orch still pulses (unchanged)");
assertEq(pulses("idle"), false, "D4: idle does not pulse");
assertEq(pulses("error"), false, "D5: error does not pulse (untouched by this session)");

console.log(fail === 0 ? "ALL TESTS PASS" : `${fail} FAILED, ${pass} PASSED`);
if (fail > 0) process.exit(1);
```
