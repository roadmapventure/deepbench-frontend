## 2. CONTEXT

Four items from John's live review of the deployed canvas, worked one at a time:

1. **Agent cards render underneath the prompt box and the legend/toggle.** Both overlays are positioned in real screen pixels (`.lav-topright{right:12,top:12}`, the prompt box's `right:314,bottom:8`), completely outside the canvas's own 1200×640 coordinate system — while the bench/queue column is anchored to the canvas's right edge in that same coordinate system. All three cluster on the right side, and nothing today keeps agent cards clear of the other two. **Confirmed with John: reserve a fixed safe band in the canvas's own coordinate space (no runtime DOM measurement) — trades a little of the breathing room `LAV-9b` just added for correctness, and John accepted that trade explicitly.**
2. **Remove the agent code** (e.g. "LB-01") from the card — pure removal, no other element reads it.
3. **Show the name on every actively-animated line, not just the top 2-3.** Since `LAV-9a` made every active-this-run edge animate simultaneously, capping labels to a recency-ranked top-3 means most of the moving lines go silent — the exact mismatch John flagged. The recency-based fade is still the right behavior for the settled view *after* a run ends (nothing's "active" then); it just shouldn't throttle labels while a line is genuinely still animating.
4. **A line can point at a card still sitting in the queue.** Root-caused, not the same bug as `LAV-9b`: `sessionEdges()` is fed the WHOLE BROWSER SESSION's hops (`canvasHops`, "edges persist across runs" — a deliberate, documented `LAV-1b`/`1c`-era choice), while node/queue state (`engaged`, `benched`) is scoped to the CURRENT run only. Run question A, a line forms; run question B, that line is still drawn even though its endpoint hasn't done anything in run B yet — reads exactly like "hiding in the queue." **Confirmed with John: reset edges per question, matching every other run-scoped piece of this canvas (nodes, bubbles, pulses, the burst).** Live-instrumented a single fresh run for 82 samples over 2+ minutes (matching every line's actual on-screen endpoint against every card's actual on-screen position) and found zero within-a-run violations — this row's own mechanism (`LAV-9b`) is sound; the cross-question staleness is the whole explanation.

```js
// test-lav-12.mjs -- LAV-12 Category B logic tests (pure, no app imports)
let pass = 0, fail = 0;
function assertEq(actual, expected, msg) {
  const ok = JSON.stringify(actual) === JSON.stringify(expected);
  if (ok) { pass++; } else { fail++; console.error(`FAIL: ${msg} -- got ${JSON.stringify(actual)}, expected ${JSON.stringify(expected)}`); }
}
function assertTrue(cond, msg) { if (cond) pass++; else { fail++; console.error(`FAIL: ${msg}`); } }

const TOP_SAFE_Y = 85, BOTTOM_SAFE_Y = 555;
const MID_Y = 305, ARC_RY = 215;
const BENCH_SPAN = 430;

// A: arc's full vertical swing stays inside the safe band
const arcMin = MID_Y - ARC_RY, arcMax = MID_Y + ARC_RY;
assertTrue(arcMin >= TOP_SAFE_Y, `A: arc min ${arcMin} clears TOP_SAFE_Y ${TOP_SAFE_Y}`);
assertTrue(arcMax <= BOTTOM_SAFE_Y, `A2: arc max ${arcMax} clears BOTTOM_SAFE_Y ${BOTTOM_SAFE_Y}`);

// B: bench stack's full span (any roster size) stays inside the safe band
function benchRange(m) {
  const step = m > 1 ? Math.min(58, BENCH_SPAN / (m - 1)) : 0;
  const y0 = 305 - (step * (m - 1)) / 2;
  return { min: y0, max: y0 + step * (m - 1) };
}
for (const m of [1, 2, 5, 8, 11, 20]) {
  const r = benchRange(m);
  assertTrue(r.min >= TOP_SAFE_Y - 0.01, `B(${m}): bench min ${r.min} clears TOP_SAFE_Y`);
  assertTrue(r.max <= BOTTOM_SAFE_Y + 0.01, `B2(${m}): bench max ${r.max} clears BOTTOM_SAFE_Y`);
}

// C: home grid's existing formula (untouched) already fits, for realistic roster sizes
function homeYRange(n) {
  const cols = Math.min(4, n), rows = Math.ceil(n / cols);
  if (rows === 1) return { min: 320, max: 320 };
  const ys = Array.from({length: rows}, (_, r) => 640 * (0.16 + 0.68 * (r / (rows - 1))));
  return { min: Math.min(...ys), max: Math.max(...ys) };
}
for (const n of [1, 4, 8, 11, 16]) {
  const r = homeYRange(n);
  assertTrue(r.min >= TOP_SAFE_Y, `C(${n}): home grid min ${r.min} clears TOP_SAFE_Y`);
  assertTrue(r.max <= BOTTOM_SAFE_Y, `C2(${n}): home grid max ${r.max} clears BOTTOM_SAFE_Y`);
}

// D: label opacity -- running always full for an active edge; settled view keeps the rank tiers
function labelOpacityForRank(rank) { if (rank < 0) return 0; if (rank < 2) return 1; if (rank < 3) return 0.45; return 0; }
function labelOpacity(running, rank) { return running ? 1 : labelOpacityForRank(rank); }
assertEq(labelOpacity(true, 5), 1, "D: running, low-rank (old) active edge -- still full opacity");
assertEq(labelOpacity(true, 0), 1, "D2: running, top-rank -- full opacity (unchanged)");
assertEq(labelOpacity(false, 0), 1, "D3: settled, rank 0 -- full (existing tier)");
assertEq(labelOpacity(false, 2), 0.45, "D4: settled, rank 2 -- fading (existing tier, unchanged)");
assertEq(labelOpacity(false, 5), 0, "D5: settled, old rank -- silent (existing tier, unchanged)");

// E: edges reset per question -- mirrors sessionEdges()'s own scan, fed runHops instead of the
// whole-session ledger. A delegation from a PRIOR question must not appear once a new one starts.
function edgeKey(a, b) { return `${a}>${b}`; }
function sessionEdgeKeys(hopsForEdges) {
  const seen = new Set();
  for (const h of hopsForEdges) {
    if (h.type !== "delegation" || !h.agentId || !h.secondaryAgentId) continue;
    seen.add(edgeKey(h.agentId, h.secondaryAgentId));
  }
  return seen;
}
const fullSession = [
  { type: "delegation", agentId: "michelle", secondaryAgentId: "marcus" },   // question 1
  { type: "question_boundary" },
  { type: "delegation", agentId: "owen", secondaryAgentId: "eleanor" },      // question 2
];
const runOnly = fullSession.slice(fullSession.findIndex(h => h.type === "question_boundary") + 1);
assertTrue(sessionEdgeKeys(fullSession).has(edgeKey("michelle", "marcus")), "E: old behavior (whole session) -- stale edge present (control)");
assertTrue(!sessionEdgeKeys(runOnly).has(edgeKey("michelle", "marcus")), "E2: new behavior (run-scoped) -- stale edge gone");
assertTrue(sessionEdgeKeys(runOnly).has(edgeKey("owen", "eleanor")), "E3: new behavior -- this run's real edge still present");

console.log(fail === 0 ? "ALL TESTS PASS" : `${fail} FAILED, ${pass} PASSED`);
if (fail > 0) process.exit(1);
```
