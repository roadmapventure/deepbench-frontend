```js
// test-log-92.mjs — S-LOG-92 (v6.3.170) — AI Audit header correctness
// Categories: B (logic/formatting), F (Supabase alignment — the un-filtered read + reconciliation identity)
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(process.env.VITE_SUPABASE_URL, process.env.VITE_SUPABASE_ANON_KEY);
let pass = 0, fail = 0;
const ok = (name, cond, detail = '') => {
  if (cond) { pass++; console.log(`  PASS  ${name}`); }
  else { fail++; console.log(`  FAIL  ${name} ${detail}`); }
};

// --- B. Formatting logic (the exact expressions the panel will use) ---
const fmt$ = n => n < 0.01 ? `<$0.01` : `$${n.toLocaleString(undefined,{minimumFractionDigits:2,maximumFractionDigits:2})}`;
ok('comma: 21835 -> "21,835"', (21835).toLocaleString() === '21,835');
ok('comma: 999 stays "999"', (999).toLocaleString() === '999');
ok('fmt$: 119.874 -> "$119.87"', fmt$(119.874) === '$119.87');
ok('fmt$: 1234.5 -> "$1,234.50"', fmt$(1234.5) === '$1,234.50');
ok('fmt$: 0.004 -> "<$0.01"', fmt$(0.004) === '<$0.01');

// --- F. Un-filtered paged read matches a direct count, and exceeds global-only ---
const PAGE_SIZE = 1000;
async function pagedCount(tenantId) {
  let n = 0, from = 0;
  while (true) {
    let q = supabase.from('ai_activity_log').select('id').order('created_at', { ascending: false }).range(from, from + PAGE_SIZE - 1);
    if (tenantId) q = q.eq('tenant_id', tenantId);
    const { data, error } = await q;
    if (error) throw new Error('paged read failed: ' + error.message);
    n += (data || []).length;
    if (!data || data.length < PAGE_SIZE) break;
    from += PAGE_SIZE;
  }
  return n;
}
const allRows = await pagedCount(null);
const globalRows = await pagedCount('global');
const { count: directAll, error: cErr } = await supabase.from('ai_activity_log').select('id', { count: 'exact', head: true });
ok('direct count query works', !cErr, cErr?.message);
ok(`paged all-tenant read (${allRows}) === direct count (${directAll})`, allRows === directAll);
ok(`all-tenant (${allRows}) > global-only (${globalRows})`, allRows > globalRows);

// --- F. Reconciliation identity: distinct classified + reclassification = total rows ---
const matchedIds = new Set();
{
  let from = 0;
  while (true) {
    const { data, error } = await supabase.from('ai_call_patterns').select('ai_activity_log_id').range(from, from + PAGE_SIZE - 1);
    if (error) throw new Error('ai_call_patterns read failed: ' + error.message);
    (data || []).forEach(r => matchedIds.add(r.ai_activity_log_id));
    if (!data || data.length < PAGE_SIZE) break;
    from += PAGE_SIZE;
  }
}
const { data: recl, error: rErr } = await supabase.from('ai_pattern_reclassification_count').select('reclassification_count').single();
ok('reclassification view readable', !rErr, rErr?.message);
ok(`identity: distinct classified (${matchedIds.size}) + reclassification (${recl?.reclassification_count}) === total rows (${allRows})`,
   matchedIds.size + (recl?.reclassification_count ?? -1) === allRows);

// --- B. Tile source agreement: rollup pattern count is what the tile will show ---
const { data: rollup, error: roErr } = await supabase.from('ai_pattern_classification_rollup').select('pattern_slug');
ok('rollup view readable', !roErr, roErr?.message);
ok(`rollup pattern count (${rollup?.length}) >= 1`, (rollup?.length ?? 0) >= 1);

console.log(fail === 0 ? `\nALL TESTS PASS (${pass})` : `\n${fail} FAILURES (${pass} passed)`);
if (fail > 0) process.exit(1);
```
