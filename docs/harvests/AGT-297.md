```sql
 note text, updated_at timestamptz not null default now());

returns table(caller_name text, permission text, spend_limit_usd numeric, llm_calls bigint, total_tokens bigint, cost_usd numeric)
language sql stable set search_path = public, pg_catalog as $f$

 left join public.mcp_caller_gates g on g.caller_name = s.name

  select count(*)::bigint n, coalesce(sum(i + o + cw + cr),0)::bigint t,
   coalesce(sum((i/1000.0) * p.input_per_1k + (cw/1000.0) * p.input_per_1k * 1.25
    + (cr/1000.0) * p.input_per_1k * 0.10 + (o/1000.0) * p.output_per_1k),0)::numeric c
  from (select model, coalesce(input_tokens,0) i, coalesce(output_tokens,0) o,
```

## A

```js
const MCP_KEY_HEADER = 'x-deepbench-mcp-key';

function isMcpRequest(url) {
  const p = url.pathname;
  if (p === '/api/mcp' || p.startsWith('/api/mcp/')) return true;
  return p === '/api/capabilities/execute' && url.searchParams.get('transport') === 'mcp';
}

// The matched key's gate row, or null for an unknown key. Throws on failure.
async function fetchCallerGate(url, key, presented) {
  const d = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(presented));
  const hex = Array.from(new Uint8Array(d), b => b.toString(16).padStart(2, '0')).join('');
  const res = await fetch(`${url}/rest/v1/rpc/mcp_caller_gate`, {
    method: 'POST',
    headers: { ...supabaseAuth(key), 'Content-Type': 'application/json' },
    body: JSON.stringify({ p_key_sha256: hex }),
    signal: timeoutSignal(SUPABASE_TIMEOUT_MS),
  });
  if (!res.ok) throw new Error(`status ${res.status}`);
  const rows = await res.json();
  return Array.isArray(rows) && rows[0] && rows[0].caller_name ? rows[0] : null;
}
```

## 2. The design in one sentence, and my calls

A request to an MCP route that presents a KNOWN key is gated on that key's own permission, cap and ledger and its address is never consulted; no key, an unknown key or a failed lookup all fall through to today's address gate, unchanged.

Governing: `docs/ARCHITECTURE.md` §19t (IP Access Gate), §19k (the `visitor_id` identity slot), §19v (migration down capture); `.claude/rules/supabase-column-grants.md` (a new table is born readable by `anon`; a new function is born callable by `anon`, `authenticated` and `service_role` by name).

- D1. The key is hashed at the edge (SHA-256) and matched inside Postgres, so no secret value reaches the edge runtime. Standard hashed-key lookup, not a homegrown scheme (pattern:101).
- D2. MCP routes only (`/api/mcp`, `/api/mcp/<x>`, and the rewrite target `/api/capabilities/execute?transport=mcp`). No other route writes the key's name on its rows, so honouring the key elsewhere would be a cap with an empty ledger -- a way around the gate. Test arm 5 holds it.
- D3. A keyed caller skips the country check: the country header describes the egress it shares, not the caller.
- D4. Per-key permission and cap live in a NEW table, `mcp_caller_gates`; an absent row is `trial` / $5.00. Rejected: two columns on `runner_secrets` (an in-place ALTER, which `capture_migration_down()` refuses, making the down card-only); `known_callers` (`anon` holds table-level SELECT on it); `ip_org_cache` rows keyed by a key name (`ip_spend_report` and the block alert read that table as addresses).
- D5. `MCP_API_KEY` is seeded `unlimited`: it is the platform's own caller (1,016 rows), and a $5 wall on the governance lane would be a self-inflicted outage.
- D6. A lookup failure is not a pass. It logs one line and the address gate still runs -- closed on a blocked address, open on a trial one. Test arm 4 holds both.
- Also: an index `ai_activity_visitor_id_idx (visitor_id) where visitor_id is not null`, mirroring `ai_activity_caller_ip_idx`. Without it the designed SELECT is a sequential scan of about 69,000 rows (21 ms measured at the first assembly); with it, three lookups took 3 ms in total. Spend is the rows `mcpAttribution()` already wrote under the key's NAME, priced with the same expression as `get_ip_stats`.

Risk (John's rule, decision 6e94edcb): nothing here is irreversible and nothing changes an agent's autonomy, so no part must ship alone. T1 is three new objects, each dropped by identity; T2-T4 are `git revert`.

## 6. What stays open after this ships -- for tickets, not this build

- **The ordinary connector user is still keyless.** The only key a caller can present is the `x-deepbench-mcp-key` request header, and most people who add a DeepBench agent to Claude as a custom connector cannot send one (request-header authentication is described as a limited beta; web search 2026-11-11, not verifiable from here). After this ticket a script, Claude Code (`--header`) or a connector in an organization with that beta is counted and capped as itself; an ordinary connector user still arrives keyless from the shared pool, attributed to no one. Closing that needs the per-caller signal to ride in the address and a Connect page that hands each person their own -- a visible change, a new ticket with its own design.
- No key exists for the "outside tester" of `q-agt-163-tester-key`; the interviewquestions client appears (inferred from dates and user agent) to be calling under the governance key.
- A keyed caller refused at its cap writes no block row and raises no alert; its spend also still counts on the ledger of whatever address it came from (`get_ip_stats` is untouched).
- `api/_lib/mcp.js` line 144 and `docs/runbooks/mcp-server.md` line 40 still say "per-IP", which stays true for a keyless caller.

```js
// DeepBench v7.0.1001 | tests/regression/agt-297-mcp-caller-gate.test.mjs | AGT-297
import assert from "node:assert/strict";
import { createHash } from "node:crypto";
import { selfRun } from "./_lib/self-run.js";
import middleware from "../../middleware.js";

const IP = "160.79.106.1";
const sha = s => createHash("sha256").update(s).digest("hex");
const CALLERS = {
  [sha("key-a")]: { caller_name: "MCP_KEY_a", permission: "trial", spend_limit_usd: 5, llm_calls: 12, total_tokens: 3400, cost_usd: 6.5 },
  [sha("key-b")]: { caller_name: "MCP_KEY_b", permission: "trial", spend_limit_usd: 5, llm_calls: 1, total_tokens: 90, cost_usd: 0.25 },
  [sha("key-c")]: { caller_name: "MCP_KEY_c", permission: "unlimited", spend_limit_usd: 5, llm_calls: 9, total_tokens: 9, cost_usd: 999 },
  [sha("key-d")]: { caller_name: "MCP_KEY_d", permission: "blocked", spend_limit_usd: 5, llm_calls: 0, total_tokens: 0, cost_usd: 0 },
};
const BLOCKED_IP = { permission: "blocked", block_reason: "spend_cap", spend_limit_usd: 5, blocked_attempts: 1, blocked_at: "2026-11-01T00:00:00Z", country: "US" };
const TRIAL_IP = { ...BLOCKED_IP, permission: "trial", block_reason: null };

async function call(path, key, { ipRow = BLOCKED_IP, gateStatus = 200 } = {}) {
  const seen = [];
  const json = (body, status = 200) => new Response(JSON.stringify(body), { status, headers: { "Content-Type": "application/json" } });
  globalThis.fetch = async (url, init = {}) => {
    const u = String(url);
    seen.push(u + " " + JSON.stringify(init.headers || {}) + " " + String(init.body || ""));
    if (u.includes("/rpc/mcp_caller_gate")) {
      if (gateStatus !== 200) return json({ message: "down" }, gateStatus);
      const row = CALLERS[JSON.parse(init.body).p_key_sha256];
      return json(row ? [row] : []);
    }
    if (u.includes("/rpc/get_ip_stats")) return json([{ llm_calls: 40, total_tokens: 1000, cost_usd: 1 }]);
    if (u.includes("/ip_org_cache") && (!init.method || init.method === "GET")) return json([ipRow]);
    return json({}, 201);
  };
  const headers = { "x-forwarded-for": IP, "x-vercel-ip-country": "US" };
  if (key) headers["x-deepbench-mcp-key"] = key;
  const res = await middleware(new Request("https://deepbench.test" + path, { method: "POST", headers }), {});
  const n = part => seen.filter(s => s.includes(part)).length;
  return { status: res.status, next: res.headers.get("x-middleware-next"), body: res.status === 403 ? await res.json() : null, seen, gate: n("/rpc/mcp_caller_gate"), ip: n("/ip_org_cache") + n("/rpc/get_ip_stats") > 0 };
}

export default async function run() {
  const saved = { fetch: globalThis.fetch, url: process.env.SUPABASE_URL, key: process.env.SUPABASE_SERVICE_KEY, bypass: process.env.GATE_BYPASS_SECRET, error: console.error };
  process.env.SUPABASE_URL = "https://sb.test";
  process.env.SUPABASE_SERVICE_KEY = "service-test";
  delete process.env.GATE_BYPASS_SECRET;
  console.error = () => {};
  try {
    // 1. TWO CALLERS, ONE EGRESS IP, and that IP's own row is already blocked.
    const a = await call("/api/mcp/harlan", "key-a");
    assert.equal(a.status, 403);
    assert.deepEqual(a.body, { deepbench_gate: true, reason: "spend_cap", ip: IP, stats: { llm_calls: 12, total_tokens: 3400, cost_usd: 6.5 } });
    const b = await call("/api/mcp/harlan", "key-b");
    assert.equal(b.status, 200);
    assert.equal(b.next, "1");
    assert.deepEqual([a.gate, a.ip, b.gate, b.ip], [1, false, 1, false]);
    // 2. The key leaves the edge only as its SHA-256.
    assert.equal(a.seen.some(s => s.includes("key-a")), false);
    assert.equal(a.seen.filter(s => s.includes(sha("key-a"))).length, 1);
    // 3. NO KEY: the IP gate, as before.
    const none = await call("/api/mcp/harlan", null);
    assert.deepEqual([none.status, none.body.reason, none.body.stats.llm_calls, none.gate], [403, "spend_cap", 40, 0]);
    // 4. UNKNOWN KEY and a FAILED LOOKUP both fall to the IP gate -- closed on a blocked IP, open on a trial one.
    const unknown = await call("/api/mcp", "key-zzz");
    assert.deepEqual([unknown.status, unknown.gate, unknown.ip], [403, 1, true]);
    const down = await call("/api/mcp", "key-b", { gateStatus: 500 });
    assert.deepEqual([down.status, down.gate], [403, 1]);
    const downTrial = await call("/api/mcp", "key-a", { gateStatus: 500, ipRow: TRIAL_IP });
    assert.deepEqual([downTrial.status, downTrial.next, downTrial.ip], [200, "1", true]);
    // 5. MCP ROUTES ONLY.
    const other = await call("/api/capabilities/execute", "key-b");
    assert.deepEqual([other.status, other.gate], [403, 0]);
    const rewritten = await call("/api/capabilities/execute?transport=mcp", "key-b");
    assert.deepEqual([rewritten.status, rewritten.gate, rewritten.ip], [200, 1, false]);
    // 6. The row's permission is obeyed.
    const c = await call("/api/mcp", "key-c");
    assert.deepEqual([c.status, c.next], [200, "1"]);
    const d = await call("/api/mcp", "key-d");
    assert.deepEqual(d.body, { deepbench_gate: true, reason: "manual", ip: IP });
  } finally {
    globalThis.fetch = saved.fetch;
    console.error = saved.error;
    for (const [k, v] of [["SUPABASE_URL", saved.url], ["SUPABASE_SERVICE_KEY", saved.key], ["GATE_BYPASS_SECRET", saved.bypass]]) {
      if (v === undefined) delete process.env[k]; else process.env[k] = v;
    }
  }
}

selfRun(import.meta.url, run);
```
