## 3. The design and what it does not do

One new read, `commitFilesFor(repoRoot, sha)` = `git diff --name-only <sha>^ <sha>`, taken beside the graded sha the verifier already reads, and handed to the pin check as a second list. The pin check passes a test that is in either list.

- **D1 -- pin check only.** The ticket's wording ("derive the changed files from the graded sha's own parent") would also feed the self-certification and auto-done checks. I did not do that. A last-commit list can be a part of a delivery (a delivery spread over two commits), and for the self-certification check a shorter list is the lenient direction: it would let attended runs that fail closed today grade as clean. That is a change in what a class may auto-complete, which the risk rule says ships as its own ticket. For the pin check the direction is the safe one: a test that is in the graded commit was changed by the delivery, so the extra list can only clear a true re-pin.
- **D2 -- an unreadable pair still blocks.** When neither list is readable (no git, or a depth-1 clone with no parent) the finding is `pin-not-graded`, worded as not graded and saying no re-pin is shown missing. It still blocks: a degrade does not remove a stop (pattern:166). `agt-328` arm E keeps passing unedited because it asserts the test list and the words `fail closed`, both kept.
- **D3 -- no runbook edit.** The ticket offers code or the runbook, "one of the two, in code". With the code fix the sanctioned order is safe as written, and leaving `session-setup.md` alone keeps the file list at three.
- **D4 -- the ladder reset is not reversed.** `runner_before_images` for cycle 84536354 holds bug_fix at rung 17, streak 9 before verdict 0d74df34 reset it. The row now reads rung 30, streak 68. Restoring the before-image would lower the class, and adding 9 to the current streak would invent signals that did not happen. Reversible: the before-image is still there.
- **D5 -- re-grade through the existing lane.** `public.stale_blocked_verdicts(<current fingerprint>)` already lists AGT-770 and AGT-781 (67 rows in the set, oldest first, 3 a cycle, so they would wait about 22 cycles). T4 names them with `--ticket`. That lane grades the three gates through `gradeDelivery()`; it does not run the pin check. The pin claim itself is re-graded by arm c of the new test, which reads the three real kickoffs against their commits' tests.

Not done: a verifier run after a later commit has landed on HEAD reads that later commit, not the build commit. All three measured cases graded the build commit itself.

Alternatives set aside: passing `--base=<push sha>^` from the runbook (leaves every other caller of the default exposed, and the ticket asks for code); folding the commit list into `changedFilesFor()` (one list everywhere, the D1 autonomy change); making `pin-not-graded` a non-block (removes a stop).

```js
// DeepBench v7.0.1116 | tests/regression/agt-791-pin-reads-graded-commit.test.mjs | AGT-791
import assert from "node:assert";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";
import { spawnSync } from "node:child_process";
import { fileURLToPath } from "node:url";
import { selfRun } from "./_lib/self-run.js";
import { commitFilesFor, pinsShipFinding, resolveDeliveryFiles } from "../../scripts/verifier.js";

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "../..");
const H = "tests/regression/holds.test.mjs";
const K = "<!-- DeepBench v7.0.1116 | kickoff | FIXTURE -->\n```pins\n" + H + " | repin | T2\n```\n";
const T = "tests/regression/";
const CASES = [
  ["docs/kickoffs/v7.0.1095-AGT-523-measured-model-params.md", ["agt-70-auditor.test.mjs", "agt-90-fable-temperature.test.mjs"]],
  ["docs/kickoffs/v7.0.1096-AGT-770-site-key-gate.md", ["agt-297-mcp-caller-gate.test.mjs"]],
  ["docs/kickoffs/v7.0.1106-AGT-781-company-knowledge-records.md", ["agt-542-dilbert.test.mjs", "agt-643-harry.test.mjs", "agt-721-laura.test.mjs"]],
].map(([k, t]) => [k, t.map(n => T + n), t.length]);

async function run() {
  // a
  assert.strictEqual(pinsShipFinding({ text: K, changedFiles: null, commitFiles: ["a.js", H] }), null);
  assert.strictEqual(pinsShipFinding({ text: K, changedFiles: ["a.js"], commitFiles: ["./" + H] }), null);
  const left = pinsShipFinding({ text: K, changedFiles: null, commitFiles: ["a.js"] });
  assert.strictEqual(left.kind, "pin-left-behind");
  assert.ok(!left.reason.includes("could not be read"), left.reason);
  for (const args of [{ text: K, changedFiles: null, commitFiles: null }, { text: K, changedFiles: null }]) {
    const ng = pinsShipFinding(args);
    assert.strictEqual(ng.kind, "pin-not-graded");
    assert.deepStrictEqual(ng.tests, [H]);
    assert.ok(ng.reason.startsWith("pin claim NOT GRADED (AGT-791)") && ng.reason.includes("fail closed"), ng.reason);
    assert.ok(!ng.reason.includes("pin left behind"), ng.reason);
  }
  // b
  const R = fs.mkdtempSync(path.join(os.tmpdir(), "agt791-"));
  try {
    const git = (...a) => {
      const r = spawnSync("git", ["-c", "user.name=t", "-c", "user.email=t@t.invalid", "-c", "commit.gpgsign=false", ...a], { cwd: R, encoding: "utf8" });
      assert.strictEqual(r.status, 0, `git ${a.join(" ")}: ${r.stderr}`);
      return String(r.stdout).trim();
    };
    const put = (p, t) => { fs.mkdirSync(path.dirname(path.join(R, p)), { recursive: true }); fs.writeFileSync(path.join(R, p), t); };
    git("init", "-q");
    put("a.js", "1"); git("add", "-A"); git("commit", "-q", "-m", "base");
    const base = git("rev-parse", "HEAD");
    put("a.js", "2"); put(H, "x"); git("add", "-A"); git("commit", "-q", "-m", "ship");
    const ship = git("rev-parse", "HEAD");
    const afterPush = resolveDeliveryFiles({ explicit: null, gitList: [] });
    assert.strictEqual(afterPush, null);
    assert.deepStrictEqual(commitFilesFor(R, ship), ["a.js", H]);
    assert.strictEqual(pinsShipFinding({ text: K, changedFiles: afterPush, commitFiles: commitFilesFor(R, ship) }), null);
    assert.strictEqual(commitFilesFor(R, base), null);
    assert.strictEqual(commitFilesFor(R, "HEAD"), null);
    assert.strictEqual(commitFilesFor(os.tmpdir(), ship), null);
  } finally { fs.rmSync(R, { recursive: true, force: true }); }
  // c
  for (const [k, files, n] of CASES) {
    const text = fs.readFileSync(path.join(ROOT, k), "utf8");
    assert.strictEqual(pinsShipFinding({ text, changedFiles: null, commitFiles: files }), null, k);
    const ng = pinsShipFinding({ text, changedFiles: null });
    assert.strictEqual(ng.kind, "pin-not-graded", k);
    assert.strictEqual(ng.tests.length, n, k);
  }
}
export default run;
selfRun(import.meta.url, run);
```
