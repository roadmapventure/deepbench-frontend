## Governing sections and rules

## Migration text

The content of `docs/design/agt-570-own-row-exemption.sql` after its header line; apply as migration `agt570_own_row_exemption` after `capture_migration_down`. Replace `:C` with the cycle id (three places).

```sql
do $up$ declare v text; k text; n int; a int; b int; begin
  v := pg_get_functiondef('public.runner_should_boot()'::regprocedure);
  v := replace(v, E'public.runner_should_boot()\n RETURNS TABLE', E'public.runner_should_boot(p_cycle_id uuid DEFAULT NULL)\n RETURNS TABLE');
  k := E'   WHERE rc.ended_at IS NULL\n';
  v := replace(v, k, k || E'     AND (p_cycle_id IS NULL OR rc.id <> p_cycle_id)  -- AGT-570: the asker is not a peer\n');
  drop function public.runner_should_boot();
  execute v;
  revoke all on function public.runner_should_boot(uuid) from public, anon, authenticated;
  grant execute on function public.runner_should_boot(uuid) to service_role;
  v := pg_get_functiondef('public.record_leverage(uuid, jsonb)'::regprocedure);
  v := replace(v, 'IF NOT (v_id = ANY (COALESCE(v_queue, ARRAY[]::text[]))) THEN', 'IF NOT (v_id = ANY (COALESCE(v_queue, ARRAY[]::text[])) OR EXISTS (SELECT 1 FROM public.backlog_items b WHERE b.backlog_id = v_id AND b.claimed_by = p_cycle_id::text AND b.claimed_at >= now() - make_interval(hours => (SELECT s.claim_stale_hours FROM public.runner_settings s WHERE s.id = 1)))) THEN');
  execute v;
  select count(*) into n from pg_proc p join pg_namespace s on s.oid = p.pronamespace where s.nspname = 'public' and p.proname = 'runner_should_boot';
  if n <> 1 then raise exception 'agt570: % overloads', n; end if;
  if has_function_privilege('anon', 'public.runner_should_boot(uuid)', 'EXECUTE') or has_function_privilege('authenticated', 'public.runner_should_boot(uuid)', 'EXECUTE') or not has_function_privilege('service_role', 'public.runner_should_boot(uuid)', 'EXECUTE') or has_function_privilege('anon', 'public.record_leverage(uuid, jsonb)', 'EXECUTE') then raise exception 'agt570: grants'; end if;
  select (detail ->> 'live_lanes')::int into a from public.runner_should_boot();
  select (detail ->> 'live_lanes')::int into b from public.runner_should_boot(':C');
  if a - b <> 1 then raise exception 'agt570: live_lanes % without, % with :C (heartbeat?)', a, b; end if;
  begin
    update public.backlog_items set claimed_by = ':C', claimed_at = now() where backlog_id = 'AGT-570' and (claimed_by is null or claimed_by = ':C');
    if not found then raise exception 'agt570: AGT-570 held elsewhere'; end if;
    perform public.record_leverage(':C', '[{"backlog_id":"AGT-570","improves":"probe","why":"rolled back"}]'::jsonb);
    raise exception using errcode = 'P0570', message = 'rollback';
  exception when sqlstate 'P0570' then null; end;
  begin
    perform public.record_leverage(gen_random_uuid(), '[{"backlog_id":"AGT-570","improves":"probe","why":"refused"}]'::jsonb);
    raise exception 'agt570: stranger admitted';
  exception when others then if sqlerrm not like '%is not a ref of prime_directive_queue()%' then raise; end if; end;
end $up$;
```
