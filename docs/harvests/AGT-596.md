A writer who types the phrase has seen the ship and is arguing about what remains.

-- DeepBench v7.0.1100 | docs/harvests/AGT-596.md `## T1 up B` | AGT-596, migration `agt596_removal_guard`
-- A TICKET ITS OWN KICKOFF SHIPPED CANNOT BE WRITTEN `removal proposed` WITHOUT A BASIS.
-- Three functions and one trigger; nothing is dropped. The undo is the `-- down` at the foot: the
-- trigger function replaced by a pass-through, so the trigger stays and does nothing.
-- "Shipped" is read from the ledger, never from git: a `shipped` runner_cycles row holding a
-- push_sha, or a runner_verdicts row, for the version the row's own kickoff_link names.
-- The basis is one of three (pattern:171): a ruling, a superseding ticket, or evidence.

create or replace function public.removal_basis(
  p_backlog_id text, p_version text, p_session_name text, p_kind text, p_reasoning text)
returns text
language plpgsql
stable
set search_path = public
as $fn$
declare
  v_sup text;
begin
  -- RULING: a named session decided it, or the decision is a ruling by kind.
  if p_session_name is not null and btrim(p_session_name) <> '' then return 'ruling'; end if;
  if p_kind in ('gate', 'ticket-validation', 'john-ruling', 'design-ruling') then return 'ruling'; end if;
  -- SUPERSEDED: the reasoning names another ticket that is itself still standing.
  v_sup := substring(coalesce(p_reasoning, '') from 'superseded-by ([A-Z]+-[0-9]+[a-z]?)');
  if v_sup is not null and v_sup <> p_backlog_id and exists (
       select 1 from public.backlog_items s
        where s.backlog_id = v_sup and s.status not in ('removed', 'removal proposed')) then
    return 'superseded';
  end if;
  -- EVIDENCE: the writer says so in these words AND names the ship it has seen. The version alone
  -- is not a basis: "closed by v7.0.942" is the defect's own sentence.
  if p_version is not null
     and position('removal-basis: evidence' in coalesce(p_reasoning, '')) > 0
     and p_reasoning ~ (replace(p_version, '.', '\.') || '([^0-9]|$)') then
    return 'evidence';
  end if;
  return null;
end
$fn$;

create or replace function public.removal_refusal(p_backlog_id text)
returns text
language plpgsql
stable
set search_path = public
as $fn$
declare
  v_link text;
  v_ver  text;
  v_ship text;
  v_dec  record;
begin
  select b.kickoff_link into v_link from public.backlog_items b where b.backlog_id = p_backlog_id;
  if v_link is null then return null; end if;
  v_ver := substring(v_link from 'v[0-9]+\.[0-9]+\.[0-9]+');
  if v_ver is null then return null; end if;

  select coalesce(
    (select 'cycle ' || c.id::text from public.runner_cycles c
      where c.item_id = p_backlog_id and c.version = v_ver
        and c.outcome = 'shipped' and c.push_sha is not null
      order by c.started_at desc limit 1),
    (select 'verdict ' || v.id::text from public.runner_verdicts v
      where v.backlog_id = p_backlog_id and v.version = v_ver
      order by v.created_at desc limit 1)) into v_ship;
  if v_ship is null then return null; end if;

  -- A ruling recorded in THIS transaction (apply_gate_rulings, apply_ticket_validation): the
  -- ruling and the status write are one act. decided_at defaults to now(), the transaction's start.
  if exists (select 1 from public.runner_decisions d
              where d.decided_at = now() and d.status <> 'reversed'
                and d.kind in ('gate', 'ticket-validation', 'john-ruling', 'design-ruling')) then
    return null;
  end if;

  -- Otherwise the newest standing decision on this ticket, at most an hour old, carries the basis.
  for v_dec in
    select d.session_name, d.kind, d.reasoning from public.runner_decisions d
     where d.backlog_id = p_backlog_id and d.status <> 'reversed'
       and d.decided_at > now() - interval '60 minutes'
     order by d.decided_at desc limit 1
  loop
    if public.removal_basis(p_backlog_id, v_ver, v_dec.session_name, v_dec.kind, v_dec.reasoning) is not null then
      return null;
    end if;
  end loop;

  return format('AGT-596: %s cannot go to removal proposed -- its own kickoff shipped as %s (%s). If that ship closed the gap the premise is delivered, not dead: verify it and settle it. To remove what remains, first record a decision on %s that is a ruling, or whose reasoning carries superseded-by <ID>, or removal-basis: evidence and %s.',
    p_backlog_id, v_ver, v_ship, p_backlog_id, v_ver);
end
$fn$;

create or replace function public.backlog_removal_guard()
returns trigger
language plpgsql
set search_path = public
as $fn$
declare
  v_why text;
begin
  if new.status = 'removal proposed' and old.status in ('open', 'partial', 'delivered', 'done') then
    v_why := public.removal_refusal(new.backlog_id);
    if v_why is not null then
      raise exception '%', v_why using errcode = 'P0001';
    end if;
  end if;
  return new;
end
$fn$;

  before update of status on public.backlog_items
